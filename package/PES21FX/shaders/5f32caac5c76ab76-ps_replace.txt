void main(
  float2 v0 : TEXCOORD0,
  float4 v1 : SV_Position0,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2,r3,r4;
  uint4 bitmask, uiDest;
  float4 fDest;

  float4 P1 = IniParams.Load(int2(1,0));
  float4 P2 = IniParams.Load(int2(2,0));
  bool  grade    = P1.w > 0.5;
  float exposure = (grade && P1.x > 0) ? P1.x : 1.0;
  float sat      = (grade && P1.y > 0) ? P1.y : 1.0;
  float contrast = grade ? P1.z : 0.0;
  float bloomInt = (grade && P2.z > 0) ? P2.z : 1.0;
  float shadows  = grade ? P2.w : 0.0;

  r0.xy = g_psSystem.m_renderBuffer.zw * g_psObject.m_localParam[1].yy;
  r0.xy = float2(8,8) * r0.xy;
  r0.zw = v1.xy * g_psSystem.m_renderBuffer.zw + r0.xy;
  r0.xy = v1.xy * g_psSystem.m_renderBuffer.zw + -r0.xy;
  r0.xy = max(float2(0,0), r0.xy);
  r0.xy = min(g_psObject.m_localParam[2].zw, r0.xy);
  r1.z = inBloom.Sample(g_samplerLinear_Clamp_s, r0.xy).z;
  r0.xy = max(float2(0,0), r0.zw);
  r0.xy = min(g_psObject.m_localParam[2].zw, r0.xy);
  r1.x = inBloom.Sample(g_samplerLinear_Clamp_s, r0.xy).x;
  r0.xy = g_psSystem.m_renderBuffer.zw * v1.xy;
  r0.zw = max(float2(0,0), r0.xy);
  r2.xyzw = inImage.Sample(g_samplerPoint_Wrap_s, r0.xy).xyzw;
  r0.xy = min(g_psObject.m_localParam[2].zw, r0.zw);
  r1.y = inBloom.Sample(g_samplerLinear_Clamp_s, r0.xy).y;
  {
    // PES21: el bloom se "derrama" en los bordes de la pantalla (halo naranja): se atenua cerca del borde
    float2 uvE = v1.xy * g_psSystem.m_renderBuffer.zw;
    float edgeM = smoothstep(0.0, 0.07, uvE.x) * smoothstep(0.0, 0.07, 1.0 - uvE.x) * smoothstep(0.0, 0.09, uvE.y) * smoothstep(0.0, 0.09, 1.0 - uvE.y);
    r1.xyz *= lerp(0.2, 1.0, edgeM);
  }
  r0.xyz = (r2.xyz + r1.xyz * bloomInt) * exposure;
  float ecK = IniParams.Load(int2(128,0)).x; ecK = ecK > 0.05 ? ecK : 1.0; r0.xyz *= ecK;   // PES21 (20/10) exposicion estable entre camaras (la calcula la DLL con el medidor de luz)
  float expoK = exposure * ecK;   // (21/10) factor de exposicion total del tonemap (para el cielo fisico)
  // === PES21 v2: look moderno (x6 luces altas suaves, y6 tono partido, z6 claridad, w6 viñeta) ===
  float4 P6 = IniParams.Load(int2(6,0));
  float3 colS_g = max(pL_color.xyz, 0.0); colS_g = colS_g / max(dot(colS_g, float3(0.2126, 0.7152, 0.0722)), 1e-3);   // (20/10) color del sol normalizado (perspectiva aerea)
  float4 P7 = IniParams.Load(int2(7,0));
  float skyK = grade ? saturate(P7.y) : 0.0;
  float skyM = 0;
  float4 PH = IniParams.Load(int2(149,0));   // (21/10) cielo fisico: x on | y brillo | z cuanto | w conservar las nubes del estadio
  bool physOn = PH.x > 0.5;
  if (skyK > 0 || physOn) {
    float2 uvS = v1.xy * g_psSystem.m_renderBuffer.zw;
    float2 pxS = g_psSystem.m_renderBuffer.zw * 1.5;
    skyM += PESDepth.SampleLevel(g_samplerLinear_Clamp_s, uvS, 0).x < 1e-6 ? 1.0 : 0.0;
    skyM += PESDepth.SampleLevel(g_samplerLinear_Clamp_s, uvS + float2(pxS.x, 0), 0).x < 1e-6 ? 1.0 : 0.0;
    skyM += PESDepth.SampleLevel(g_samplerLinear_Clamp_s, uvS - float2(pxS.x, 0), 0).x < 1e-6 ? 1.0 : 0.0;
    skyM += PESDepth.SampleLevel(g_samplerLinear_Clamp_s, uvS + float2(0, pxS.y), 0).x < 1e-6 ? 1.0 : 0.0;
    skyM += PESDepth.SampleLevel(g_samplerLinear_Clamp_s, uvS - float2(0, pxS.y), 0).x < 1e-6 ? 1.0 : 0.0;
    skyM *= 0.2;
    float skyRaw = skyM;   // mascara de cielo (0..1)
    skyM *= skyK;
    // PES21 cielo v2 (Claude, 16/10): antes el cielo se multiplicaba por 0.55 y quedaba azul marino a mediodia.
    // Ahora: compresion suave de luces (luminoso sin quemar), zenit azul, horizonte claro con bruma.
    // x28 brillo (0.8; 1 = sin bajar), y28 azul del zenit (0..1), z28 contraste de nubes, w28 bruma del horizonte
    float4 SK = IniParams.Load(int2(28,0));
    // === PES21 cielo fisico (21/10 b): el COLOR del cielo despejado sale de un modelo de atmosfera (PESSkyPhys,
    // pes_common.hlsli) con el sol real: azul profundo arriba, horizonte mas claro, al atardecer horizonte tibio y cenit mas
    // azul. El BRILLO sigue siendo el del cielo del estadio (el modelo solo lo modula un poco, con un gradiente fijo en el
    // mundo): antes el brillo absoluto del modelo salia mucho mas oscuro que el del estadio y, para emparejar las nubes
    // pintadas, se usaba una grilla de puntos de pantalla que cambiaba al mover la camara (el cielo "se movia" y quedaba
    // azul marino con nubes grises). Las nubes pintadas (lo que no es azul) no se tocan. 149: x on | y brillo | z cuanto
    float physW = 0.0;
    if (physOn && skyRaw > 0.01 && IniParams.Load(int2(114,0)).w > 0.5 && dot(pL_dir.xyz, pL_dir.xyz) > 1e-6 && abs(pS_projParam.x) > 1e-6 && abs(pS_projParam.y) > 1e-6) {
      const float3 LWp = float3(0.2126, 0.7152, 0.0722);
      float4 Q86p = IniParams.Load(int2(86,0));
      float3 vXp = float3(pS_mat[2]._m00, pS_mat[2]._m01, pS_mat[2]._m02);
      float3 vYp = float3(pS_mat[2]._m10, pS_mat[2]._m11, pS_mat[2]._m12);
      float3 vZp = float3(pS_mat[2]._m20, pS_mat[2]._m21, pS_mat[2]._m22);
      float3 Lnp = normalize(pL_dir.xyz);
      float3 sunWp = normalize(float3(dot(Lnp, vXp), dot(Lnp, vYp), dot(Lnp, vZp)));
      float kEl = smoothstep(0.02, 0.10, sunWp.y);
      if (kEl > 0.001) {
        float4 ATp = IniParams.Load(int2(121,0));
        float turbP = ATp.y > 0.5 ? ATp.y : 2.5;
        float2 ndcP = float2(uvS.x * 2.0 - 1.0, (Q86p.y > 0.5 ? 1.0 : -1.0) * (uvS.y * 2.0 - 1.0));
        float3 dvP = normalize(float3((ndcP - pS_centerOff.xy) * pS_projParam.xy, Q86p.z > 0.5 ? -1.0 : 1.0));
        float3 dWp = normalize(float3(dot(dvP, vXp), dot(dvP, vYp), dot(dvP, vZp)));
        float3 phys = PESSkyPhys(dWp, sunWp, turbP, 12);
        float lP = max(dot(phys, LWp), 1e-6);
        // referencia fija en el mundo: 35 grados de altura, del lado opuesto al sol (el cielo "medio")
        float2 sh = normalize(sunWp.xz + float2(1e-4, 0.0));
        float3 dRef = normalize(float3(-sh.x * 0.819, 0.574, -sh.y * 0.819));
        float lRef = max(dot(PESSkyPhys(dRef, sunWp, turbP, 8), LWp), 1e-6);
        // modulacion del brillo acotada: el horizonte un poco mas claro y la zona del sol mas luminosa, sin quemar
        float gK = clamp(pow(lP / lRef, 0.5), 0.7, 1.8) * (PH.y > 0 ? PH.y : 1.0);
        float3 gIn = r0.xyz; float lumG = dot(gIn, LWp);
        float clr = smoothstep(0.30, 0.80, saturate((gIn.b - gIn.r) / max(lumG, 1e-4) * 3.0));   // 1 = cielo azul, 0 = nube / cielo cubierto
        float3 tgt = (phys / lP) * lumG * gK;
        physW = saturate(PH.z) * kEl;
        r0.xyz = lerp(r0.xyz, tgt, skyRaw * physW * lerp(1.0 - saturate(PH.w), 1.0, clr));
        physW *= clr;
      }
    }
    float topH = saturate(1.0 - uvS.y * 1.6);
    float3 skyIn = r0.xyz;
    float lumS = dot(skyIn, float3(0.2126, 0.7152, 0.0722));
    float brightS = SK.x > 0 ? SK.x : 0.8;
    float kneeS = 1.2;
    float lumC = lumS <= kneeS ? lumS : kneeS + (lumS - kneeS) / (1.0 + (lumS - kneeS) / (kneeS * 1.5));
    float3 cS = skyIn * (lumC / max(lumS, 1e-4)) * brightS;
    // 16/10: el azul del zenit solo donde el cielo ya es azul (nublado/gris queda gris, no gris azulado)
    float blueS = saturate((skyIn.b - skyIn.r) / max(lumS, 1e-3) * 3.0);
    float3 tintS = lerp(float3(1.0, 1.0, 1.0), float3(0.62, 0.78, 1.12), topH * saturate(SK.y) * blueS * (1.0 - physW));   // (21/10) el cielo fisico ya trae su azul
    cS *= tintS;
    float3 hazeS = float3(0.80, 0.88, 0.98) * 0.6;
    cS = lerp(cS, max(cS, hazeS), (1.0 - topH) * saturate(SK.w) * (1.0 - physW));
    r0.xyz = lerp(r0.xyz, cS, max(skyM, skyRaw * physW));
    // --- sol visible: x37 tamano, y37 intensidad, z37 = 1 invierte el eje vertical, w37 on/off [Alt+Shift+4] ---
    float4 SN = IniParams.Load(int2(37,0));
    float3 Ls = pL_dir.xyz;
    float3 pesCloudCol = 0.0; float pesCloudT = 1.0;   // se aplican despues del sol (las nubes lo tapan)
    // (21/10) con x159 el sol y las nubes volumetricas los dibuja el cielo del juego (9adf9d46, con la camara y la luz de su
    // propio dibujo): aca solo si ese cielo no se dibujo (w159 lo escribe la DLL)
    float4 Q159t = IniParams.Load(int2(159,0)); bool domeSky = Q159t.x > 0.5 && Q159t.w > 0.5;
    if (!domeSky && SN.w > 0.5 && skyM > 0.01 && dot(Ls, Ls) > 1e-6 && abs(pS_projParam.x) > 1e-6 && abs(pS_projParam.y) > 1e-6) {
      Ls = normalize(Ls);
      if (Ls.z > 0.02) {
        float2 scS = Ls.xy / (Ls.z * pS_projParam.xy) + pS_centerOff.xy;
        float2 sunUV = float2(scS.x * 0.5 + 0.5, (SN.z > 0.5 ? scS.y : -scS.y) * 0.5 + 0.5);
        float aspS = g_psSystem.m_renderBuffer.w > 0 ? g_psSystem.m_renderBuffer.w / max(g_psSystem.m_renderBuffer.z, 1e-6) : 1.7778;
        float2 dS = (uvS - sunUV) * float2(aspS, 1.0);
        float rS = length(dS);
        float4 SP = IniParams.Load(int2(130,0));   // (20/10) sol fisico: x on | y oscurecimiento del limbo | z destello (glare) | w bruma alrededor
        float3 colS = max(pL_color.xyz, 0.0);
        colS = colS / max(dot(colS, float3(0.2126, 0.7152, 0.0722)), 1e-3);
        if (SP.x > 0.5) {
          // disco de 0.53 grados (radio 0.265) en unidades de pantalla: 1 unidad vertical = 2 tan(fov/2) -> radio = tan(0.265 deg) / projParam.y
          float radP = (0.004625 / max(abs(pS_projParam.y), 1e-4)) * 0.5 * max(SN.x, 0.1);
          float u = saturate(rS / max(radP, 1e-5));
          // oscurecimiento del limbo (Hestroffer-Magnan, aprox.): I(u) = 1 - a (1 - sqrt(1 - u^2))
          float limb = 1.0 - saturate(SP.y) * 0.6 * (1.0 - sqrt(max(1.0 - u * u, 0.0)));
          float aaP = max(g_psSystem.m_renderBuffer.w, 1e-4);
          float diskP = (1.0 - smoothstep(radP - aaP, radP + aaP, rS)) * limb;
          // glare de camara de TV: PSF con cola 1/r^2.2 (difraccion + bloom del sensor), mas un halo ancho de la bruma
          float rr = rS / max(radP, 1e-5);
          float psf = (0.35 / pow(1.0 + rr, 2.2)) * saturate(SP.z);
          float haze = exp(-rS / (radP * 22.0)) * 0.12 * saturate(SP.w);
          // energia del disco: con el limbo y el tamano real el disco es chico: brillo alto (saturado por el tonemap)
          r0.xyz += colS * (diskP * 22.0 + psf * 2.5 + haze * 0.8) * saturate(SN.y) * skyM;   // (21/10 b) menos energia: apuntando al sol quemaba todo el cielo

          float radS = 0.011 * max(SN.x, 0.1);
          float diskS = 1.0 - smoothstep(radS * 0.8, radS, rS);
          float glowS = exp(-rS / (radS * 2.5)) * 0.7 + exp(-rS / (radS * 14.0)) * 0.18;
          r0.xyz += colS * (diskS * 14.0 + glowS * 1.6) * saturate(SN.y) * skyM;
        }
      }
    }
    // === PES21 nubes volumetricas (18/10): raymarch en una capa de nubes sobre los pixeles de cielo ===
    // El rayo de cada pixel se arma con la proyeccion (como el sol visible) y se pasa a mundo con las filas de m_view
    // (misma convencion que la brujula del sol). Campo de nubes compartido con la sombra de nubes de los shaders de luz.
    { PESCloudParams CP = PESCloudLoad(g_psSystem.m_timeParam.z);
      float4 Q85 = IniParams.Load(int2(85,0)); float4 Q86 = IniParams.Load(int2(86,0));
      // etapa de diagnostico (w12 = 5): 1 = nubes apagadas, 2 = sin cbuffer de escena, 3 = sin sol, 4 = rayo bajo el horizonte, 5 = sin densidad, 6 = nube
      int cloudStage = 0;
      if (CP.on < 0.5) cloudStage = 1; else if (!(abs(pS_projParam.x) > 1e-6 && abs(pS_projParam.y) > 1e-6)) cloudStage = 2; else if (!(dot(Ls, Ls) > 1e-6)) cloudStage = 3;
      if (!domeSky && CP.on > 0.5 && skyM > 0.01 && abs(pS_projParam.x) > 1e-6 && abs(pS_projParam.y) > 1e-6 && dot(Ls, Ls) > 1e-6) {

        float3 vX = float3(pS_mat[2]._m00, pS_mat[2]._m01, pS_mat[2]._m02);
        float3 vY = float3(pS_mat[2]._m10, pS_mat[2]._m11, pS_mat[2]._m12);
        float3 vZ = float3(pS_mat[2]._m20, pS_mat[2]._m21, pS_mat[2]._m22);

        float3 Ln = normalize(Ls);

          cloudStage = 5;
          float t0 = CP.hBase / dW.y, t1 = (CP.hBase + CP.thick) / dW.y;

          int steps = (int)clamp(Q85.w > 0 ? Q85.w : 24.0, 8.0, 48.0);
          float dt = (t1 - t0) / steps;

          float mu = dot(dW, sunW);
          float g1 = 0.65, g2 = -0.25;
          float ph = lerp(0.0597 * (1.0 - g1 * g1) / pow(1.0 + g1 * g1 - 2.0 * g1 * mu, 1.5), 0.0597 * (1.0 - g2 * g2) / pow(1.0 + g2 * g2 - 2.0 * g2 * mu, 1.5), 0.3) * 4.0;
          float T = 1.0; float3 C = 0.0;
          float sunK = Q85.x > 0 ? Q85.x : 1.0, ambK = Q85.y > 0 ? Q85.y : 1.0;
          float sigma = 0.006 * dt;                                // extincion por metro x paso (18/10 b: x2, mas cuerpo)
          [loop] for (int i = 0; i < steps && T > 0.02; i++) {
            float3 pW = dW * (t0 + (i + jit) * dt);
            float d = PESCloudDensity(pW, CP, 4);
            if (d < 0.002) continue;
            // luz del sol: 3 pasos hacia el sol dentro de la capa
            float dl = 0.0;
            [unroll] for (int k = 1; k <= 3; k++) dl += PESCloudDensity(pW + sunW * (k * k * 90.0), CP, 3);
            float Tsun = exp(-dl * 0.55);                            // 18/10 b: sombra propia mas marcada
            float powder = 1.0 - exp(-d * 2.0);
            float hN = saturate((pW.y - CP.hBase) / CP.thick);
            float3 amb = ambC * lerp(0.18, 0.95, hN) * ambK;         // bases oscuras, topes claros
            float3 S = sunC * Tsun * ph * 2.4 * sunK * lerp(0.5, 1.0, powder) + amb;
            float a = 1.0 - exp(-d * sigma);
            C += T * S * a; T *= 1.0 - a;
          }
          // bruma: las nubes lejanas se funden con el horizonte
          float hz = saturate((dW.y - 0.015) / 0.12);
          pesCloudT = lerp(1.0, T, hz);
          pesCloudCol = lerp(ambC * (1.0 - T), C, hz);   // radiancia ya premultiplicada por la opacidad
          if (T < 0.98) cloudStage = 6;
        }
      }
      r0.xyz = lerp(r0.xyz, r0.xyz * pesCloudT + pesCloudCol, skyM);
      // diagnostico w12 = 5: sobre el cielo, color = direccion del rayo en MUNDO (debe girar con la camara; rojo = este, verde = arriba, azul = norte) y la opacidad de nube en el canal rojo
      if (IniParams.Load(int2(12,0)).w > 4.5 && IniParams.Load(int2(12,0)).w < 5.5 && skyM > 0.01) {
        float2 ndcD = float2(uvS.x * 2.0 - 1.0, (Q86.y > 0.5 ? 1.0 : -1.0) * (uvS.y * 2.0 - 1.0));
        float3 dirVD = normalize(float3((ndcD - pS_centerOff.xy) * pS_projParam.xy, Q86.z > 0.5 ? -1.0 : 1.0));
        float3 vXd = float3(pS_mat[2]._m00, pS_mat[2]._m01, pS_mat[2]._m02), vYd = float3(pS_mat[2]._m10, pS_mat[2]._m11, pS_mat[2]._m12), vZd = float3(pS_mat[2]._m20, pS_mat[2]._m21, pS_mat[2]._m22);
        float3 dWd = normalize(float3(dot(dirVD, vXd), dot(dirVD, vYd), dot(dirVD, vZd)));
        r0.xyz = lerp(abs(dWd) * 2.0, float3(3.0, 0.0, 0.0), 1.0 - pesCloudT);

  // === PES21 perspectiva aerea fisica (20/10): la luz que llega de lejos pierde parte por la atmosfera (extincion) y se le
  // suma luz dispersada (in-scatter): Rayleigh azula lo lejano, Mie (bruma) blanquea hacia el sol. Se calcula por pixel con
  // la profundidad real, no con la niebla del juego. 131: x on | y turbiedad extra (bruma, 0..3) | z distancia de referencia (m) | w fuerza

        float k = saturate(AP.w) * (1.0 - skyM);                                    // el cielo ya trae su color
        r0.xyz = lerp(r0.xyz, r0.xyz * T + inS, k);
      }
    }

          }
        }
      }
    }
  }
  // === PES21 reflejos en la cancha mojada (20/10): reflexion en espacio de pantalla (SSR) solo sobre el plano de la cancha ===
  // Con lluvia (estado del suelo humedo) el cesped se vuelve un espejo rugoso: se reconstruye la posicion en mundo con la
  // profundidad, y para los pixeles que estan sobre el plano de la cancha (y ~ 0, dentro del rectangulo) se marcha el rayo

      float depR = PESDepth.SampleLevel(g_samplerPoint_Wrap_s, uvR, 0).x;
      if (depR > 1e-6) {

        float2 ndcR = float2(uvR.x * 2.0 - 1.0, ySgn * (uvR.y * 2.0 - 1.0));

        bool onPitch = abs(posW.y) < 0.16 && abs(posW.x) < 56.0 && abs(posW.z) < 37.0;
        if (SR.w > 0.5) r0.xyz = onPitch ? float3(0.0, 2.0, 0.0) : float3(abs(posW.y) < 2.0 ? 2.0 : 0.0, 0.0, frac(posW.x * 0.1) * 2.0);
        if (onPitch) {
          float3 V = normalize(posW - eyeR);
          if (V.y < -0.01) {
            float3 R = float3(V.x, -V.y, V.z);
            float cosT = saturate(-V.y);

            float t = 0.35 * (1.0 + 0.5 * jitR); float hitK = 0.0; float2 hitUV = 0.0; float travel = 0.0;
            float3 p0 = posW + float3(0.0, 0.02, 0.0);

              float2 uvH = float2(sc.x * 0.5 + 0.5, ySgn * sc.y * 0.5 + 0.5);
              if (any(uvH < 0.0) || any(uvH > 1.0)) break;

  float clarK = grade ? saturate(P6.z) : 0.0;
  float lcDiff = 0.0;

      lc = max(lc, 0.0);
      float3 Vc = lc < 0.018 ? 4.5 * lc : 1.099 * pow(max(lc, 0.018), 0.45) - 0.099;
      // contraste en los medios alrededor del gris medio de la camara
      float zc = TV.z > 0.0 ? TV.z : 1.1;
      float3 wm = saturate(1.0 - abs(Vc - 0.41) * 2.2);
      Vc = lerp(Vc, 0.41 + (Vc - 0.41) * zc, wm);
      r0.xyz = lerp(saturate(Vc), r0.xyz, saturate(TV.y));
    }
  }
  // ================= GRADACION =================
  if (grade) {
    float3 c = saturate(r0.xyz);
    // sombras: levanta o hunde de forma suave sin tocar las luces altas
    c = saturate(c + shadows * (1.0 - c) * (1.0 - c) * (1.0 - c));
    // contraste en S alrededor del gris medio (curva suave, sin recorte)
    float3 s = c * c * (3.0 - 2.0 * c);
    c = lerp(c, s, saturate(contrast));
    // saturacion preservando luminancia, con proteccion de piel/colores ya saturados
    float lum = dot(c, float3(0.2126, 0.7152, 0.0722));
    float chroma = max(c.r, max(c.g, c.b)) - min(c.r, min(c.g, c.b));
    float satAmt = lerp(sat, 1.0 + (sat - 1.0) * 0.4, saturate(chroma * 2.0));
    c = saturate(lerp(lum.xxx, c, satAmt));
    // PES21 vibrancia (20/10, x133): sube los colores apagados (pasto lavado, cielo palido, tribuna gris) y casi no toca
    // lo ya saturado ni la piel (tonos calidos de saturacion media): mas color sin pisar la LUT del estadio
    { float4 VB = IniParams.Load(int2(133,0));
      if (VB.x > 0.001) {
        float l3 = dot(c, float3(0.2126, 0.7152, 0.0722));
        float mx = max(c.r, max(c.g, c.b)), mn = min(c.r, min(c.g, c.b));
        float satC = (mx - mn) / max(mx, 1e-3);
        float skin = saturate(1.0 - abs(c.r - c.b * 1.6 - 0.12) * 4.0) * step(c.b, c.g) * step(c.g, c.r) * smoothstep(0.1, 0.35, satC) * (1.0 - smoothstep(0.55, 0.75, satC));
        float k = saturate(VB.x) * (1.0 - satC) * (1.0 - 0.8 * skin) * smoothstep(0.03, 0.12, l3);
        c = saturate(lerp(l3.xxx, c, 1.0 + k * 0.9));
        // verde del pasto un poco mas vivo (y133): solo tonos verdes de saturacion media/baja
        float gK = saturate(VB.y) * saturate((c.g - max(c.r, c.b)) * 6.0) * (1.0 - satC * 0.6);
        c = saturate(lerp(dot(c, float3(0.2126, 0.7152, 0.0722)).xxx, c, 1.0 + gK * 0.35));
      } }
    // claridad (contraste local suave)
    c = saturate(c * exp2(lcDiff * clarK * 0.35 * (1.0 - skyM)));   // 16/10: sin claridad en el cielo (halos oscuros en las nubes)
    // tono partido: sombras apenas frias, luces apenas calidas (look de transmision moderna)
    float stK = saturate(P6.y);
    float lum2 = dot(c, float3(0.2126, 0.7152, 0.0722));
    float3 tint = lerp(float3(0.975, 1.0, 1.035), float3(1.03, 1.0, 0.965), smoothstep(0.15, 0.75, lum2));
    c = saturate(lerp(c, c * tint, stK));
    // nubes con volumen (solo cielo)
    if (skyM > 0) {   // PES21 cielo v2: la curva S solo en lo claro (nubes), no oscurece el azul
      float ls = dot(c, float3(0.2126, 0.7152, 0.0722));
      float3 s2 = c * c * (3.0 - 2.0 * c);
      c = lerp(c, s2, skyM * saturate(IniParams.Load(int2(28,0)).z) * smoothstep(0.45, 0.75, ls));
    }
    // viñeta muy leve
    float2 uvV = v1.xy * g_psSystem.m_renderBuffer.zw - 0.5;
    c *= 1.0 - saturate(P6.w) * smoothstep(0.35, 0.85, length(uvV * float2(1.0, 0.75)));
    r0.xyz = c;
  }
  // ==============================================
  r0.xyz = r0.xyz * g_psObject.m_localParam[4].xxx + float3(-0.5,-0.5,-0.5);
  r0.xyz = g_psObject.m_localParam[4].yyy * r0.xyz + float3(0.5,0.5,0.5);
  r1.xyz = float3(0.25,0.25,0.25) * r0.xyz;
  r0.w = cmp(g_psObject.m_localParam[1].x == 0.000000);
  o0.xyz = r0.www ? r1.xyz : r0.xyz;
  // ---- diagnostico: reloj del partido (w9 = 1): barra magenta = segundos, celeste = minutos ----
  {
    float4 P9 = IniParams.Load(int2(9,0));
    if (false) {
      float tsec = IniParams.Load(int2(10,0)).x * 60.0;   // reloj del partido
      // barra 1 (fila 40-52): segundos dentro del minuto, 0..1000 px
      if (v1.y < 52 && v1.x < 20 + fmod(tsec, 60.0) / 60.0 * 1000.0 && v1.x > 20) o0.xyz = float3(1, 0, 1);
      // barra 2 (fila 56-68): minutos enteros, 20 px por minuto, marcas cada 5
      if (v1.y > 56 && v1.x > 20 && v1.x < 20 + floor(tsec / 60.0) * 20.0) o0.xyz = (fmod(floor((v1.x - 20) / 20.0), 5) == 4) ? float3(1,1,0) : float3(0,1,1);
    }
  }
  return;
}
