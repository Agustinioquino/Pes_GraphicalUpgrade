// Hash original: 5ce3ed8702271bc1  (volcado con 3Dmigoto 1.3.16)
//
// Este shader NO ilumina: escribe el G-buffer que despues usa la pasada de luz.
//   o0.rgb = albedo (+ "velo" de borde por fresnel)
//   o1     = normal octaedrica (2x11 bits) + translucidez (6 bits, la usa el SSS del motor)
//   o2.x   = rugosidad codificada, o2.y = intensidad especular, o2.w = tipo de material
//   o3.xy  = vectores de movimiento
//
// Parametros en vivo (d3dx.ini, seccion [Constants]):
//   x  = multiplicador de translucidez / SSS   (1.0 = original)
//   y  = saturacion del albedo de la piel        (1.0 = original)
//   z  = multiplicador de especular              (1.0 = original)
//   w  = 1 activa las mejoras, 0 = shader original
//   x2 = calidez de las zonas finas (orejas, nariz, labios) usando el mapa de translucidez
//   y2 = multiplicador de rugosidad -> <1 piel mas brillosa, >1 mas mate
//   x8..w8 = gotas de sudor: cantidad, tamano, velocidad, on/off
//   x9/y9 = minuto de inicio / maximo del sudor, z9 = realismo del vello facial
//   x32/y32 = sesgo de mip del color / normal de la cara (negativo = mas nitido), w32 on/off  (16/10)
//
// ARRUGAS DE EXPRESION (independiente de w; w13 = 1 las activa, 0 = arrugas originales)
//   x7  = ganancia global de las arrugas animadas      (1.0 = original)
//   x13 = curva de respuesta: >0 las expresiones suaves ya marcan arrugas (0 = lineal)
//   y13 = sombreado de pliegues (oscurece y quita brillo dentro del surco)  (0 = nada)
//   z13 = sonrisa -> patas de gallo: refuerza la zona de ojos cuando boca y ojos
//         se activan juntos (sonrisa real / Duchenne)                      (0 = nada)
//   x14/y14/z14/w14 = ganancia por zona: frente+sienes / entrecejo / ojos / boca
//   w15 = signo del eje Y del mapa de arrugas para detectar surcos (-1 = correcto para PES)
//   x15 = 1 usa las texturas remasterizadas (t20/t21), 0 = las del juego
//   y15 = zona muerta: por debajo de este peso no aparecen arrugas (cara en reposo lisa)
//
// FISICA DE LA EXPRESION (no sube las arrugas: cambia como reacciona la piel)  w16 = on/off
//   x16 = sangre: pliegue mas oscuro/rojizo, piel estirada mas palida   (0 = nada)
//   y16 = brillo: piel estirada mas lisa y brillante, pliegue mas mate   (0 = nada)
//   z16 = translucidez: el pliegue deja pasar menos luz, la piel tensa mas (0 = nada)

// ===================== gotas de sudor procedurales =====================
float h21(float2 p) { p = frac(p * float2(123.34, 456.21)); p += dot(p, p + 45.32); return frac(p.x * p.y); }

// Devuelve: xy = gradiente para la normal, z = mascara mojada, w = altura
float vn21(float2 p) { float2 i = floor(p), f = frac(p); f = f * f * (3 - 2 * f);
  return lerp(lerp(h21(i), h21(i + float2(1,0)), f.x), lerp(h21(i + float2(0,1)), h21(i + 1), f.x), f.y); }
float2 PoreGrad(float2 uv, float F) {
  float e = 0.35;
  float2 q = uv * F;
  return float2(vn21(q + float2(e,0)) - vn21(q - float2(e,0)), vn21(q + float2(0,e)) - vn21(q - float2(0,e)));
}
float4 SweatLayer(float2 uv, float cells, float amount, float size, float t, bool moving, float fp)
{
  float2 g = uv * float2(cells, cells);
  float2 c = floor(g);
  float2 f = frac(g);
  float r0 = h21(c);
  if (r0 > amount) return 0;
  float r1 = h21(c + 17.1);
  float r2 = h21(c + 41.7);
  float r3 = h21(c + 73.3);
  float2 pos = float2(0.3 + 0.4 * r1, 0.5);
  float slide = 0;
  if (moving) {
    // la gota se queda quieta un rato y despues cae (ciclo)
    float cyc = frac(r2 + t * (0.05 + 0.08 * r3));
    slide = smoothstep(0.55, 1.0, cyc);
    pos.y = lerp(0.15, 0.95, slide);
    pos.x += sin(cyc * 18.0 + r1 * 6.28) * 0.025 * slide;   // zigzag leve
  }
  float sz = size * (0.55 + 0.6 * r3);
  float2 rad = float2(0.16, 0.20 + 0.10 * slide) * sz;     // se alarga al caer
  float2 d = (f - pos) / rad;
  float d2 = dot(d, d);
  float h = saturate(1 - d2);
  float2 grad = (d2 < 1) ? (-2 * d / rad) * 0.045 : 0;
  float wet = smoothstep(0.0, 0.2, h);                        // PES21 (20/10): borde suave (antes 0/1 duro)
  if (moving) {
    // estela mojada por encima de la gota
    float dx = abs(f.x - pos.x);
    float trail = (f.y < pos.y && f.y > 0.05) ? saturate(1 - dx / (0.035 * sz)) * slide : 0;
    wet = max(wet, trail * 0.8);
    grad.x += (f.x > pos.x ? -1 : 1) * trail * 0.25;
  }
  // PES21 (20/10): antialiasing de las gotas. Una gota de menos de ~1 pixel de radio era un espejo diminuto: titilaba como
  // brillantina en los planos medios. Se apaga la gota y el que llama suma la humedad promedio pareja (pelicula fina)
  float px = rad.x / max(fp, 1e-5);
  float k = (fp > 0.0) ? saturate((px - 0.8) / 1.6) : 1.0;
  return float4(grad * k, wet * k, h * k);
}

void main(
  float4 v0 : SV_Position0,
  float4 v1 : COLOR0,
  float2 v2 : TEXCOORD0,
  float2 w2 : TEXCOORD9,
  float4 v3 : TEXCOORD4,
  float4 v4 : TEXCOORD5,
  float4 v5 : TEXCOORD6,
  float4 v6 : TEXCOORD7,
  float4 v7 : TEXCOORD8,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,
  out float4 o3 : SV_Target3,
  out float4 o4 : SV_Target4)
{
  float4 r0,r1,r2,r3,r4;
  uint4 bitmask, uiDest;
  float4 fDest;

  // ---- parametros en vivo ----
  float4 P0 = IniParams.Load(int2(0,0));   // x y z w
  float4 P2 = IniParams.Load(int2(2,0));   // x2 y2 z2 w2
  float4 P8 = IniParams.Load(int2(8,0));   // sudor: x8 cantidad, y8 tamano, z8 velocidad, w8 on
  float4 P9 = IniParams.Load(int2(9,0));   // x9 minuto inicio sudor, y9 minuto maximo, z9 vello facial
  // detalle de textura de la cara (16/10): x32 sesgo del color, y32 sesgo del normal map (poros), w32 on/off
  float4 P32 = IniParams.Load(int2(32,0));
  float faceBD = P32.w > 0.5 ? P32.x : 0.0;
  float faceBN = P32.w > 0.5 ? P32.y : 0.0;
  float tmin = IniParams.Load(int2(10,0)).x;   // minutos reales de partido (los calcula d3dx.ini)
  float sweatProg = saturate((tmin - P9.x) / max(P9.y - P9.x, 0.1));
  bool  enhance      = P0.w > 0.5;
  float sssBoost     = enhance ? P0.x : 1.0;
  float albedoSat    = enhance ? P0.y : 1.0;
  float specScale    = enhance ? P0.z : 1.0;
  float thinWarmth   = enhance ? P2.x : 0.0;
  float glossScale   = enhance ? P2.y : 1.0;
  // ---- arrugas de expresion ----
  float4 P7  = IniParams.Load(int2(7,0));
  float4 P13 = IniParams.Load(int2(13,0));
  float4 P14 = IniParams.Load(int2(14,0));
  float4 P15 = IniParams.Load(int2(15,0));
  bool  wrOn      = P13.w > 0.5;
  float wrGain    = (P7.x  > 0) ? P7.x  : 1.0;
  float wrCurve   = max(P13.x, 0.0);
  float wrCrease  = max(P13.y, 0.0);
  float wrSmile   = max(P13.z, 0.0);
  float4 wrZone   = float4(P14.x > 0 ? P14.x : 1.0, P14.y > 0 ? P14.y : 1.0,
                           P14.z > 0 ? P14.z : 1.0, P14.w > 0 ? P14.w : 1.0);
  float wrYSign   = (P15.w > 0) ? 1.0 : -1.0;
  bool  wrHi      = wrOn && P15.x > 0.5;
  float wrWeight  = 0;     // peso final de arrugas (para el sombreado de pliegues)
  float wrFold    = 0;     // 0..1: cuanto "surco" hay en este pixel
  float4 P16 = IniParams.Load(int2(16,0));
  bool  exOn      = wrOn && P16.w > 0.5;
  float exBlood   = max(P16.x, 0.0);
  float exGloss   = max(P16.y, 0.0);
  float exTrans   = max(P16.z, 0.0);
  float exFoldZ   = 0;     // zona de pliegue (ancha y suave) * peso de la expresion
  float exRidgeZ  = 0;     // zona de piel estirada
  // valores seguros si el ini no los define (IniParams vale 0)
  if (sssBoost   <= 0) sssBoost   = 1.0;
  if (albedoSat  <= 0) albedoSat  = 1.0;
  if (specScale  <= 0) specScale  = 1.0;
  if (glossScale <= 0) glossScale = 1.0;

  // ---- vectores de movimiento (arreglado: el decompilador generaba w2.xzww) ----
  r0.xy = float2(-0.5,-0.5) + v0.xy;
  r1.xyzw = v7.xyzw / w2.xxyy;
  r1.xyzw = r1.xyzw * float4(0.5,0.5,0.5,0.5) + float4(0.5,0.5,0.5,0.5);
  r1.xyzw = r1.zwxy * float4(1,-1,1,-1) + float4(0,1,0,1);
  r0.zw = r1.zw + -r1.xy;
  o3.xy = g_psSystem.m_renderBuffer.xy * r0.zw;
  float2 pesMotionRaw = r0.zw;   // PES21 cejas: movimiento sin escalar (para las hebras)

  // ---- arrugas animadas ----
  r0.z = cmp(v2.x < 0.5);
  if (r0.z != 0) {
    r1.x = g_psMaterial.m_materials[0].y;
  } else {
    r1.x = g_psMaterial.m_materials[4].x;
  }
  r0.zw = float2(1,0.5) * v2.xy;
  r2.xyzw = wrHi ? pesWrinkleM.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw : g_tex_wrinkle_mask.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw;
  r0.zw = v2.xy * float2(1,0.5) + float2(0,0.5);
  r3.xyzw = wrHi ? pesWrinkleM.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw : g_tex_wrinkle_mask.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw;
  // pesos por zona que manda el motor segun la pose de la cara:
  //   mitad superior de la mascara: frente, sienes, entrecejo, ojos  -> m_materials[3]
  //   mitad inferior: boca (lado izq/der), mandibula der/izq, cuello -> r1.x, m_materials[4].yzw
  float4 wTop = g_psMaterial.m_materials[3].xyzw;
  float4 wBot = float4(r1.x, g_psMaterial.m_materials[4].yzw);
  if (wrOn) {
    // sonrisa real: boca y ojos activos a la vez -> mas patas de gallo
    float smile = sqrt(saturate(max(g_psMaterial.m_materials[0].y, g_psMaterial.m_materials[4].x)) * saturate(wTop.w));
    wTop *= float4(wrZone.x, wrZone.x, wrZone.y, wrZone.z * (1.0 + wrSmile * smile));
    wBot.x *= wrZone.w;
  }
  r0.z = dot(wTop, r2.xyzw);
  r0.w = dot(wBot, r3.xyzw);
  r0.z = saturate(r0.z + r0.w);
  if (wrOn) {
    // zona muerta: los pesos chicos de la cara en reposo no dibujan arrugas (y15)
    float wrDead = saturate(P15.y);
    r0.z = saturate((r0.z - wrDead) / max(1.0 - wrDead, 0.05));
    r0.z = saturate(r0.z * wrGain);
    r0.z = 1.0 - pow(1.0 - r0.z, 1.0 + wrCurve);
  }
  wrWeight = r0.z;

  // ---- normal de detalle + normal de animacion ----
  r1.xyz = g_tex_normal.SampleBias(g_samplerAnisotropic_Clamp_s, v2.xy, faceBN).xyw;
  r1.x = r1.x * r1.z;
  r1.xy = r1.xy * float2(2,2) + float2(-1,-1);
  r0.w = -r1.x * r1.x + 1;
  r0.w = -r1.y * r1.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.w = rsqrt(r0.w);
  r1.z = r1.w * r0.w;
  r2.xyz = wrHi ? pesWrinkleN.Sample(g_samplerAnisotropic_Clamp_s, v2.xy).xyw : g_tex_normal_animation.Sample(g_samplerAnisotropic_Clamp_s, v2.xy).xyw;
  r2.x = r2.x * r2.z;
  r3.xy = r2.xy + r2.xy;
  r2.xy = r2.xy * float2(2,2) + float2(-1,-1);
  r0.w = -r2.x * r2.x + 1;
  r0.w = -r2.y * r2.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.w = rsqrt(r0.w);
  r3.z = r1.w * r0.w;
  // surcos: divergencia del mapa de arrugas (las normales "miran" hacia el fondo del pliegue)
  float wrLod = wrHi ? pesWrinkleN.CalculateLevelOfDetail(g_samplerAnisotropic_Clamp_s, v2.xy) : g_tex_normal_animation.CalculateLevelOfDetail(g_samplerAnisotropic_Clamp_s, v2.xy);
  if (wrOn && (wrCrease > 0 || exOn) && wrWeight > 0.002) {
    uint wtw, wth;
    if (wrHi) pesWrinkleN.GetDimensions(wtw, wth); else g_tex_normal_animation.GetDimensions(wtw, wth);
    float2 wdu = float2(1.5 / max(wtw, 1u), 0);
    float2 wdv = float2(0, 1.5 / max(wth, 1u));
    float4 wa = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdu, wrLod) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdu, wrLod);
    float4 wb = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdu, wrLod) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdu, wrLod);
    float4 wc = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdv, wrLod) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdv, wrLod);
    float4 wd = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdv, wrLod) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdv, wrLod);
    float divX = (wa.x * wa.w - wb.x * wb.w) * 2.0;
    float divY = (wc.y - wd.y) * 2.0;
    float wdiv = divX + wrYSign * divY;
    wrFold = saturate(-wdiv * 4.0) * wrWeight;
    if (exOn) {
      // misma deteccion con un radio 4 veces mayor -> zonas suaves, no lineas
      float2 wdu2 = wdu * 4.0, wdv2 = wdv * 4.0;
      float4 za = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdu2, wrLod + 1) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdu2, wrLod + 1);
      float4 zb = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdu2, wrLod + 1) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdu2, wrLod + 1);
      float4 zc = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdv2, wrLod + 1) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy + wdv2, wrLod + 1);
      float4 zd = wrHi ? pesWrinkleN.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdv2, wrLod + 1) : g_tex_normal_animation.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy - wdv2, wrLod + 1);
      float zdiv = (za.x * za.w - zb.x * zb.w) * 0.5 + wrYSign * (zc.y - zd.y) * 0.5;
      exFoldZ  = saturate(-zdiv * 2.0) * wrWeight;
      exRidgeZ = saturate( zdiv * 2.0) * wrWeight;
    }
  }
  r2.xyz = float3(-1,-1,-1) + r3.xyz;
  r1.xyz = r2.xyz * r0.zzz + r1.xyz;
  r0.z = dot(r1.xyz, r1.xyz);
  r0.z = rsqrt(r0.z);
  r1.xyz = r1.xyz * r0.zzz;
  // === PES21 (19/10) micro normal de poros: mosaico fino sobre la normal del juego, solo de cerca.
  // x94 intensidad, y94 escala (repeticiones sobre la cara), w94 on/off
  float pesPoreA = 0.0, pesMicroVar = 0.0, pesPoreOcc = 1.0, pesTZone = 0.0;
  float4 P94 = IniParams.Load(int2(94,0));   // (declarado aca: lo usa tambien el bloque del especular mas abajo)
  float3 pesMicroDbg = float3(-1, 0, 0);     // dbg: intensidad = 2.00 pinta el mosaico;  pesMicroVar: varianza de la micro normal (Toksvig) -> rugosidad extra
  { if (enhance && P94.w > 0.5) {
      // (27/10) poros a tamano real: antes el mosaico se repetia y94 veces sobre la textura de la cara, y donde la UV esta
      // estirada (nariz, orejas, mandibula, cuello) el poro salia estirado o gigante. Ahora se mide cuanta piel cubre la UV en
      // este punto (area de la superficie / area de UV, con las derivadas de la posicion en vista) y el mosaico se repite segun
      // eso: un mosaico = 22 mm de piel con y94 = 7 (mas = poros mas chicos). La escala se redondea a potencias de 2 y se
      // funden las dos vecinas, asi el dibujo no se deforma donde la escala cambia.
      float3 vdir = normalize(v3.xyz);
      float3 pV = -vdir * ((1.0 / max(v0.w, 1e-6)) / max(abs(vdir.z), 0.05));   // posicion en vista (metros)
      float3 dPx = ddx(pV), dPy = ddy(pV);
      float2 dUx = ddx(v2.xy), dUy = ddy(v2.xy);
      float mPerUv = sqrt(length(cross(dPx, dPy)) / max(abs(dUx.x * dUy.y - dUx.y * dUy.x), 1e-14));
      float tileM = 0.022 * 7.0 / max(P94.y, 1.0);
      float lr = log2(clamp(mPerUv / tileM, 1.0, 4096.0));
      float k0 = floor(lr), kf = lr - k0;
      float sc = exp2(k0);
      float2 muv = v2.xy * sc;
      float2 mdx = dUx * sc, mdy = dUy * sc;
      // fundido por tamano en pantalla: con el pixel mas grande que ~2 celdas de la piel (22 por mosaico) el relieve ya no se ve
      float pxM = sqrt(length(dPx) * length(dPy));
      float near = saturate((2.5 - pxM / (tileM / 22.0)) / 1.5);
      if (near > 0.001) {
        float4 guide = pesFacePores.Sample(g_samplerLinear_Clamp_s, v2.xy);   // por zona: densidad, tamano, zona T, oclusion
        pesTZone = guide.b;
        float4 mnA = pesSkinMicro.SampleGrad(g_samplerAnisotropic_Clamp_s, frac(muv), mdx, mdy);   // frac + grad: sin costura de mip
        float4 mnB = pesSkinMicro.SampleGrad(g_samplerAnisotropic_Clamp_s, frac(muv * 2.0 + 0.53), mdx * 2.0, mdy * 2.0);
        float4 mn = lerp(mnA, mnB, kf);
        // poros grandes (nariz, barbilla): el mismo mosaico a la mitad de escala, mezclado segun el tamano de la zona
        float4 mn2 = pesSkinMicro.SampleGrad(g_samplerAnisotropic_Clamp_s, frac(muv * 0.5 + 0.37), mdx * 0.5, mdy * 0.5);
        float big = guide.g;
        mn = lerp(mn, mn2, big * 0.7);
        float3 tn = mn.xyz * 2.0 - 1.0;
        // solo piel: se atenua donde la quinta textura marca pelo/barba (A) y donde el mapa de zonas no es piel (A)
        float skinM = 1.0;
        { float4 P68e = IniParams.Load(int2(68,0));
          float4 FBe = pesFaceB.Sample(g_samplerLinear_Clamp_s, v2.xy);
          if (P68e.x > 0.5 && FBe.a > 0.5) { float4 FAe = pesFaceA.Sample(g_samplerLinear_Clamp_s, v2.xy); skinM *= 1.0 - saturate(FAe.a * 1.5); }
          skinM *= lerp(0.5, 1.0, pesZones.Sample(g_samplerLinear_Clamp_s, v2.xy).a); }
        float str = clamp(P94.x, 0.0, 2.0) * near * 1.1 * skinM * (0.25 + 1.0 * guide.r);   // 19/10 d: densidad por zona (guia)
        r1.xy += tn.xy * str;
        r1.xyz = normalize(r1.xyz);
        pesPoreA = mn.a * near * (0.3 + guide.r);
        // oclusion de poro: el fondo del poro recibe menos luz (ambiente y directa). Mas donde la guia marca oclusion.
        pesPoreOcc = 1.0 - pesPoreA * guide.a * 0.55 * saturate(P94.x);
        if (P94.x >= 1.995) pesMicroDbg = float3(mn.xy, near);   // diagnostico: con Intensidad al tope se ve el mosaico en crudo (rojo/verde) y el fundido en azul
        // Toksvig (GPU Gems 3 / dqlin): la micro normal promediada en un area mas grande pierde largo; ese acortamiento
        // es la varianza de la micro-geometria y se convierte en rugosidad para el brillo (y de lejos, donde el mosaico ya no se ve,
        // la rugosidad se mantiene y el brillo no destella)
        { float4 ma = pesSkinMicro.SampleGrad(g_samplerAnisotropic_Clamp_s, frac(muv), mdx * 6.0, mdy * 6.0);
          float ln = length(ma.xyz * 2.0 - 1.0);
          pesMicroVar = saturate((1.0 - ln) * 8.0) * saturate(P94.x * 0.7) * skinM; } 
      } } }
  // 16/10: oclusion de cavidades a partir del normal map (pliegues de la oreja, arco de la ceja, aletas de la nariz):
  // donde la normal se inclina mucho la superficie esta dentro de un hueco y le llega menos luz. x35 fuerza, w35 on/off
  float4 P35 = IniParams.Load(int2(35,0));
  float pesCav = 1.0;
  if (P35.w > 0.5) {
    float tilt = saturate(1.0 - r1.z);                       // 0 = plano, crece en los pliegues
    pesCav = 1.0 - saturate(P35.x) * smoothstep(0.02, 0.35, tilt) * 0.55;
  }

  // ---- gotas de sudor: relieve en la normal (espacio tangente) ----
  float sweatWet = 0;
  float fhMask = 0;
  // === PES21 quinta textura (18/10): generada en vivo a partir de las 4 texturas de ESTA cara (regla facegen) ===
  // x68 on | y68 cavidades | z68 grosor | w68 vello/pelo
  float4 P68 = IniParams.Load(int2(68,0));
  float4 FB = pesFaceB.Sample(g_samplerLinear_Clamp_s, v2.xy);
  bool fgOk = enhance && P68.x > 0.5 && FB.a > 0.5;
  float4 FA = fgOk ? pesFaceA.Sample(g_samplerLinear_Clamp_s, v2.xy) : float4(0, 0.5, 0, 0);
  // === PES21 (20/10) caras genericas sin corte (x147): la cara generica de Konami es un MONTAJE (textura FaceSKIN pegada en
  // el rectangulo u 0.148..0.852, v 0.152..0.856 del plano de la cabeza; medido registrando las texturas) y el resto de la
  // cabeza (cuello, orejas, nuca) es otra pieza que dibuja el shader de piel del cuerpo. Todo lo propio de este shader
  // terminaba de golpe en ese borde: debajo de la mandibula la cara quedaba naranja (translucidez x1.41 + mapa de SSS) y el
  // cuello no. En una franja de ~5% junto al borde: la quinta textura se apaga siempre; y si la DLL no iguala la pieza del
  // cuello (w147, lo pone la DLL nueva), tambien vuelven al juego color, translucidez y brillo (igual que el cuello).
  float4 P147 = IniParams.Load(int2(147,0));
  float pesEdge = 0.0;
  {
    // (20/10 b) en este shader NO: las genericas en partido usan una sola malla para toda la cabeza (textura de 1024 con el
    // montaje ya pegado en el plano de la cabeza), sin pieza aparte para el cuello. Fundir el borde aca le sacaba la mascara de
    // pelo de la quinta textura al cuero cabelludo (el pelo rapado se veia gris y brilloso). Solo aplica a ce188 (dos UV).
    bool gen = false;
    float2 gr = float2(0.5, 0.5);
    float dE = min(min(gr.x, 1.0 - gr.x), min(gr.y, 1.0 - gr.y)) * 0.7044;   // distancia al borde en el plano de la cabeza
    pesEdge = gen ? 1.0 - smoothstep(0.012, 0.07, dE) : 0.0;
  }
  float pesEdgeAll = (P147.w > 0.5) ? pesEdge * (1.0 - smoothstep(0.50, 0.62, v2.y)) : pesEdge;   // con la DLL nueva la cabeza solo iguala cuello y orejas
  FA = lerp(FA, float4(0, 0.5, 0, 0), pesEdge);
  // === PES21 zonas de la cara (18/10): sudor por zonas, labios, cuello ===
  // x72 pelicula de sudor, y72 contraste entre zonas, z72 color de labios, w72 on/off | x73 brillo de labios, y73 tono (rosado..rojo), z73 gotas por zona
  float4 P72 = IniParams.Load(int2(72,0));
  float4 P73 = IniParams.Load(int2(73,0));
  float4 FZ = pesZones.Sample(g_samplerLinear_Clamp_s, v2.xy);
  bool zonesOn = enhance && P72.w > 0.5;
  float zoneSw = zonesOn ? lerp(1.0, saturate(FZ.r * 1.25), saturate(P72.y)) : 1.0;   // propension al sudor de esta zona
  float lipM = zonesOn ? FZ.g : 0.0;
  // no toda la cara suda a la vez: la frente, las sienes y la nariz arrancan antes; mandibula y mejillas despues
  float sweatProgZ = zonesOn ? saturate(sweatProg * lerp(1.0, 0.35 + 1.1 * saturate(FZ.r * 1.25), saturate(P73.z))) : sweatProg;
  float filmWet = zonesOn ? saturate(sweatProgZ * zoneSw) * saturate(P72.x) : 0.0;
  // (25/10) lluvia en la piel (z12): con lluvia en el partido (114.x = 1, lo publica la DLL) la cara se moja entera, no por
  // zonas: pelicula de agua (menos rugosidad, mas brillo, albedo apenas mas oscuro) y gotas aunque todavia no haya sudor
  float4 P12 = IniParams.Load(int2(12,0));
  float pesRainV = IniParams.Load(int2(114,0)).x;
  float rainK = (enhance && pesRainV > 0.5 && pesRainV < 1.5) ? saturate(P12.z) : 0.0;
  filmWet = max(filmWet, rainK * 0.85);
  sweatProgZ = max(sweatProgZ, rainK * 0.6);
  if (enhance && P9.z > 0) {
    // vello facial pintado (barba, cejas, bigote): mascara por color oscuro y poco saturado
    float3 albE = g_tex_diffuse.Sample(g_samplerAnisotropic_Clamp_s, v2.xy).xyz;
    float lumE = dot(albE, float3(0.2126, 0.7152, 0.0722));
    float mxE = max(albE.r, max(albE.g, albE.b)), mnE = min(albE.r, min(albE.g, albE.b));
    float satE = (mxE - mnE) / max(mxE, 1e-3);
    // relativo al tono de piel local (mip alto = color medio de la zona): sirve para pieles claras y oscuras
    float3 albA = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy, 5.0).xyz;
    float lumA = dot(albA, float3(0.2126, 0.7152, 0.0722));
    float ratioE = lumE / max(lumA, 0.005);
    fhMask = smoothstep(0.62, 0.38, ratioE) * smoothstep(0.15, 0.06, lumE) * smoothstep(0.45, 0.20, satE) * saturate(P9.z);
    if (fhMask > 0.01) {
      // relieve de pelitos: ruido fino y alargado -> textura de barba/cejas en vez de pintura plana
      float2 hc = floor(v2.xy * float2(1500.0, 520.0));
      float2 jit = float2(h21(hc), h21(hc + 7.3)) - 0.5;
      r1.xy += jit * 0.55 * fhMask;
      r1.xyz = normalize(r1.xyz);
    }
  }
  if (fgOk) fhMask = max(fhMask, FA.a * saturate(P68.w));   // PES21 quinta textura: vello y nacimiento del pelo exactos
  fhMask *= 1.0 - lipM;   // PES21 zonas: los labios no son barba
  float poreAmt = 0;
  if (enhance) {
    // poros y microrrelieve de la piel (se apaga a distancia para no titilar)
    float fw = max(length(ddx(v2.xy)), length(ddy(v2.xy))) * 1400.0;
    poreAmt = saturate(1.5 - fw) * (1.0 - fhMask);
    float2 pg = PoreGrad(v2.xy, 1400.0) * 0.10 + PoreGrad(v2.xy + 0.13, 520.0) * 0.06;
    r1.xy += pg * poreAmt;
    r1.xyz = normalize(r1.xyz);
  }
  float pesSwFp = max(length(ddx(v2.xy)), length(ddy(v2.xy)));   // PES21 (20/10): tamano del pixel en uv (fuera de la rama)
  if (enhance && P8.w > 0.5 && (sweatProg > 0.001 || rainK > 0.001)) {
    float t = g_psSystem.m_timeParam.z * max(P8.z, 0.01);
    float4 sA = SweatLayer(v2.xy, 70.0, P8.x * sweatProgZ, max(P8.y, 0.1), t, true, pesSwFp * 70.0);          // PES21 zonas: gotas donde se suda
    float4 sB = SweatLayer(v2.xy + 0.37, 150.0, P8.x * 1.3 * sweatProgZ, max(P8.y, 0.1) * 0.8, t, false, pesSwFp * 150.0);
    float2 sg = sA.xy + sB.xy * 0.7;
    // gotas que ya no se resuelven (menos de ~1 pixel): su humedad promedio, pareja (sin el mosaico de celdas)
    float kA = saturate((0.16 * max(P8.y, 0.1) / max(pesSwFp * 70.0, 1e-5) - 0.8) / 1.6);
    float kB = saturate((0.128 * max(P8.y, 0.1) / max(pesSwFp * 150.0, 1e-5) - 0.8) / 1.6);
    float farW = saturate(P8.x * sweatProgZ * (0.10 * (1.0 - kA) + 0.08 * (1.0 - kB)));
    sweatWet = saturate(max(max(sA.z, sB.z * 0.9), farW));
    r1.xy += sg;
    r1.xyz = normalize(r1.xyz);
  }

  // ---- espacio tangente -> mundo ----
  r0.z = dot(v4.xyz, v4.xyz);
  r0.z = rsqrt(r0.z);
  r2.xyz = v4.xyz * r0.zzz;
  r0.z = dot(v5.xyz, v5.xyz);
  r0.z = rsqrt(r0.z);
  r3.xyz = v5.xyz * r0.zzz;
  r3.xyz = r3.xyz * r1.yyy;
  r1.xyw = r2.xyz * r1.xxx + r3.xyz;
  r0.z = dot(v6.xyz, v6.xyz);
  r0.z = rsqrt(r0.z);
  r2.xyz = v6.xyz * r0.zzz;
  r1.xyz = r2.xyz * r1.zzz + r1.xyw;

  // ---- mascara de rotacion (ojos) ----
  r2.xyz = g_tex_rotation_mask.Sample(g_samplerAnisotropic_Clamp_s, v2.xy).xyz;
  r0.zw = float2(-10,-10) + g_psMaterial.m_materials[5].yz;
  r0.zw = min(float2(0,0), r0.zw);
  r2.x = g_psMaterial.m_materials[5].x * r2.x;
  r0.zw = r0.zw * r2.zy;
  r0.z = -0.0375000015 * r0.z;
  r0.z = r2.x * -0.00999999978 + r0.z;
  r0.z = r0.w * -0.0375000015 + r0.z;
  r1.w = r1.y + r0.z;
  r0.z = dot(r1.xzw, r1.xzw);
  r0.z = rsqrt(r0.z);
  r1.xyz = r1.xwz * r0.zzz;

  // ---- especular / brillo / albedo / suciedad ----
  r0.zw = g_tex_srm.SampleBias(g_sampler_srm_s, v2.xy, faceBD).xy;
  float pesSpec0 = r0.z, pesSrmY = r0.w;   // (20/10) valores del juego, para el borde de las genericas
  r1.w = g_tex_diffuse.CalculateLevelOfDetail(g_samplerAnisotropic_Clamp_s, v2.xy);
  r1.w = clamp(r1.w + faceBD, 0, 5);
  r2.xyz = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy, r1.w).xyz;
  // (27/10) marcas de la piel (y124): lunares, pecas, manchas y venitas que ya estan pintadas en la textura de la cara se
  // pierden (la textura es chica y se ve lavada). Se realza el detalle fino del albedo: la luminancia de este punto contra el
  // promedio de su zona (2.5 niveles de mip mas arriba), solo en la piel (no en barba ni pelo pintados) y solo de cerca.
  // No inventa marcas: solo marca mas las que la cara ya tiene. 0 = como el juego
  { float mk = saturate(IniParams.Load(int2(124,0)).y) * (1.0 - fhMask) * saturate(3.5 - r1.w);
    if (enhance && mk > 0.001) {
      float3 aB = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy, r1.w + 2.5).xyz;
      float rat = (dot(r2.xyz, float3(0.2126, 0.7152, 0.0722)) + 0.01) / (dot(aB, float3(0.2126, 0.7152, 0.0722)) + 0.01);
      r2.xyz *= pow(clamp(rat, 0.25, 4.0), 0.9 * mk);
    } }
  r3.xyz = g_tex_Dirty.Sample(g_sampler_Dirty_s, v2.xy).xyz;
  r1.w = saturate(dot(g_psMaterial.m_materials[1].yzw, r3.xyz));
  r3.xy = r0.ww * float2(255,255) + float2(-205,-170);
  r4.x = 0.0110000009 * r3.y;
  r2.w = r3.x * -0.00800000038 + 0.930000007;
  r2.w = max(0.730000019, r2.w);
  r2.w = min(0.930000007, r2.w);
  r4.x = saturate(r4.x);
  r3.x = min(0.660000026, r4.x);
  r3.yzw = r2.xyz * r2.www + -r2.xyz;
  r2.xyz = r1.www * r3.yzw + r2.xyz;
  r2.w = r0.w * r3.x + -r0.w;
  r0.w = r1.w * r2.w + r0.w;
  float3 pesAlb0 = r2.xyz; float pesRough0 = max(min(0.56078434, pesSrmY), r0.w);   // (20/10) como los deja el shader del cuerpo

  // ---- translucidez (mapa de grosor de la piel) ----
  r1.w = g_tex_translucent.Sample(g_samplerLinear_Clamp_s, v2.xy).y;
  float pesTrans0 = r1.w;
  // === PES21 mapa de SSS (18/10): generado a partir de 5 caras reales del juego (orejas, nariz, labios, parpados).
  // x62 on | y62 cuanto sube la translucidez en las zonas finas | z62 sangre (tinte rojizo donde la piel es fina) | w62 reservado
  float4 P62 = IniParams.Load(int2(62,0));
  float4 sssMap = pesSSSMap.Sample(g_samplerLinear_Clamp_s, v2.xy);
  float mapBlood = 0.0;
  if (enhance && P62.x > 0.5) {
    float k = sssMap.w;                                           // solo piel (el pelo del mapa queda afuera)
    float mapThin = sssMap.x * saturate(P62.y);
    r1.w = lerp(r1.w, max(r1.w, mapThin), k);                     // nunca baja la del juego: solo afina donde corresponde
    r1.w = saturate(r1.w + 0.12 * sssMap.z * saturate(P62.y) * k);  // la sangre bajo la piel tambien dispersa
    mapBlood = sssMap.z * k * saturate(P62.z);
  }
  // 18/10 v2: solo sube la translucidez donde el grosor generado dice 'fino' (orejas, aletas, labios, parpados);
  // antes la subia en casi toda la cara y el SSS se veia ceroso y naranja
  if (fgOk) r1.w = max(r1.w, FA.b * saturate(P68.z));   // 18/10: grosor de la quinta textura tal cual (la version con umbral no gusto)
  r1.w = max(r1.w, 0.85 * lipM * saturate(P72.z));         // PES21 zonas: los labios son finos e irrigados (mas luz adentro)
  float thin = r1.w;

  // ---- dithering de LOD / fade ----
  r0.xy = float2(0.125,0.125) * r0.xy;
  r3.yz = g_tex_mesh.Sample(g_samplerPoint_Wrap_s, r0.xy).xw;
  r3.x = g_psSystem.m_param.w * r3.z;
  r0.x = 1;
  r0.y = v1.w;
  r0.xy = r0.xy + -r3.xy;
  r0.xy = cmp(r0.xy < float2(0,0));
  r0.x = (int)r0.y | (int)r0.x;
  if (r0.x != 0) discard;

  // ================= MEJORAS DE PIEL =================
  // 1) Albedo: la piel de PES sale grisacea/"plastica". Saturacion suave
  //    preservando luminancia + calidez en zonas finas (donde la sangre
  //    se ve a traves de la piel: orejas, aletas de la nariz, labios).
  {
    float lum = dot(r2.xyz, float3(0.2126, 0.7152, 0.0722));
    r2.xyz = max(0, lerp(lum.xxx, r2.xyz, albedoSat));
    float3 bloodTint = float3(1.04, 0.98, 0.96);   // 21/10: mas sutil
    r2.xyz *= lerp(float3(1,1,1), bloodTint, saturate(thin * thinWarmth));
    r2.xyz *= lerp(float3(1,1,1), float3(1.0, 0.95, 0.94), saturate(mapBlood));   // 21/10: mas sutil   // PES21 mapa de SSS: zonas irrigadas
    // PES21 zonas (18/10): color de labios propio (rosado natural .. rojo), conservando la textura del juego
    { float3 lipTint = lerp(float3(1.10, 0.84, 0.86), float3(1.16, 0.70, 0.72), saturate(P73.y));
      r2.xyz = lerp(r2.xyz, r2.xyz * lipTint, lipM * saturate(P72.z)); }
    if (fgOk) {   // PES21 quinta textura: cavidades (pliegues, alas de nariz, comisuras) mas oscuras y rojizas, y mas mates
      float cavK = FA.r * saturate(P68.y);
      r2.xyz *= lerp(float3(1,1,1), float3(0.88, 0.83, 0.81), cavK);   // 21/10: casi neutro (con el rojo-marron fuerte la cara quedaba naranja y manchada; el AO ya oscurece el hueco)
      r0.w = lerp(r0.w, max(r0.w, 0.7), cavK * 0.4);
    }
    r2.xyz = saturate(r2.xyz);
    // rubor progresivo por el esfuerzo (sube con el reloj del partido)
    float flushF = sweatProg * P12.x;
    // (25/10) sangre por zona (y12): la sangre sube donde la piel esta mas irrigada (mejillas, nariz, orejas: canal de sangre
    // del mapa de SSS y lo fino), no pareja en toda la cara; y con mas melanina se ve mucho menos (piel oscura: casi nada).
    // 0 = rubor parejo como antes
    float flushZ = 1.0;
    if (P12.y > 0.001) {
      float bl = sssMap.w > 0.01 ? sssMap.z * sssMap.w : 0.5;               // sin mapa de SSS: zona media
      float zoneB = saturate(0.30 + 1.10 * bl + 0.45 * saturate(thin));
      float lA = dot(r2.xyz, float3(0.2126, 0.7152, 0.0722));
      float melA = saturate(log2(0.47 / max(lA, 1e-3)) / 2.85);             // indice de melanina (PESMelanin)
      flushZ = lerp(1.0, 1.6 * zoneB * (1.0 - 0.7 * melA), saturate(P12.y));
    }
    flushF *= flushZ;
    r2.xyz *= float3(1.0 + flushF, 1.0 - flushF * 0.35, 1.0 - flushF * 0.45);
  }
  // cavidades (x35): menos luz difusa y menos brillo dentro de los huecos; la translucidez se mantiene (las orejas siguen rojas a contraluz)
  r2.xyz *= pesCav;
  r0.z  *= lerp(1.0, pesCav, 0.8);
  // Pliegues de expresion: dentro del surco llega menos luz y la piel brilla menos
  if (wrOn && wrFold > 0) {
    float fold = saturate(wrFold * wrCrease);
    r2.xyz *= (1.0 - 0.35 * fold);
    r0.z  *= (1.0 - 0.5 * fold);
    r0.w   = lerp(r0.w, max(r0.w, 0.6), 0.5 * fold);
  }
  // Fisica de la expresion: sangre, brillo y translucidez segun pliegue / estiramiento
  if (exOn && (exFoldZ > 0 || exRidgeZ > 0)) {
    float fz = saturate(exFoldZ  * exBlood);
    float rz = saturate(exRidgeZ * exBlood);
    r2.xyz *= lerp(float3(1,1,1), float3(0.93, 0.86, 0.86), fz);              // sangre acumulada en el pliegue
    r2.xyz  = lerp(r2.xyz, saturate(r2.xyz * float3(1.0, 1.04, 1.05) + 0.012), rz * 0.6); // piel tensa: mas palida
    r0.w = saturate(r0.w * (1.0 - 0.25 * exRidgeZ * exGloss) + 0.08 * exFoldZ * exGloss);  // rugosidad
    r0.z *= (1.0 + 0.25 * exRidgeZ * exGloss);                                // especular
    r1.w *= (1.0 - 0.35 * exFoldZ * exTrans) * (1.0 + 0.2 * exRidgeZ * exTrans); // translucidez
  }
  // 2) Especular: la piel real refleja ~2.8% (F0 0.028). El valor del juego
  //    suele quedar alto -> efecto plastico. Se escala de forma controlada.
  r0.z = saturate(r0.z * specScale);
  // 3) Brillo/gloss (entrada de la curva de rugosidad)
  r0.w = saturate(r0.w * glossScale);
  // PES21 zonas (18/10): labios mas lisos y brillosos que la piel; pelicula de sudor progresiva por zonas (antes de las gotas)
  r0.w = lerp(r0.w, min(r0.w, 0.30), lipM * saturate(P73.x));
  r0.z = lerp(r0.z, max(r0.z, 0.55), lipM * saturate(P73.x) * 0.8);
  r0.w = lerp(r0.w, r0.w * 0.45, filmWet);
  r0.z = lerp(r0.z, max(r0.z, 0.5), filmWet * 0.8);
  r2.xyz *= 1.0 - 0.06 * filmWet;
  // superficie mojada: muy brillante y lisa, albedo apenas mas oscuro
  r0.w = lerp(r0.w, 0.05, sweatWet);  // r0.w es rugosidad: mojado = liso (PES21 20/10: 0.05, el agua no es un espejo perfecto)
  // vello facial: no es traslucido, es mas aspero y refleja menos
  r0.w = lerp(r0.w, max(r0.w, 0.75), fhMask);
  // micro-geometria de poros en el especular (19/10): dentro del poro el brillo baja y la superficie es mas aspera;
  // la varianza de la micro normal (Toksvig) suma rugosidad para que el brillo no sea un plastico liso
  r0.w = saturate(r0.w + pesMicroVar * 0.30 + pesPoreA * 0.20);
  // zona T mas brillante (grasa natural), mejillas mas mate; y el albedo se oscurece en el fondo de los poros
  r0.w = saturate(r0.w * lerp(1.06, 0.86, pesTZone * saturate(P94.x)));
  r0.z *= (1.0 - 0.45 * pesPoreA) * lerp(0.95, 1.12, pesTZone * saturate(P94.x));
  r2.xyz *= pesPoreOcc;
  r0.z *= (1.0 - 0.6 * fhMask);
  r0.z = lerp(r0.z, max(r0.z, 0.75), sweatWet);
  r2.xyz *= lerp(1.0, 0.9, sweatWet);
  // 4) Translucidez: la pasada de luz del motor usa este valor para el
  //    scattering subsuperficial. Subirlo da piel con luz "dentro",
  //    terminador rojizo y sombras menos duras.
  r1.w = saturate(r1.w * sssBoost) * (1.0 - fhMask);
  if (pesEdgeAll > 0.0) {   // (20/10) borde de la mascara generica: igual que la pieza del cuello (shader del cuerpo)
    r2.xyz = lerp(r2.xyz, pesAlb0 * pesCav, pesEdgeAll);
    r0.z = lerp(r0.z, saturate(pesSpec0 * specScale) * lerp(1.0, pesCav, 0.8), pesEdgeAll);
    r0.w = lerp(r0.w, pesRough0, pesEdgeAll);
    r1.w = lerp(r1.w, pesTrans0, pesEdgeAll);
  }
  // ===================================================

  // ---- velo de borde (fresnel) del material ----
  r0.x = g_psMaterial.m_materials[2].w * r0.z;
  r0.y = dot(v3.xyz, v3.xyz);
  r0.y = rsqrt(r0.y);
  r3.xyz = v3.xyz * r0.yyy;
  r0.y = dot(r1.xyz, r3.xyz);
  r0.y = saturate(1 + -r0.y);
  r0.y = log2(r0.y);
  r0.y = g_psMaterial.m_materials[1].x * r0.y;
  r0.y = exp2(r0.y);
  r0.x = r0.x * r0.y;
  float3 stVeil = g_psMaterial.m_materials[2].xyz;
  {   // (24/10 c) tono de piel propio del jugador (skintone.cpp + pes_skintone.hlsli): Oklab, tambien el velo de borde; el
      // VS faab87ee (reemplazo propio) dice el jugador en o6.w
#ifdef PES_HL_ACC
    {   // (24/10 e) medicion: alto de la cara en pantalla (lugar 6: y minimo, y maximo) y color de la franja de abajo (el
        // cuello, 12 % inferior del alto del cuadro anterior: z192 / w192) -> lugar 5
      uint yq = (uint)clamp(v0.y, 0.0, 8191.0), d;
      gHlAcc.InterlockedMax(6u * 16 + 0, 8192u - yq, d); gHlAcc.InterlockedMax(6u * 16 + 4, yq, d); gHlAcc.InterlockedAdd(6u * 16 + 12, 1u, d);
      float4 FB = IniParams.Load(int2(192, 0));   // z = y minimo, w = y maximo de la cara (cuadro anterior; skintone.cpp)
      if (FB.w > FB.z + 8.0 && v0.y > FB.z + 0.88 * (FB.w - FB.z)) PesSkinAccum(gHlAcc, 5u, r2.xyz);
    }
#endif
    int stE = PesSkinEntry(v6.w);
    if (stE >= 0) { float3 stP = PesSkinParams(stE); r2.xyz = PesSkinApply(r2.xyz, stP); stVeil = PesSkinApply(stVeil, stP); }
  }
  r3.xyz = stVeil + -r2.xyz;
  o0.xyz = r0.xxx * r3.xyz + r2.xyz;
  if (P147.y > 0.5) o0.xyz = lerp(o0.xyz, float3(1.0, 0.0, 0.0), pesEdge);   // (20/10) diagnostico y147: rojo = franja del borde de la mascara generica

  // ---- rugosidad codificada ----
  r0.x = r0.w * 0.0361702144 + 0.0904255286;
  r0.x = r0.x * r0.w + 0.180851057;
  r0.x = r0.x * r0.w + 0.271276593;
  r0.x = r0.x * r0.w + 0.271276593;
  o2.x = r0.x * r0.w + 0.150000006;

  // === PES21 cejas con hebras (18/10): esta cara escribe, para cada celda con semilla, la posicion y normal animadas
  // (espacio de vista), la direccion de crecimiento y el color, en el buffer que lee el dibujo de hebras de la DLL.
  // x74 on/off | w75 = -1 invierte la direccion (si las cejas crecen al reves)
  { float4 P74 = IniParams.Load(int2(74,0));
    { float dbgT = IniParams.Load(int2(75,0)).z;   // diagnostico 2, prueba de paso: verde = el shader llega hasta aca pero el bloque de cejas no entra
      if (dbgT > 1.5 && dbgT < 2.5 && !(enhance && P74.x > 0.5)) o0.xyz = float3(0.0, 1.0, 0.0); }
    if (enhance && P74.x > 0.5) {
      // mapa de pelo (automatico o hair\<id>.png del usuario): R densidad, G largo, B angulo de peinado, A zona (1 = ceja)
      float2 cell = floor(v2.xy * 256.0);
      float4 sm = pesSeedMap.SampleLevel(g_samplerLinear_Clamp_s, (cell + 0.5) / 256.0, 0);   // (s11: el s8 del juego puede no estar enlazado en este dibujo)
      bool isBrowZ = sm.a > 0.95;
      float4 P76s = IniParams.Load(int2(99,0));   // x99 = pelo con hebras en el cuero cabelludo on/off, y99 densidad (19/10 c: era 76, chocaba con la dispersion v2)
      float densMul = pesHairMisc.x > 0.5 ? (isBrowZ ? pesHairB.y * pesHairBrow.y : pesHairB.y) : 1.0;
      if (!isBrowZ) densMul *= (P76s.x > 0.5 ? max(P76s.y, 0.0) * 0.35 : 0.0);   // cuero cabelludo: solo si esta prendido, menos denso (las tarjetas siguen)
      bool seedHere = sm.r > 0.01 && h21(cell + 0.37) < saturate(sm.r * densMul);
      float dbg75 = IniParams.Load(int2(75,0)).z;
      // diagnostico 2: semilla = magenta, zona de ceja sin semilla = azul; el resto de la cara muestra el mapa crudo
      // (rojo = densidad, verde = alfa; azul oscuro parejo = t26 vacio o sin enlazar)
      if (dbg75 > 1.5 && dbg75 < 2.5) { if (seedHere) o0.xyz = float3(1.0, 0.0, 1.0); else o0.xyz = float3(FB.a * 0.3, saturate(sm.r * 4.0), sm.a > 0.95 ? 1.0 : 0.15); }   // (19/10: apagado, para que el magenta sea solo semilla)   // rojo = t24 (quinta textura) llega; verde = densidad del mapa de pelo; azul = alfa de t26
      if (seedHere) {
        float2 jit = float2(h21(cell + 0.37), h21(cell + 1.7)) - 0.5;
        float2 suv = (cell + 0.5 + jit * 0.6) / 256.0;
        float fp = max(fwidth(v2.x), fwidth(v2.y)) * 2048.0;         // texeles por pixel
        float rad = max(1.2, fp * 0.8);
        if (length((v2.xy - suv) * 2048.0) < rad) {
          // 18/10 b: semillas compactas: cada una toma el siguiente lugar libre (contador en los argumentos del dibujo)
          uint perS = (uint)clamp(P74.y, 1.0, 3.0);
          uint slotS; pesSeedArgs.InterlockedAdd(4, perS, slotS);
          uint idx = slotS / perS;
          float3 dirC = -normalize(v3.xyz);
          // v0 es SV_Position: su w es 1/w (ERROR corregido 18/10: antes se usaba como si fuera la profundidad y las
          // hebras nacian a medio camino entre la camara y la cara)
          float depthV = 1.0 / max(v0.w, 1e-6);
          float3 posV = dirC * (depthV / max(abs(dirC.z), 1e-4));
          float angH = sm.z * 6.2831853;
          float2 d2 = float2(cos(angH), sin(angH));
          float flipD = IniParams.Load(int2(75,0)).w < -0.5 ? -1.0 : 1.0;
          float3 dirV = normalize(v4.xyz) * d2.x * flipD - normalize(v5.xyz) * d2.y * flipD;
          PesSeed sd;
          sd.posV = float4(posV, 1.0);
          sd.nrmV = float4(normalize(r1.xyz), isBrowZ ? 0.0 : 1.0);   // w = 1: semilla de cuero cabelludo (pelo), 0: ceja
          sd.dirV = float4(normalize(dirV + 1e-5), sm.y);
          sd.col = float4(o0.xyz, asfloat(f32tof16(pesMotionRaw.x) | (f32tof16(pesMotionRaw.y) << 16)));
          if (idx < 65536u) { pesSeeds[idx] = sd; pesSeedArgs.InterlockedMax(0, 24u); }
          if (dbg75 > 1.5) o0.xyz = float3(0.0, 1.0, 1.0);   // diagnostico 2: pixel que escribio la semilla en cian
        }
      }
    } }
  // ---- empaquetado normal octaedrica + translucidez ----
  r0.x = r1.w * 63 + 0.5;
  r0.y = abs(r1.x) + abs(r1.y);
  r0.y = r0.y + abs(r1.z);
  r0.yw = r1.xy / r0.yy;
  r1.x = cmp(0 < -r1.z);
  r1.yz = float2(1,1) + -abs(r0.wy);
  r2.xy = cmp(r0.yw >= float2(0,0));
  r2.xy = r2.xy ? float2(1,1) : float2(-1,-1);
  r1.yz = r2.xy * r1.yz;
  r0.yw = r1.xx ? r0.yw : r1.yz;
  r0.yw = r0.yw * float2(0.5,0.5) + float2(0.5,0.5);
  r0.yw = float2(2047,2047) * r0.yw;
  r0.xyw = (uint3)r0.xyw;
  r1.xy = (uint2)r0.yw >> int2(1,1);
  bitmask.y = ((~(-1 << 1)) << 1) & 0xffffffff;  r0.y = (((uint)r0.y << 1) & bitmask.y) | ((uint)0 & ~bitmask.y);
  bitmask.y = ((~(-1 << 1)) << 0) & 0xffffffff;  r0.y = (((uint)r0.w << 0) & bitmask.y) | ((uint)r0.y & ~bitmask.y);
  r1.xy = (uint2)r1.xy;
  r1.w = (uint)r0.y;
  o1.xyw = float3(0.000977517106,0.000977517106,0.333333343) * r1.xyw;
  r0.x = (uint)r0.x << 4;
  r0.x = (uint)r0.x;
  o1.z = 0.000977517106 * r0.x;
  o0.w = 1;
  if (pesMicroDbg.x >= 0) o0.xyz = pesMicroDbg;
  o2.y = r0.z;
  o2.z = enhance ? 0.98039 : 1;   // marca de piel para los shaders de luz (250/255)
  o2.w = g_psMaterial.m_materials[0].x;
  o3.zw = float2(0,0);
  o4.xyzw = float4(0,0,0,1);
  return;
}
