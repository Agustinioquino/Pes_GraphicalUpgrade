  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2)
{
  float4 r0, r1, r2, r3, r4, r5;
  float4 v0 = 0;
  v0.xyzw = i_texcoord0;
  float4 v1 = 0;
  v1.xyzw = i_texcoord1;
  float4 v2 = 0;
  v2.xyzw = i_pos;
  r0.xy = ((v1.xyxx / v1.wwww)).xy;
  float4 skSt = g_SpecularTexture.Sample(g_samplerPoint_Wrap_s, r0.xy);   // (04/11) piel: z marca, x aspereza, y intensidad
  r1.xyzw = ((g_NormalTexture.Sample(g_samplerPoint_Wrap_s, (r0.xyxx).xy)).xyzw).xyzw;
  r1.xyw = ((r1.xyxw * float4(asfloat(0x447fc000U), asfloat(0x447fc000U), 0.0, asfloat(0x40400000U)))).xyw;

  r0.w = (float4((uint4(int4(r1.wwww)) >> (uint4(1U, 1U, 1U, 1U) & 31)))).w;
  r0.z = (float4(0.0, 0.0, (float)((int)r0.w | (int)r0.z), 0.0)).z;

  r1.xy = (float4(uint4(int4(r0.zwzz)))).xy;
  r0.zw = ((r1.xxxy * float4(0.0, 0.0, asfloat(0x3a801002U), asfloat(0x3a801002U)) + float4(0.0, 0.0, asfloat(0xbf800000U), asfloat(0xbf800000U)))).zw;
  r1.xyw = ((-(abs(r0.zwzz)) + float4(asfloat(0x3f800000U), asfloat(0x3f800000U), 0.0, asfloat(0x3f800000U)))).xyw;
  r2.z = ((-(abs(r0.wwww)) + r1.xxxx)).z;
  r1.x = (((r2.zzzz >= float4(0.0, 0.0, 0.0, 0.0)) ? float4(-1.0, -1.0, -1.0, -1.0) : float4(0.0, 0.0, 0.0, 0.0))).x;
  r3.xy = (((r0.zwzz >= float4(0.0, 0.0, 0.0, 0.0)) ? float4(-1.0, -1.0, -1.0, -1.0) : float4(0.0, 0.0, 0.0, 0.0))).xy;
  r3.xy = (((r3.xyxx != 0) ? float4(asfloat(0x3f800000U), asfloat(0x3f800000U), 0.0, 0.0) : float4(asfloat(0xbf800000U), asfloat(0xbf800000U), 0.0, 0.0))).xy;
  r1.yw = ((r1.yyyw * r3.xxxy)).yw;
  r2.xy = (((r1.xxxx != 0) ? r0.zwzz : r1.ywyy)).xy;
  r0.z = ((dot((r2.xyzx).xyz, (r2.xyzx).xyz)).xxxx).z;
  r0.z = (rsqrt(r0.zzzz)).z;
  r2.xyz = ((r0.zzzz * r2.xyzx)).xyz;
  r0.z = ((r1.zzzz * float4(asfloat(0x447fc000U), asfloat(0x447fc000U), asfloat(0x447fc000U), asfloat(0x447fc000U)) + float4(asfloat(0x3f000000U), asfloat(0x3f000000U), asfloat(0x3f000000U), asfloat(0x3f000000U)))).z;
  r0.z = (float4(uint4(r0.zzzz))).z;
  r0.w = (float4(0.0, 0.0, 0.0, (float)((int)r0.z & 15))).w;
  int skMat = (int)r0.w;
  r1.x = (((int4(r0.wwww) == int4(1, 1, 1, 1)) ? float4(-1.0, -1.0, -1.0, -1.0) : float4(0.0, 0.0, 0.0, 0.0))).x;
  if ((r1.xxxx).x != 0) {
    r3.y = (float4(0.0, 0.0, 0.0, 0.0)).y;
    r4.x = (float4(0.0, 0.0, 0.0, 0.0)).x;
  } else {
    r0.z = (float4((uint4(int4(r0.zzzz)) >> (uint4(4U, 4U, 4U, 4U) & 31)))).z;
    r1.y = (((int4(r0.wwww) == int4(3, 3, 3, 3)) ? float4(-1.0, -1.0, -1.0, -1.0) : float4(0.0, 0.0, 0.0, 0.0))).y;
    if ((r1.yyyy).x != 0) {
      r1.y = (float4(uint4(int4(r0.zzzz)))).y;
      r3.y = ((r1.yyyy * float4(asfloat(0x3c820821U), asfloat(0x3c820821U), asfloat(0x3c820821U), asfloat(0x3c820821U)))).y;
      r4.x = (float4(0.0, 0.0, 0.0, 0.0)).x;
    } else {
      r4.x = ((g_SpecularTexture.Sample(g_samplerPoint_Wrap_s, (r0.xyxx).xy)).wxyz).x;
      r0.z = (float4(uint4(int4(r0.zzzz)))).z;
      r3.y = ((r0.zzzz * float4(asfloat(0x3c820821U), asfloat(0x3c820821U), asfloat(0x3c820821U), asfloat(0x3c820821U)))).y;
    }
  }
  r0.x = ((g_DepthTexture.Sample(g_samplerPoint_Wrap_s, (r0.xyxx).xy)).xyzw).x;
  r0.yz = ((v0.xxyx / v0.wwww)).yz;
  r0.yz = ((r0.yyzy + -(g_psScene.m_cameraCenterOffset.xxyx))).yz;
  r0.yz = ((r0.yyzy * g_psScene.m_projectionParam.xxyx)).yz;
  r0.x = ((r0.xxxx + -(g_psScene.m_projectionParam.wwww))).x;
  r5.z = ((g_psScene.m_projectionParam.zzzz / r0.xxxx)).z;
  r5.xy = ((r0.yzyy * r5.zzzz)).xy;

  r0.xyz = ((-(r5.xyzx) + g_psLight.m_lightSourcePos.xyzx)).xyz;

  float3 skN = normalize(r2.xyw);

  o0.w = (float4(asfloat(0x3f800000U), asfloat(0x3f800000U), asfloat(0x3f800000U), asfloat(0x3f800000U))).w;
  o1.xyzw = (float4(0.0, 0.0, 0.0, asfloat(0x3f800000U))).xyzw;
  if (skOn) o1.xyz = min(PesSkSpec(0.0.xxx, skN, skV, skL, skSt.x, skSt.y, skC * g_psLight.m_lightSourceParam3.y, 1.0) * skFS, 10000.0.xxx);   // (04/11) brillo de piel
  o2.xyzw = (float4(asfloat(0x3f800000U), asfloat(0x3f800000U), asfloat(0x3f800000U), asfloat(0x3f800000U))).xyzw;
