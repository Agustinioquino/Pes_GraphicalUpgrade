        float2x2 Ai = float2x2(dyu2.y, -dxu2.y, -dyu2.x, dxu2.x) / det;      // inversa de A = [dxu; dyu]

        float2x2 B2 = float2x2(bxx, bxy, bxy, byy);

        float lmax = tr * 0.5 + disc, lmin = max(tr * 0.5 - disc, 1e-12);

        float sens = max(WR.y, 0.01);
        float amtC = smoothstep(1.0 + sens * 0.5, 1.0 + sens * 2.5, aniso);

        float curvW = saturate((dxn.x + dxn.y) / max(fpW * 0.002, 1e-4) * 0.08);
