float3 PESDecN(float2 uv) {
  float4 nb = inGBufferNormal.SampleLevel(g_samplerPoint_Clamp_s, uv, 0);
  uint3 q = (uint3)(float3(1023,1023,3) * nb.xyw);
  float2 e = float2((q.x << 1) | (q.z >> 1), (q.y << 1) | (q.z & 1)) * 0.00097703957 - 1;
  float3 n = float3(e.xy, 1 - abs(e.x) - abs(e.y));
  if (n.z < 0) n.xy = (1 - abs(e.yx)) * (e.xy >= 0 ? 1 : -1);
  return normalize(n);
}

// PES21 (14/11, Agustin: "pq aveces el pasto se pone rojo, y no toma bien el color de el sol en las puntas para la translucidez") x512
// CONTRALUZ CON EL COLOR DE LA HOJA: la luz que atraviesa una hoja sale del color de la hoja (la clorofila se come el rojo y el azul), no del
// color del sol. La luz que sumaban y57, x414 y x415 a contraluz era el color del sol (a las 15 h R/G x1.29) por el albedo amarillo verdoso:
// con el sol de frente el pasto pasaba a amarillo, marron y rojo. Aca esa luz toma el BRILLO del sol (su luminancia), un cuarto de su tono y
// el verde de la propia hoja (el albedo, que despues la vuelve a multiplicar: lo que pasa por la hoja es mas verde que lo que rebota), acotado
// a los verdes (nunca mas rojo que 0.95 del verde ni mas azul que 0.6). Mismo brillo que antes. k 0 = el color del sol, como antes
float3 PESLeafTrans(float3 sunC, float3 alb, float k) {
  const float3 LWt = float3(0.2126, 0.7152, 0.0722);
  float lS = dot(sunC, LWt);
  if (k <= 0.001 || lS <= 1e-6) return sunC;
  float3 a = alb / max(alb.g, 1e-4);
  a = clamp(a, float3(0.35, 1.0, 0.10), float3(0.95, 1.0, 0.60));
  float3 t = lerp(float3(1.0, 1.0, 1.0), a, 0.5) * lerp(float3(1.0, 1.0, 1.0), sunC / lS, 0.25);
  t *= lS / max(dot(t, LWt), 1e-6);
  return lerp(sunC, t, saturate(k));
}
// x512 el halo del sol: la hoja se enciende en un lobulo suave alrededor del sol (mas ancho con w512) en vez de una potencia fija del angulo
float PESSunLobe(float fwd, float w) {
  return 1.2 * exp((saturate(fwd) - 1.0) / clamp(w, 0.08, 1.0)) * saturate(fwd * 4.0);
}
// PES21 (14/11) y513 SOMBRAS DE CONTACTO ENTRE HOJAS: desde el punto se camina unos cm hacia el sol en la pantalla; si la profundidad de
// algo delante (otra hoja, la pelota, un botin) tapa el camino a menos de 6 cm, la hoja queda a su sombra. Solo cerca de la camara (la sombra
// del sol del juego no ve hojas sueltas). 1 = al sol, 0 = tapada
float PESGrassContact(float3 P0, float3 L, float len, float jit) {
  float occ = 0.0;
  [unroll] for (int i = 0; i < 6; i++) {
    float t = len * (i + 0.5 + 0.5 * jit) / 6.0;
    float3 Q = P0 + L * t;
    if (Q.z < 0.06) break;
    float2 sQ = Q.xy / (g_psScene.m_projectionParam.xy * Q.z) + g_psScene.m_cameraCenterOffset.xy;
    float2 uvQ = (sQ + float2(1, -1)) * float2(0.5, -0.5);
    if (any(uvQ < 0.0) || any(uvQ > 1.0)) break;
    float dz = Q.z - PESLinZ(uvQ);
    occ = max(occ, saturate((dz - 0.004) / 0.01) * saturate((0.06 - dz) / 0.03) * (1.0 - 0.5 * t / len));
  }
  return 1.0 - occ;
}

  // 2) oclusion especular (Lagarde): el brillo se apaga en los huecos segun la rugosidad y el angulo de vista,
  //    en vez de oscurecer todo igual. x48 multirebote, y48 oclusion especular, z48 fuerza total, w48 on/off
  float4 AOP = IniParams.Load(int2(48,0));
  // PES21 rasgos de la cara (20/10, x124): comparando con el juego original, la cara del mod salia mas clara y "sin rasgos":
  // el multirebote del GTAO aclara la oclusion en superficies claras (en la piel una cuenca de 0.5 pasaba a 0.72), y el SSGI
  // rellena las cuencas. En la piel: oclusion mas honda (potencia) y sin multirebote, segun x124 (0 = como antes).
  float faceK = (cmpOn && skinC) ? saturate(IniParams.Load(int2(124,0)).x) : 0.0;
  if (faceK > 0.0) { aoRaw = pow(aoRaw, 1.0 + 0.8 * faceK); occRGB = pow(max(occRGB, 1e-4), 1.0 + 0.8 * faceK); }
  if (cmpOn && AOP.w > 0.5) {
    float3 diffC = r7.xyz * r2.xyz;
    float3 specC = r1.xyz;
    // (02/11) pasto a la sombra del estadio: el reflejo del cielo en las gotas y puntas (rocio x334 / x329 bajan la rugosidad en el
    // G-buffer) no se atenuaba y quedaban puntos blancos dentro de la sombra (captura de Agustin 10:25 PC). Bajo el techo o la tribuna
    // se ve mucho menos cielo: el reflejo baja al 40 % donde no da el sol
    if (tyC == 3) specC *= lerp(0.4, 1.0, saturate(inShadow.SampleLevel(g_samplerPoint_Clamp_s, uvC, 0).x));
    float aoK = lerp(1.0, aoRaw, saturate(AOP.z));
    float3 alb = saturate(r7.xyz);
    float3 ma = 2.0404 * alb - 0.3324, mb = -4.7951 * alb + 0.6417, mc = 2.7552 * alb + 0.6903;
    float3 multi = max(aoK.xxx, ((aoK * ma + mb) * aoK + mc) * aoK);
    float3 aoD = lerp(occRGB, occRGB * multi / max(aoRaw, 1e-3), saturate(AOP.x) * (1.0 - faceK));
    aoD = min(aoD, 1.0);
    float aoS = aoK;
    if (geoOk) {
      float ndvS = saturate(dot(Nsk, -Vsk));
      float roughS = saturate(gmC.x);
      float so = saturate(pow(max(ndvS + aoK, 1e-4), exp2(-16.0 * roughS - 1.0)) - 1.0 + aoK);
      aoS = lerp(aoK, so, saturate(AOP.y));
    }
    // PES21 micro-sombras (18/10, Naughty Dog "micro-shadowing"): los poros, pliegues y la trama de la tela se hacen
    // sombra a si mismos con la luz rasante del sol. La apertura la da la oclusion ambiental; solo del lado iluminado.
    // x88 fuerza, y88 on/off, z88 apertura: 0 = oclusion de pantalla (cambia con la camara), 1 = relieve (31/10): el angulo
    // entre la normal de detalle (poros, pliegues, trama) y la geometrica de la profundidad; no depende de la camara
    // (13/11, pedido del hilo del pasto) LA MICRO SOMBRA VIEJA SIGUE para todo lo que no es piel ni tela (tyC != 0: pasto, pelota,
    // estadio) hasta que tengan su version v4: el pasto de cerca depende de ella para la profundidad entre hojas. z88 ahora es la
    // suavidad de la v4: aca queda siempre el modo relieve (el que estaba por defecto)
    { float4 MS = IniParams.Load(int2(88,0));
      if (tyC != 0 && MS.y > 0.5 && MS.x > 0.0 && geoOk && dot(pLc_dir.xyz, pLc_dir.xyz) > 1e-6) {
        float ndlM = dot(Nsk, normalize(pLc_dir.xyz));
        float aper = 2.0 * aoRaw * aoRaw;
        if (true) {
          // (01/11) relieve v2: la normal de detalle contra el PROMEDIO de las normales del G-buffer en un anillo de ~1.2 cm del
          // mundo (no contra la normal de la profundidad). La de la profundidad salia a facetas (cuadrados de 2x2 y triangulos de
          // mallas con pocos poligonos, como la mano) y cada faceta se leia como un pliegue hondo. El promedio de un anillo
          // simetrico quita poros, trama y pliegues finos y deja la curvatura suave; sigue sin depender de la camara.
          float3 Nc0 = PESDecN(uvC);
          float pxmM = 1.0 / max(2.0 * abs(g_psScene.m_projectionParam.x) * zsk * g_psSystem.m_renderBuffer.z, 1e-5);
          float rM = clamp(0.012 * pxmM, 1.5, 16.0);
          float3 nAcc = 0; float nW = 0;
          // (02/11) solo vecinos sobre el plano tangente del pixel (tolerancia 35 % del radio del anillo, ~4 mm): antes el corte era
          // 3 % de la profundidad + 2 cm (~6 cm en un primer plano) y en el cuello entraban normales del menton y de la malla del
          // torso -> linea dura de sombra bajando del menton y franja de borde recto en la union (Agustin: "es por las micro sombras")
          float3 NvC = float3(Nc0.xy, -Nc0.z);
          float tolM = 0.35 * rM / pxmM + 0.001;
          [unroll] for (int mi = 0; mi < 8; mi++) {
            float angM = mi * 0.78539816 + 0.3927;
            float2 uvM = uvC + float2(cos(angM), sin(angM)) * rM * g_psSystem.m_renderBuffer.zw;
            float zM = PESLinZ(uvM);
            float3 Pm = float3(((uvM * float2(2, -2) + float2(-1, 1)) - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zM, zM);
            float wM = abs(dot(Pm - Pgeo, NvC)) < tolM ? 1.0 : 0.0;
            nAcc += PESDecN(uvM) * wM; nW += wM;
          }
          float sB = nW > 2.5 ? pow(saturate(dot(Nc0, normalize(nAcc + 1e-6))), 6.0) : 1.0;
          aper = 2.0 * sB * sB;
        }
        float micro = saturate(abs(ndlM) + aper - 1.0);
        float litSide = saturate(ndlM * 4.0);                         // solo donde pega el sol
        // (01/11) piso 0.4: un pliegue nunca queda negro, y en la piel el brillo baja igual que la difusa (antes quedaba el reflejo
        // azulado del cielo sobre poca difusa rojiza: nudillos y dedos violetas)
        // (12/11, piel v2 w124) en la piel la micro sombra sale de poros y del normal map fino: solo mientras se resuelven (hasta
        // 0.8 mm de piel por pixel entera, nada a 2 mm). Mas lejos comparaba normales que el mip ya no muestra bien y dejaba manchas
        // en la franja del terminador que cambiaban con la distancia (mejillas, orejas)
        float msFar = (skinC && IniParams.Load(int2(124,0)).w > 0.5) ? saturate((2.0 - 2000.0 * abs(g_psScene.m_projectionParam.x) * zsk * g_psSystem.m_renderBuffer.z) / 1.2) : 1.0;
        float msK = lerp(1.0, max(micro, 0.4), saturate(MS.x) * litSide * msFar);
        diffC *= msK; specC *= skinC ? msK : lerp(1.0, msK, 0.5);
      } }
    // (13/11) MICRO SOMBRAS v4 (pes_microsh.hlsli): sombra de relieve con direccion, hacia el sol, para piel, tela y todo lo
    // que no es pasto ni pelota. x88 fuerza, y88 on/off, w88 largo (0 = 1). z88 ya no se usa. Solo sobre la parte de la luz que
    // viene del sol (su color x N.L x sombra del juego contra el total): en la sombra del estadio o de la cabeza no hace nada.
    // Largo por material (piel 6 mm, lo demas 15 mm), nunca menos de 5 pixeles: de cerca marca poros y trama, mas lejos arrugas,
    // pliegues y costuras, sin puntos sueltos; pasados 3 largos se apaga. Piso por material: en la piel la sombra chica nunca es
    // negra y deja pasar rojo (la luz cruza un poro por debajo), en la tela baja pareja
    { float4 MS = IniParams.Load(int2(88,0));
      if (MS.y > 0.5 && MS.x > 0.0 && geoOk && tyC == 0 && !mouthM && zsk < 25.0 && dot(pLc_dir.xyz, pLc_dir.xyz) > 1e-6) {   // (13/11) nada en la boca (regla)
        float3 LdS = normalize(pLc_dir.xyz);
        float ndlM = dot(Nsk, LdS);
        float sunSh = saturate(inShadow.SampleLevel(g_samplerPoint_Clamp_s, uvC, 0).x);
        float3 sunD = max(pLc_color.xyz, 0.0) * saturate(ndlM) * sunSh;
        float sunF = saturate(dot(sunD, float3(0.2126, 0.7152, 0.0722)) / max(dot(r2.xyz, float3(0.2126, 0.7152, 0.0722)), 1e-4)) * saturate(ndlM * 6.0) * sunSh;
        if (sunF > 0.01) {
          float pxmM = 1.0 / max(2.0 * abs(g_psScene.m_projectionParam.x) * zsk * g_psSystem.m_renderBuffer.z, 1e-5);   // pixeles por metro
          float rBase = (skinC ? 0.006 : 0.015) * (MS.w > 0.001 ? clamp(MS.w, 0.25, 4.0) : 1.0);
          float rM = max(rBase, 5.0 / pxmM);
          float farK = saturate((3.0 * rBase - rM) / (1.5 * rBase) + 1.0) * saturate((25.0 - zsk) / 5.0);
          if (farK > 0.01) {
            // forma lisa: promedio de las normales en un anillo del largo del camino, solo vecinos sobre el plano del pixel
            float3 Nc0 = PESDecN(uvC);
            float3 NvC = float3(Nc0.xy, -Nc0.z);
            float rPx = clamp(rM * pxmM, 2.0, 24.0);
            float tolM = 0.35 * rM + 0.001;
            float3 nAcc = 0; float nW = 0;
            [unroll] for (int mi = 0; mi < 8; mi++) {
              float angM = mi * 0.78539816 + 0.3927;
              float2 uvM = uvC + float2(cos(angM), sin(angM)) * rPx * g_psSystem.m_renderBuffer.zw;
              float zM = PESLinZ(uvM);
              float3 Pm = float3(((uvM * float2(2, -2) + float2(-1, 1)) - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zM, zM);
              float wM = abs(dot(Pm - Pgeo, NvC)) < tolM ? 1.0 : 0.0;
              nAcc += PESDecN(uvM) * wM; nW += wM;
            }
            if (nW > 2.5) {
              float3 NsM = normalize(nAcc + Nc0); NsM.z = -NsM.z;
              // ESTABLE: el ruido por pixel solo con el TAA de la DLL prendido (x454), que lo promedia; sin TAA paso fijo (sin hormigueo)
              float jitM = IniParams.Load(int2(454,0)).x > 0.5 ? frac(52.9829189 * frac(dot(v0.xy, float2(0.06711056, 0.00583715)))) : 0.5;
              float softM = MS.z > 0.001 ? clamp(MS.z, 0.25, 3.0) : 1.0;   // z88 suavidad de la penumbra (0 = 1)
              float visM = PESMicroShadowDir(uvC, Pgeo, Nsk, NsM, LdS, rM, jitM, softM);
              float3 floorM = skinC ? float3(0.62, 0.44, 0.40) : float3(0.35, 0.35, 0.35);
              // REACTIVO al sudor / lluvia: la pelicula de agua llena el poro y alisa la piel (la rugosidad del G-buffer baja de ~0.42
              // seca a ~0.2 mojada): con la piel mojada la sombra del poro se acorta hasta un 60 %; arrugas y pliegues grandes quedan
              float wetM = skinC ? saturate((0.38 - saturate(gmC.x)) / 0.18) : 0.0;
              float kM = saturate(MS.x) * farK * sunF * (1.0 - 0.6 * wetM);
              diffC *= lerp(1.0, lerp(floorM, 1.0, visM), kM);
              specC *= lerp(1.0, lerp(0.25, 1.0, visM), kM);
            }
          }
        }
      } }
    r0.xyw = diffC * aoD + specC * aoS;
  } else {
    r1.xyz = r7.xyz * r2.xyz + r1.xyz;
    r0.xyw = r1.xyz * occRGB;
  }
  { // === PES21 reflejos del cielo (21/10, 151) ===
    // Brillo ambiental: cada superficie refleja el cielo fisico (panorama que arma la DLL con el mismo modelo que el cielo,
    // capa nitida para lo liso y desenfocada para lo aspero) o, hacia abajo, el pasto iluminado. Antes no habia reflejo de
    // entorno salvo los "matcap" fijos de algunos materiales: botines, pelota, piel transpirada y carteles no veian el cielo.
    // Fresnel y rugosidad con la BRDF de entorno (Karis); en los huecos se apaga con oclusion especular (Lagarde) y con la
    // visibilidad del cielo (oclusion ambiental). 151: x on | y fuerza | z oclusion especular | w pasto en los reflejos
    float4 RS = IniParams.Load(int2(151,0)), SKr = IniParams.Load(int2(150,0));
    if (cmpOn && RS.x > 0.5 && geoOk && SKr.y > 0.01 && (tyC != 1 || pesBallSkyK > 0.001) && tyC != 3 && gmC.y > 0.004 && zsk < 300.0) {
      float3 Rv = reflect(Vsk, Nsk);
      float3 vXr = float3(g_psScene.m_view._m00, g_psScene.m_view._m01, g_psScene.m_view._m02);
      float3 vYr = float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12);
      float3 vZr = float3(g_psScene.m_view._m20, g_psScene.m_view._m21, g_psScene.m_view._m22);
      float3 Rw = normalize(float3(dot(Rv, vXr), dot(Rv, vYr), dot(Rv, vZr)));
      float roughE = saturate(gmC.x);
      float uE = atan2(Rw.z, Rw.x) * 0.15915494 + 0.5;
      float vE = clamp(sqrt(saturate(asin(saturate(Rw.y)) / 1.5707963)), 0.5 / 16.0, 1.0 - 0.5 / 16.0);
      float3 eS = PESSkyEnv.SampleLevel(g_samplerLinear_Wrap_s, float3(uE, vE, 0), 0).xyz;
      float3 eB = PESSkyEnv.SampleLevel(g_samplerLinear_Wrap_s, float3(uE, vE, 1), 0).xyz;
      float3 env = lerp(eS, eB, saturate(roughE * 1.7));
      // hacia abajo: el pasto al sol (albedo tipico x (sol + cielo)); lo aspero ya lo trae la capa desenfocada
      float4 SW = IniParams.Load(int2(111,0));
      float3 gnd = float3(0.10, 0.17, 0.05) * (saturate(SW.y) + SKr.w) * (RS.w > 0 ? saturate(RS.w) : 1.0);
      env = lerp(env, lerp(gnd, eB, saturate(roughE * 1.2)), saturate(-Rw.y * 10.0));
      // BRDF de entorno (aprox. analitica de Karis): especular del G-buffer 0.5 = F0 0.04
      float NdVe = saturate(dot(Nsk, -Vsk));
      // (27/10) arquero de Agustin: la tela oscura salia gris lavada (como mojada) con el reflejo del cielo entero encima. La
      // tela de punto no refleja como una superficie lisa (las fibras desparraman y tapan el brillo): en los kits va a 1/4.
      // La piel queda fija como la dejo Agustin (y151 = 0.1, 27/10: "me gusta como esta ahora en la piel"), sin depender del slider
      bool kitE = (tyC == 0) && (abs(gmC.z - 0.9607843) < 0.002);
      float F0 = skinC ? 0.028 : 0.08 * saturate(gmC.y);
      float4 rr = roughE * float4(-1.0, -0.0275, -0.572, 0.022) + float4(1.0, 0.0425, 1.04, -0.04);
      float a004 = min(rr.x * rr.x, exp2(-9.28 * NdVe)) * rr.x + rr.y;
      float2 AB = float2(-1.04, 1.04) * a004 + rr.zw;
      float specE = (F0 * AB.x + AB.y) * saturate(gmC.y * 10.0) * (kitE ? 0.25 : 1.0);
      // oclusion especular: Lagarde (cuenca segun rugosidad y angulo) y visibilidad del cielo
      float aoE = saturate(aoRaw);
      float soL = saturate(pow(max(NdVe + aoE, 1e-4), exp2(-16.0 * roughE - 1.0)) - 1.0 + aoE);
      // oclusion de horizonte (Jimenez 2016): con el mapa de normales el reflejo puede apuntar hacia adentro de la superficie
      // real (pliegues de la tela, poros): se apaga contra la normal geometrica, que sale de la profundidad
      float hzK = 1.0;
      { float2 pxh = g_psSystem.m_renderBuffer.zw;
        float zX = PESLinZ(uvC + float2(pxh.x, 0)), zY = PESLinZ(uvC + float2(0, pxh.y));
        float2 s0 = uvC * float2(2, -2) + float2(-1, 1), sX = (uvC + float2(pxh.x, 0)) * float2(2, -2) + float2(-1, 1), sY = (uvC + float2(0, pxh.y)) * float2(2, -2) + float2(-1, 1);
        float3 P0 = float3((s0 - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zsk, zsk);
        float3 PX = float3((sX - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zX, zX);
        float3 PY = float3((sY - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zY, zY);
        float3 Ng = cross(PX - P0, PY - P0);
        if (abs(zX - zsk) < 0.05 * zsk && abs(zY - zsk) < 0.05 * zsk && dot(Ng, Ng) > 1e-12) {
          Ng = normalize(Ng); if (dot(Ng, -Vsk) < 0) Ng = -Ng;
          hzK = saturate(1.0 + 1.1 * dot(Rv, Ng)); hzK *= hzK;
        } }
      float soK = lerp(1.0, soL * lerp(1.0, aoE, 0.5) * hzK, RS.z > 0 ? saturate(RS.z) : 1.0);
      float sunLe = max(dot(max(pLc_color.xyz, 0.0), float3(0.2126, 0.7152, 0.0722)), 0.0) * g_psScene.m_exposure.z;
      r0.xyw += env * sunLe * specE * soK * (skinC ? 0.1 : max(RS.y, 0.0)) * (tyC == 1 ? min(pesBallSkyK, 1.0) : 1.0);
    }
  }
  { // === PES21 (12/11) y436 b) BRILLO DEL CIELO EN EL CESPED: 151 salta el pasto (tipo 3). La cera de las hojas refleja el cielo en angulo
    // rasante: la cancha de lejos se ve un poco plateada / celeste con cielo azul, calida al atardecer. Panorama fisico desenfocado (150 /
    // pes_skyenv), Fresnel de la cera (F0 0.035) con la rugosidad del G-buffer; a la sombra del estadio menos (se ve menos cielo)
    float4 SKg2 = IniParams.Load(int2(150,0)); float kSk2 = clamp(IniParams.Load(int2(436,0)).y, 0.0, 2.0);
    if (cmpOn && geoOk && kSk2 > 0.001 && tyC == 3 && SKg2.y > 0.01 && zsk < 400.0) {
      float3 RvG = reflect(Vsk, Nsk);
      float3 RwG = normalize(float3(dot(RvG, float3(g_psScene.m_view._m00, g_psScene.m_view._m01, g_psScene.m_view._m02)),
                                    dot(RvG, float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12)),
                                    dot(RvG, float3(g_psScene.m_view._m20, g_psScene.m_view._m21, g_psScene.m_view._m22))));
      float rgG = saturate(gmC.x);
      float uSkG = atan2(RwG.z, RwG.x) * 0.15915494 + 0.5;
      float vSkG = clamp(sqrt(saturate(asin(saturate(RwG.y)) / 1.5707963)), 0.5 / 16.0, 1.0 - 0.5 / 16.0);
      float3 envG = lerp(PESSkyEnv.SampleLevel(g_samplerLinear_Wrap_s, float3(uSkG, vSkG, 0), 0).xyz, PESSkyEnv.SampleLevel(g_samplerLinear_Wrap_s, float3(uSkG, vSkG, 1), 0).xyz, saturate(rgG * 1.7));
      envG *= saturate(RwG.y * 6.0 + 0.15);                                       // hacia abajo la hoja ve pasto, no cielo
      float NdVg = saturate(dot(Nsk, -Vsk));
      float4 rrG = rgG * float4(-1.0, -0.0275, -0.572, 0.022) + float4(1.0, 0.0425, 1.04, -0.04);
      float aG = min(rrG.x * rrG.x, exp2(-9.28 * NdVg)) * rrG.x + rrG.y;
      float2 ABg = float2(-1.04, 1.04) * aG + rrG.zw;
      float specG = (0.035 * ABg.x + ABg.y) * saturate(gmC.y * 10.0);
      float shG2 = lerp(0.4, 1.0, saturate(inShadow.SampleLevel(g_samplerPoint_Clamp_s, uvC, 0).x));
      float sunLg = max(dot(max(pLc_color.xyz, 0.0), float3(0.2126, 0.7152, 0.0722)), 0.0) * g_psScene.m_exposure.z;
      r0.xyw += envG * sunLg * specG * shG2 * saturate(aoRaw) * 0.5 * kSk2;
    }
  }
  { // PES21 carteles LED (20/10, 141): los shaders de los carteles marcan el pixel (rugosidad 253/255, especular 1/255, metal 0).
    // Un LED emite: se le suma luz propia en unidades fisicas (luminancia fija x la exposicion del motor), asi se ve encendido
    // en la sombra y de noche brilla mucho mas que lo iluminado (como en TV). El sol que le pega lo lava un poco (y141 = brillo,
    // w141 = caida por angulo: los paneles LED se ven mas oscuros de costado).
    float4 LD = IniParams.Load(int2(141,0));
    if (LD.x > 0.5 && abs(gmC.x - 0.99215686) < 0.0025 && abs(gmC.y - 0.00392157) < 0.0018 && gmC.z < 0.002) {
      float3 albL = inAlbedo.SampleLevel(g_samplerPoint_Clamp_s, uvC, 0).xyz;
      float B = LD.y > 0.0 ? LD.y : 1.0;
      float angK = 1.0;
      if (geoOk) angK = lerp(1.0, pow(saturate(dot(Nsk, -Vsk)), 0.6), saturate(LD.w));
      r0.xyw = r0.xyw * (1.0 - 0.35 * saturate(B)) + albL * (B * 2500.0 * g_psScene.m_exposure.z * angK);
    }
  }
  // === PES21 (14/11, Agustin: "si ademas si quiero cambiar el color del pasto este se volvera de unos colores poco realistas no?") y512
  // VERDE SIEMPRE VERDE: la luz no vuelve marron ni rojo al pasto: el rojo del color final (luz por albedo mas brillo) queda como mucho en
  // 0.92 del verde, o en el del propio albedo x1.08 (la punta seca, la tierra gastada y el color elegido x456 quedan como son: solo se saca
  // lo que agrega la luz calida). El azul como mucho 0.95 del verde; mismo brillo. Lo casi blanco (cal, rocio, chispas) no se toca. 0 = como antes
  {
    float kG = saturate(P512.y);
    if (cmpOn && kG > 0.001 && tyC == 3) {
      const float3 LWg = float3(0.2126, 0.7152, 0.0722);
      float3 cG = max(r0.xyw, 0.0);
      float mxG = max(cG.r, max(cG.g, cG.b)), mnG = min(cG.r, min(cG.g, cG.b));
      float wG = kG * (1.0 - smoothstep(0.75, 0.92, mnG / max(mxG, 1e-6)));
      float lG = dot(cG, LWg);
      if (wG > 0.001 && lG > 1e-6) {
        float rgA = pesAlbG.r / max(pesAlbG.g, 1e-4);
        float3 cN = cG;
        cN.r = min(cN.r, cN.g * max(0.92, rgA * 1.08));
        cN.b = min(cN.b, cN.g * 0.95);
        cN *= lG / max(dot(cN, LWg), 1e-6);
        r0.xyw = lerp(r0.xyw, cN, wG);
      }
    }
  }
  r0.xyw *= pesCovDk;   // (12/11 20:30) x443: el pasto delante de la base de la pelota, algo a su sombra
  r0.z = log2(r0.z);
  r0.z = g_psScene.m_fogParam[2].y * r0.z;
  r0.z = r0.z * 0.693147182 + g_psScene.m_fogParam[2].z;
  r0.z = max(0, r0.z);
  r0.z = min(g_psScene.m_fogParam[2].w, r0.z);
  r1.xz = float2(0.0146484403,0.0146484403) * v1.xx + float2(0.015625,0.015625);
  r1.yw = g_psScene.m_fogParam[1].xx * v1.yy + g_psScene.m_fogParam[1].yy;
  r2.x = 1 + r0.z;
  r2.x = max(0, r2.x);
  r2.x = min(g_psScene.m_fogParam[2].w, r2.x);
  r2.y = floor(r0.z);
  r2.w = floor(r2.x);
  r2.xy = float2(0.03125,0.03125) * r2.yw;
  r2.zw = frac(r2.xy);
  r3.xz = g_psScene.m_fogParam[1].zz * r2.zw;
  r3.yw = floor(r2.xy);
  r2.xz = float2(32,32);
  r2.yw = g_psScene.m_fogParam[1].ww;
  r1.xyzw = r2.xyzw * r3.xyzw + r1.xyzw;
  r0.z = frac(r0.z);
  r2.xyzw = inFogVolume.Sample(g_samplerLinear_Wrap_s, r1.xy).xyzw;
  r1.xyzw = inFogVolume.Sample(g_samplerLinear_Wrap_s, r1.zw).xyzw;
  r3.x = 1 + -r0.z;
  r1.xyzw = r1.xyzw * r0.zzzz;
  r1.xyzw = r3.xxxx * r2.xyzw + r1.xyzw;
  // === PES21 atmosfera con sol dinamico (17/10): la bruma del archivo de luz es azul y tapa el fondo (montanas,
  // tribunas) de un solo color. x54 desatura la bruma lejana, z54 la calienta cuando el sol esta bajo (horizonte
  // naranja en vez de azul), w54 on/off. Solo con sol real (no de noche con torres).
  {
    float4 ATM = IniParams.Load(int2(54,0));
    float3 Ld = pLc_dir.xyz;
    if (cmpOn && ATM.w > 0.5 && dot(Ld, Ld) > 1e-6) {
      Ld = normalize(Ld);
      float3 upV = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
      float sunUp = saturate(abs(dot(Ld, upV)));
      float3 sc = max(pLc_color.xyz, 0.0);
      float kSun = saturate((sc.r / max(sc.b, 1e-4) - 0.85) * 4.0);          // sol de verdad vs. luz fria de torres
      float lowS = 1.0 - smoothstep(0.05, 0.40, sunUp);
      float fl = dot(r1.xyz, float3(0.3, 0.5, 0.2));
      float3 fc = lerp(r1.xyz, fl.xxx, saturate(ATM.x) * kSun);
      float3 warm = float3(1.0, 0.70, 0.46);
      fc = lerp(fc, fl * warm / dot(warm, float3(0.3, 0.5, 0.2)), saturate(ATM.z) * lowS * kSun);
      r1.xyz = fc;
    }
  }
  // === PES21 curva del dia (18/10): densidad de la bruma segun la altura del sol + hora azul ===
  {
    float3 Ld = pLc_dir.xyz;
    if (cmpOn && dot(Ld, Ld) > 1e-6) {
      Ld = normalize(Ld);
      float3 upV = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
      float3 sc = max(pLc_color.xyz, 0.0);
      float kSun = saturate((sc.r / max(sc.b, 1e-4) - 0.85) * 4.0);
      PESDay dy = PESDayCurve(saturate(abs(dot(Ld, upV))), kSun);
      if (dy.on > 0.0) {
        float T = saturate(r1.w);
        float T2 = pow(max(T, 1e-4), max(dy.fog, 0.05));
        r1.xyz *= (1.0 - T2) / max(1.0 - T, 1e-4);
        r1.w = T2;
        r1.xyz = PESBlueTint(r1.xyz, dy.blue * 0.6);
      }
    }
  }
  // niebla/bruma del estadio: es lo que "lava" la imagen; x11 = cuanto se aplica (1 = original)
  float fogK = cmpOn ? saturate(P11.x) : 1.0;
  r0.xyz = r0.xyw * lerp(1.0, r1.w, fogK) + r1.xyz * fogK;
  // === PES21 (31/10) luz volumetrica (335): rayos de sol en el aire del estadio. Se marcha por el rayo de vista desde la
  // camara hasta el pixel (o hasta y335 m) en 16 pasos con ruido que cambia por cuadro (lo limpia el TAA del juego); cada paso
  // prueba el SHADOW MAP DEL SOL igual que los shaders de luz (misma matriz y cascadas), asi la luz entra por los huecos del
  // techo y los jugadores la cortan. Fase Henyey-Greenstein (z335: mas = mas brillo mirando hacia el sol), aire mas denso
  // abajo (w335 m de altura de escala sobre la cancha). x335 = densidad (0 = apagado). w12 = 4 muestra solo esta luz.
  float3 vlDbg = 0;
  {
    float4 VL = IniParams.Load(int2(335,0));
    float3 Ld = pLc_dir.xyz;
    if (cmpOn && VL.x > 0.0 && dot(Ld, Ld) > 1e-6) {
      Ld = normalize(Ld);
      // (31/10) la fase apunta al sol de la LUZ (Agustin: 'prefiero que los rayos sean del sol de la luz'), el mismo de las sombras
      float maxD = VL.y > 1.0 ? VL.y : 150.0;
      float zP = min(PESLinZ(uvC), maxD * 8.0);
      float2 sCv = uvC * float2(2, -2) + float2(-1, 1);
      float3 Pv = float3((sCv - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zP, zP);
      float dP = length(Pv);
      if (dP > 0.05) {
        float3 Vd = Pv / dP;
        float dE = min(dP, maxD);
        float3 upV = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
        if (dot(Ld, upV) < 0) upV = -upV;                                  // como el rebote del pasto: el sol siempre arriba
        float cosT = dot(Vd, Ld);
        float g = clamp(VL.z, -0.9, 0.95);
        float hg = (1.0 - g * g) / pow(max(1.0 + g * g - 2.0 * g * cosT, 1e-4), 1.5);   // HG x 4 pi (isotropo = 1)
        float ph = lerp(1.0, hg, 0.8);
        float Hs = VL.w > 1.0 ? VL.w : 40.0;
        float resS = g_psScene.m_shadowMapResolutions.x;
        float smW = 0, smH = 0; PESShadowMap.GetDimensions(smW, smH);
        bool smOk = resS > 1.0 && smW > 1.0 && smH > 2.5 * smW;
        float bnd = 1.0 - 8.0 / max(resS, 16.0);
        float jit = frac(52.9829189 * frac(dot(v0.xy, float2(0.06711056, 0.00583715))) + IniParams.Load(int2(155,0)).y * 0.6180339);
        const int VN = 16;
        float dt = dE / VN, acc = 0.0;
        [loop] for (int vi = 0; vi < VN; vi++) {
          float t = (vi + jit) * dt;
          float3 Q = Vd * t;
          float hQ = g_psScene.m_eyepos.y + dot(Q, upV);
          float dens = exp(-max(hQ, 0.0) / Hs);
          float lit = 1.0;                                                  // fuera de las 3 cascadas: al sol
          if (smOk) {
            float4 hq = float4(PESShadowAlign(float3(Q.x, Q.y, Q.z + g_psScene.m_shadowProjection2._m33), g_psScene.m_view, IniParams.Load(int2(337,0)).w), 1.0);   // (31/10) w337
            float3 s0 = float3(dot(hq, g_psScene.m_shadowProjection._m00_m10_m20_m30), dot(hq, g_psScene.m_shadowProjection._m01_m11_m21_m31), dot(hq, g_psScene.m_shadowProjection._m02_m12_m22_m32));
            bool done = false;
            [unroll] for (int cI = 0; cI < 3; cI++) {
              float scl = cI == 0 ? 1.0 : (cI == 1 ? g_psScene.m_shadowProjection2._m03 : g_psScene.m_shadowProjection2._m13);
              float3 ofs = cI == 0 ? float3(0, 0, 0) : (cI == 1 ? g_psScene.m_shadowProjection2._m00_m01_m02 : g_psScene.m_shadowProjection2._m10_m11_m12);
              float3 sp = s0 * scl + ofs;
              float zr = g_psScene.m_shadowProjection2._m30 * (1.0 + sp.z);
              if (!done && abs(sp.x) <= bnd && abs(sp.y) <= bnd && zr >= 0.0) {
                float2 uvM = float2(sp.x * 0.5 + 0.5, (cI + 0.5 - sp.y * 0.5) / 3.0);
                float dS = PESShadowMap.Load(int3(min(uvM * float2(smW, smH), float2(smW, smH) - 1.0), 0)).x;
                lit = dS > zr ? 0.0 : 1.0;
                done = true;
              }
            }
          }
          acc += lit * dens;
        }
        float3 sunV = max(pLc_color.xyz, 0.0) * g_psScene.m_exposure.zzz;
        vlDbg = sunV * (ph * VL.x * 1e-3 * acc * dt);
        r0.xyz += vlDbg;
      }
    }
  }
  // === PES21 (12/11 20:50, Agustin con 4 fotos) w445 POLEN Y MOTAS A CONTRALUZ (foto 3): motitas en el aire a ras del pasto (6 capas fijas
  // de 0.12 a 2.2 m sobre la cancha, una grilla de 35 cm fija al mundo con una mota en algunas celdas) que el viento lleva (direccion de
  // las nubes z84, fuerza y108; cada capa a su velocidad y cada mota con su remolino) y que solo se ven cuando el sol las atraviesa:
  // mirando hacia el sol y en el aire al sol (shadow map: bajo el techo, la sombra del arco o de los jugadores no brillan). Cerca de la
  // camara son discos grandes y suaves (desenfocadas); cada una aparece y se va despacio y titila al girar. Cuantas y de que color segun
  // la hora (a la manana rocio: mas, chicas y blancas; a la tarde polen y polvo dorado), la estacion (invierno pocas), el clima (con lluvia
  // nada; suelo seco mas polvo) y la cancha (pasto alto mas polen; artificial o gastada, polvo). Solo de dia con sol. 0 = apagado
  { float kM = clamp(IniParams.Load(int2(445,0)).w, 0.0, 2.0);
    float3 LdM = pLc_dir.xyz;
    if (cmpOn && kM > 0.001 && dot(LdM, LdM) > 1e-6) {
      LdM = normalize(LdM);
      float3 scM = max(pLc_color.xyz, 0.0);
      float4 M114 = IniParams.Load(int2(114,0)), M415 = IniParams.Load(int2(415,0));
      float dayM = saturate((scM.r / max(scM.b, 1e-4) - 0.85) * 4.0) * (M114.w > 0.5 ? 1.0 : 0.2);
      float rainM = abs(M114.x - 1.0) < 0.5 ? 1.0 : 0.0, dryM = abs(M114.x - 2.0) < 0.5 ? 1.0 : 0.0;
      float seasM = M114.z < -0.5 ? 0.8 : (M114.z < 0.5 ? 1.0 : (M114.z < 1.5 ? 0.35 : 0.8));
      float dewM = (M415.z > 0.5 && M415.x >= 0.0) ? saturate((10.5 - M415.x) / 3.0) : 0.0;
      bool artM = IniParams.Load(int2(407,0)).x > 0.5;
      float mmM = IniParams.Load(int2(296,0)).w; mmM = mmM > 1.0 ? clamp(mmM, 15.0, 80.0) : 28.0;
      float wearM = saturate(IniParams.Load(int2(104,0)).w);
      float pollen = artM ? 0.15 : clamp(mmM / 28.0, 0.7, 1.8) * seasM;
      float dust = (0.25 + 0.6 * wearM + (artM ? 0.4 : 0.0)) * (1.0 + 0.8 * dryM);
      float densM = kM * dayM * (1.0 - rainM) * saturate((pollen + dust) * (1.0 + 0.4 * dewM) * 0.6);
      float3 upM = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
      if (dot(LdM, upM) < 0) upM = -upM;
      float zM = PESLinZ(uvC); if (!(zM > 0.0) || !(zM < 1e5)) zM = 1e5;
      float3 PvM = float3(((uvC * float2(2, -2) + float2(-1, 1)) - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zM, zM);
      float dSurf = length(PvM);
      float3 VdM = PvM / max(dSurf, 1e-4);
      float fwM = dot(VdM, LdM);
      float vyM = dot(VdM, upM);
      if (densM > 0.002 && fwM > 0.2 && abs(vyM) > 1e-4) {
        float ph = pow(saturate(fwM), 5.0) * 1.2 + pow(saturate(fwM), 40.0) * 4.0;   // la luz que pasa casi derecho (polen, gotitas)
        float3 vXM = normalize(float3(g_psScene.m_view._m00, g_psScene.m_view._m01, g_psScene.m_view._m02));
        float3 vZM = normalize(float3(g_psScene.m_view._m20, g_psScene.m_view._m21, g_psScene.m_view._m22));
        float twM = IniParams.Load(int2(23,0)).z;
        float aW = radians(IniParams.Load(int2(84,0)).z);
        float2 wdM = float2(cos(aW), sin(aW));
        float spM = 0.12 + 0.6 * saturate(IniParams.Load(int2(108,0)).y);   // m/s a ras del piso
        float pxK = abs(2.0 * g_psScene.m_projectionParam.y * g_psSystem.m_renderBuffer.w);   // metros por pixel por metro de distancia
        float pCell = saturate(0.45 * densM);
        float3 accM = 0;
        [loop] for (int kL = 0; kL < 6; kL++) {
          float hL = 0.12 * (1.0 + (float)kL) + 0.06 * (float)(kL * kL);   // 0.12, 0.30, 0.60, 1.02, 1.56, 2.22 m
          float tL = (hL - g_psScene.m_eyepos.y) / vyM;
          float gF = smoothstep(0.03, 0.08, abs(vyM));                        // rayo casi horizontal: la capa se apaga (no se corta)
          if (gF > 0.001 && tL > 0.08 && tL < min(dSurf, 30.0)) {
            float3 Hv = VdM * tL;
            float2 hw = g_psScene.m_eyepos.xz + float2(dot(Hv, vXM), dot(Hv, vZM));
            float2 drift = wdM * (spM * (0.7 + 0.12 * (float)kL) * twM);
            float2 g0 = floor((hw - drift) / 0.35);
            [unroll] for (int cy = -1; cy <= 1; cy++) {
              [unroll] for (int cx = -1; cx <= 1; cx++) {

                float2 ch = ci + float2(17.0, 31.0) * (float)(kL + 1);
                if (PES445H(ch) < pCell) {

                  float2 mp = (ci + 0.15 + 0.7 * float2(r1m, r2m)) * 0.35 + drift;
                  mp += 0.03 * float2(sin(twM * (0.7 + r3m) + r1m * 6.2832), cos(twM * (0.9 + r2m) + r3m * 6.2832));   // remolino
                  float my = hL;                                                // en el plano de la capa: la busqueda 3x3 siempre la encuentra
                  float2 dh = abs(mp - hw);
                  float edF = smoothstep(0.35, 0.22, max(dh.x, dh.y)) * gF;      // se apaga suave antes del borde de las 3x3 celdas (no se corta)

                  float3 Mv = vXM * dm.x + upM * dm.y + vZM * dm.z;            // al espacio de vista
                  float tM = dot(Mv, VdM);
                  if (tM > 0.08 && tM < dSurf) {
                    float dPerp = length(Mv - VdM * tM);
                    float rw = (0.0012 + 0.0018 * r3m) * (1.0 - 0.35 * dewM);     // 1-3 mm con su brillo (gotitas mas chicas)

                      if (edF > 0.001) accM += colM * (Im * life * twk * edF * PES445Lit(Mv));
                    }

                }
              }
            }
          }
        }

      }

  { float4 P89 = IniParams.Load(int2(89,0));
    if (P89.x > 0.5) {

    // punto AMARILLO = el sol esta adelante de la camara (contraluz), CELESTE = detras de la camara (luz frontal);

      float3 Ld = normalize(pLc_dir.xyz);

      float2 d = v0.xy - C;
      float r = length(d);

      float sl = length(sd);
      float2 sdn = sl > 1e-4 ? sd / sl : float2(0, -1);
      float along = dot(d, sdn), perp = abs(dot(d, float2(-sdn.y, sdn.x)));
      float3 col = Ld.z > 0 ? float3(1.0, 0.85, 0.1) : float3(0.2, 0.8, 1.0);
      if (abs(r - R) < 2.0) o0.xyz = float3(1, 1, 1);                                      // anillo
      if (along > 0 && along < R * sl && perp < 2.5) o0.xyz = col;                           // linea hacia el sol
      if (length(d - sdn * R * sl) < 7.0) o0.xyz = col;                                      // punto = el sol
      if (r < 3.0) o0.xyz = float3(1, 0, 0);                                                 // centro = camara
      float3 upV = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
      float elev = saturate(abs(dot(Ld, upV)));
      float2 b = v0.xy - float2(C.x - R, C.y + R * 1.25);
      if (b.y > 0 && b.y < 6.0 && b.x > 0 && b.x < 2.0 * R) o0.xyz = (b.x < 2.0 * R * elev) ? float3(1.0, 0.6, 0.1) : float3(0.25, 0.25, 0.25);
    } }
  if (IniParams.Load(int2(12,0)).w < -0.5) {   // PES21 debug: bits de m_shadowMapResolutions y m_exposure en la esquina
    int bx = (int)(v0.x / 24.0), by = (int)((v0.y - 300.0) / 24.0);
    if (v0.y >= 300.0 && by < 8 && bx < 32) {
      float4 vals0 = g_psScene.m_shadowMapResolutions;
      float4 vals1 = g_psScene.m_exposure;
      float val = by < 4 ? vals0[by] : vals1[by - 4];
      uint bits = asuint(val);
      o0.xyz = ((bits >> (31 - bx)) & 1) ? float3(0.5, 0.5, 0.5) : float3(0, 0, 0);
    }
  }
  PESVistaMotor(o0, v0.xy, 0.01);   // PES21 (11/11, Motor en tiempo real): vistas del motor (fila 429); va ultima para ganarle a las otras vistas de prueba
  return;
