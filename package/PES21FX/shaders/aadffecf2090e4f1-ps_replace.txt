// PES 2021 - Botines y pelota (material PBR con manchas de barro)  [hash aadffecf2090e4f1]  - PES21 (20/10, 143)
// Traducido a HLSL desde el ensamblador (catalogo PES21 FX), instruccion por instruccion. G-buffer diferido: o0 albedo,
// o1 normal octaedrica (+tipo), o2 rugosidad / especular / metal, o3 velocidad, o4 (0,0,0,1). t4 = mancha (barro) que el
// juego mezcla segun cb4[1].x. Lo que agrega PES21 FX (143, x on):
//  - antialiasing especular (Kaplanyan / Tokuyoshi): la rugosidad crece con la variacion de la normal en el pixel; los
//    botines brillantes y la pelota dejan de titilar a distancia y en movimiento (y143 fuerza)
//  - cancha mojada: el cuero sintetico y la pelota se ven mojados (menos rugosos, apenas mas oscuros) (z143)
//  - barro que se acumula con los minutos de partido (mas con lluvia), sobre la misma mancha del juego (w143)
cbuffer cPSMaterial : register(b4) { float4 m_materials[8]; }
cbuffer PESBallC : register(b12) { float4 pBall; }   // (2/11) la DLL, solo en el dibujo de la pelota: x 1, y verde del pasto, z tierra, w cal (0..1); en los botines queda lo que haya (x != 1)
// (11/11 4wp0pj) en los botines (fila 430 x > 0) la DLL manda x 2, y tierra, z pasto pegado, w cancha mojada (0..1) DE ESE JUGADOR: lo que corrio, freno y piso
cbuffer cPSSystem : register(b0) { float4 m_param; float4 m_renderInfo; float4 m_renderBuffer; float4 m_dominantLightDir; float4 m_timeParam; }

SamplerState g_sampler_diffuse_s : register(s0);
SamplerState g_sampler_normal_s : register(s1);
SamplerState g_sampler_srm_s : register(s2);
SamplerState g_sampler_metalnessmap_s : register(s3);
SamplerState g_sampler_stain_s : register(s4);
SamplerState g_samplerPoint_Wrap_s : register(s8);
Texture2D<float4> g_tex_diffuse : register(t0);
Texture2D<float4> g_tex_normal : register(t1);
Texture2D<float4> g_tex_srm : register(t2);
Texture2D<float4> g_tex_metalnessmap : register(t3);
Texture2D<float4> g_tex_stain : register(t4);
Texture2D<float4> g_tex_mesh : register(t15);
Texture1D<float4> IniParams : register(t120);

float PESEqWet() {   // misma regla que el pasto (110 / 114): automatico = lluvia del partido
  float4 Q = IniParams.Load(int2(110, 0)); float4 L = IniParams.Load(int2(114, 0));
  int m = (int)(Q.z + 0.5); float wet = 0.0;
  if (m == 0) wet = ((int)(L.x + 0.5) == 1) ? 1.0 : 0.0; else if (m == 2) wet = 1.0;
  return wet * (Q.w > 0.001 ? saturate(Q.w) : 1.0);
}

// (11/11 4wp0pj) botines sucios: grumos de tierra (cupulas en celdas, con tamano y alto al azar) y hojitas de pasto pegadas (segmentos
// finos girados), en uv del botin. Devuelve x alto de la tierra, y hojita de pasto, z al azar de la celda (color), w borde del grumo (sombra)
float2 PesH2(float2 c) { return frac(sin(float2(dot(c, float2(127.1, 311.7)), dot(c, float2(269.5, 183.3)))) * 43758.5453); }
float4 PesBootH(float2 uv, float dCov, float gCov) {
  float hD = 0.0, rimD = 0.0, rnd = 0.0, hG = 0.0;
  float2 p = uv * 42.0, ip = floor(p), fp = frac(p);
  [unroll] for (int j = -1; j <= 1; j++) [unroll] for (int i = -1; i <= 1; i++) {
    float2 c = ip + float2(i, j), h = PesH2(c), o = float2(i, j) + 0.2 + 0.6 * h - fp;
    float rad = 0.28 + 0.32 * frac(h.x * 7.3 + h.y * 3.1), r = length(o) / rad;
    float on = step(frac(h.x * 13.7 + h.y * 5.3), dCov);
    float dome = sqrt(saturate(1.0 - r * r)) * (0.55 + 0.45 * h.y) * on;
    if (dome > hD) { hD = dome; rnd = h.x; rimD = saturate(1.0 - abs(r - 1.0) * 5.0) * on; }
  }
  float2 q = uv * 118.0 + 5.7, iq = floor(q), fq = frac(q);   // migas chicas de tierra
  [unroll] for (int j2 = -1; j2 <= 1; j2++) [unroll] for (int i2 = -1; i2 <= 1; i2++) {
    float2 c = iq + float2(i2, j2), h = PesH2(c), o = float2(i2, j2) + 0.2 + 0.6 * h - fq;
    float on = step(frac(h.y * 17.1 + h.x * 2.9), dCov * 1.15);
    hD = max(hD, sqrt(saturate(1.0 - dot(o, o) / 0.09)) * 0.45 * on);
  }
  float2 g = uv * 30.0 + 11.3, ig = floor(g), fg = frac(g);   // hojitas de pasto: una por celda, girada, 0.5..0.9 de largo
  [unroll] for (int j3 = -1; j3 <= 1; j3++) [unroll] for (int i3 = -1; i3 <= 1; i3++) {
    float2 c = ig + float2(i3, j3), h = PesH2(c + 31.0);
    float on = step(frac(h.x * 9.1 + h.y * 21.7), gCov);
    float a = h.x * 6.2832, L = 0.25 + 0.2 * h.y; float2 dir = float2(cos(a), sin(a));
    float2 o = fg - (float2(i3, j3) + 0.5 + 0.3 * (h - 0.5));
    float t = clamp(dot(o, dir), -L, L); float dd = length(o - dir * t);
    float wdt = 0.035 + 0.03 * h.y * (1.0 - abs(t) / L);
    hG = max(hG, saturate(1.0 - dd / wdt) * on);
  }
  return float4(hD, hG, rnd, rimD);
}

void main(
  float4 v0 : SV_Position0,
  float4 v1 : COLOR0,
  float2 v2 : TEXCOORD0,
  float2 w2 : TEXCOORD9,
  float3 v3 : TEXCOORD5,
  float3 v4 : TEXCOORD6,
  float3 v5 : TEXCOORD7,
  float4 v6 : TEXCOORD8,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,
  out float4 o3 : SV_Target3,
  out float4 o4 : SV_Target4)
{
  float4 r0, r1, r2;
  float4 EQ = IniParams.Load(int2(143, 0));
  bool pesOn = EQ.x > 0.5;
  float4 BT = IniParams.Load(int2(430, 0));   // (11/11 4wp0pj) botines sucios: x cantidad, y relieve, z pasto pegado
  bool pesBoot = pesOn && BT.x > 0.001 && abs(pBall.x - 2.0) < 0.01;
  float pesWet = pesOn ? PESEqWet() * saturate(EQ.z) : 0.0;
  // --- fundido con tramado (aparicion / LOD) ---
  r0.xy = (v0.xy - 0.5) * 0.125;
  float4 tm = g_tex_mesh.Sample(g_samplerPoint_Wrap_s, r0.xy);
  r2 = g_tex_diffuse.Sample(g_sampler_diffuse_s, v2.xy);
  float2 cut = float2(r2.w - tm.w * m_param.w, v1.w - tm.x * m_param.w);
  if (cut.x < 0.0 || cut.y < 0.0) discard;
  // --- mancha de barro del juego (cb4[1].x = cuanto) ---
  r0 = g_tex_stain.Sample(g_sampler_stain_s, v2.xy);
  float4 pesStn = r0;   // (2/11) mascara de barro del juego: tambien da forma a las manchas de la pelota
  float stainAmt = m_materials[1].x;
  if (pesBoot) {
    // (11/11 4wp0pj) el barro del juego crece con la tierra que junto ESTE jugador (no con el reloj): el que no corre queda limpio
    stainAmt = saturate(stainAmt + saturate(pBall.y) * 0.55 * (0.6 + 0.4 * saturate(pBall.w + 0.5)));
  } else if (pesOn && EQ.w > 0.001) {
    float mins = IniParams.Load(int2(10, 0)).x;                       // minutos reales de partido (la DLL)
    stainAmt = saturate(stainAmt + saturate(EQ.w) * 0.45 * saturate(mins / 12.0) * (0.35 + 0.65 * PESEqWet()));
  }
  r1.x = 1.0 - saturate((1.0 - (r0.w - 0.25) - stainAmt) * 4.0);
  r1.y = r0.w * r1.x;
  r0.w = 1.0 - r1.x * r0.w;
  r0.xyz = r0.xyz * r2.xyz * r1.y;
  o0.xyz = r2.xyz * r0.w + r0.xyz;
  o0.w = 1.0;
  // (2/11) pelota manchada (fila 177 x, la DLL la reconoce y manda b12): verde del pasto donde rodo, tierra donde la cancha esta pelada,
  // cal al cruzar las lineas. Manchas irregulares: la mascara de barro del juego + bandas suaves por uv, mas en las costuras (zonas oscuras)
  float pesDust = 0.0, pesStainH = 0.0;   // pesStainH: "alto" de la mancha (relieve: normal de detalle + rugosidad, pedido de Agustin)
  if (pesOn && abs(pBall.x - 1.0) < 0.01 && (pBall.y + pBall.z + pBall.w) > 0.002) {
    float band = 0.5 + 0.5 * sin(v2.x * 29.0 + v2.y * 13.0) * sin(v2.y * 23.0 - v2.x * 7.0);
    float patch = saturate(pesStn.w * 1.3 + band * 0.6 - 0.25);
    float lum = dot(o0.xyz, float3(0.3, 0.59, 0.11));
    // sutil (Agustin): a tope de la fila 177 apenas se nota en camara de juego; x177 = 2 lo duplica
    float g = saturate(pBall.y) * saturate(patch * 1.2) * 0.30, dirt = saturate(pBall.z) * saturate(patch * 1.4) * 0.40, chalk = saturate(pBall.w) * saturate(0.4 + band * 0.8) * 0.35;
    o0.xyz = lerp(o0.xyz, o0.xyz * float3(0.62, 0.80, 0.38) + float3(0.02, 0.05, 0.0), g);
    o0.xyz = lerp(o0.xyz, float3(0.30, 0.22, 0.14) * (0.6 + 0.6 * lum), dirt);
    o0.xyz = lerp(o0.xyz, float3(0.86, 0.85, 0.80), chalk * (1.0 - 0.5 * lum));
    pesDust = saturate(dirt + chalk * 0.8 + g * 0.3);
    pesStainH = saturate(dirt * 1.3 + chalk * 0.6 + g * 0.5) * patch;
  }
  // (11/11 4wp0pj) botines sucios: grumos de tierra y hojitas de pasto con relieve, primero donde la mascara de barro del juego (t4 .w,
  // alta = se ensucia primero: suela, tapones, bordes) lo dice y avanzando con la cantidad de ESE jugador
  float4 bH = 0.0; float bootCov = 0.0, bootWet = 0.0, bootRel = 0.0; float2 bootCv = 0.0;
  if (pesBoot && (pBall.y + pBall.z) > 0.002) {
    bootWet = saturate(pBall.w);
    float order = saturate(pesStn.w);
    float dCov = saturate(saturate(pBall.y) * 1.3 - (1.0 - order) * 0.85 + 0.05);
    float gCov = saturate(saturate(pBall.z) * (BT.z > 0.001 ? saturate(BT.z) * 2.0 : 1.0) * 0.9 - (1.0 - order) * 0.7 + 0.05);
    bootCv = float2(dCov, gCov);
    bH = PesBootH(v2.xy, dCov, gCov);
    float lum = dot(o0.xyz, float3(0.3, 0.59, 0.11));
    float3 dirtC = lerp(float3(0.19, 0.13, 0.08), float3(0.33, 0.25, 0.16), bH.z) * lerp(1.0, 0.62, bootWet);   // mojada: barro oscuro
    float film = saturate(dCov * 1.4 - 0.15) * saturate(order * 1.5) * 0.45;                                     // capa fina entre grumos
    float dA = saturate(max(film, bH.x * 2.2));
    o0.xyz = lerp(o0.xyz, dirtC * (0.7 + 0.45 * bH.x) * (0.85 + 0.3 * lum), dA);
    o0.xyz *= 1.0 - 0.35 * bH.w * saturate(bH.x * 4.0 + 0.3);                                                   // borde del grumo: sombra
    float3 grassC = lerp(float3(0.10, 0.24, 0.05), float3(0.30, 0.31, 0.11), frac(bH.z * 5.3));                 // fresca .. seca
    o0.xyz = lerp(o0.xyz, grassC * lerp(1.0, 0.8, bootWet), bH.y);
    bootCov = saturate(dA + bH.y);
    bootRel = BT.y > 0.001 ? saturate(BT.y * 0.5) * 2.0 : 1.0;
  }
  // --- normal ---
  float3 n0 = normalize(v3), n1 = normalize(v4), n2 = normalize(v5);
  float4 tn = g_tex_normal.Sample(g_sampler_normal_s, v2.xy);
  float2 nm = float2(tn.w * tn.x, tn.y) * 2.0 - 1.0;
  float nz = max(1.0 - nm.x * nm.x - nm.y * nm.y, 0.0) + 0.0001;
  nz = nz * rsqrt(nz);
  float3 N = normalize(n0 * nm.x + n1 * nm.y + n2 * nz);
  if (pesStainH > 0.001) {
    // (2/11) relieve de las manchas (normal map de detalle derivado): grano fino de la mascara de barro del juego + poros del tinte,
    // como gradiente de altura en uv -> inclina la normal en el marco tangente (n0, n1). Tierra gruesa, cal fina
    float2 du = float2(1.0 / 512.0, 0.0), dv = float2(0.0, 1.0 / 512.0);
    float hC = g_tex_stain.Sample(g_sampler_stain_s, v2.xy).w;
    float hX = g_tex_stain.Sample(g_sampler_stain_s, v2.xy + du).w, hY = g_tex_stain.Sample(g_sampler_stain_s, v2.xy + dv).w;
    float2 grain = float2(frac(sin(dot(floor(v2.xy * 900.0), float2(12.9898, 78.233))) * 43758.5453), frac(sin(dot(floor(v2.xy * 900.0) + 7.0, float2(39.3468, 11.135))) * 24634.6345)) - 0.5;
    float2 gH = float2(hX - hC, hY - hC) * 6.0 + grain * (0.25 + 0.55 * saturate(pBall.z));
    N = normalize(N + (n0 * gH.x + n1 * gH.y) * pesStainH * 1.2);
  }
  if (bootCov > 0.001 && bootRel > 0.001) {
    // (11/11 4wp0pj) relieve de los grumos y las hojitas: gradiente del alto en uv (diferencias finitas) -> inclina la normal
    float2 du = float2(1.0 / 2048.0, 0.0), dv = float2(0.0, 1.0 / 2048.0);
    float4 hX = PesBootH(v2.xy + du, bootCv.x, bootCv.y), hY = PesBootH(v2.xy + dv, bootCv.x, bootCv.y);
    float hC = bH.x + bH.y * 0.5;
    float2 gB = float2(hX.x + hX.y * 0.5 - hC, hY.x + hY.y * 0.5 - hC) * 48.0;
    N = normalize(N - (n0 * gB.x + n1 * gB.y) * bootRel);
  }
  // codificacion octaedrica del juego (10 bits + 2 bits bajos en w)
  {
    float s = abs(N.y) + abs(N.x) + abs(N.z);
    float2 p = N.xy / s;
    float2 sg = float2(p.x >= 0.0 ? 1.0 : -1.0, p.y >= 0.0 ? 1.0 : -1.0);
    float2 wr = sg * (1.0 - abs(p.yx));
    p = (0.0 < -N.z) ? p : wr;
    uint2 q = (uint2)((p * 0.5 + 0.5) * 2047.0);
    uint lo = ((q.x & 1u) << 1) | (q.y & 1u);
    q >>= 1;
    o1.xy = (float2)q * 0.000977517106;
    o1.w = (float)lo * 0.333333343;
    o1.z = 0.000977517106;
  }
  // --- material ---
  float metal = g_tex_metalnessmap.Sample(g_sampler_metalnessmap_s, v2.xy).x;
  float2 srm = g_tex_srm.Sample(g_sampler_srm_s, v2.xy).xy;
  o2.z = r0.w * metal;
  o2.y = r0.w * srm.x;
  float rough = max(srm.y, 0.004);
  float3 dNx = ddx_fine(N), dNy = ddy_fine(N);   // fuera de cualquier rama
  if (pesOn) {
    if (EQ.y > 0.001) {   // antialiasing especular: varianza de la normal en el pixel -> rugosidad extra
      float var = 0.25 * (dot(dNx, dNx) + dot(dNy, dNy));
      float kr2 = min(2.0 * var, 0.18) * saturate(EQ.y);
      rough = sqrt(rough * rough + kr2);
    }
    if (pesWet > 0.001) {   // mojado: agua sobre el cuero / la pelota
      rough = lerp(rough, max(rough * 0.45, 0.06), pesWet);
      o2.y = lerp(o2.y, saturate(o2.y * 1.6 + 0.1), pesWet * (1.0 - o2.z));
      o0.xyz *= lerp(1.0, 0.88, pesWet * (1.0 - o2.z));
    }
  }
  if (pesDust > 0.001) { rough = lerp(rough, 0.85, pesDust); o2.y *= 1.0 - 0.6 * pesDust; }
  if (bootCov > 0.001) {   // (11/11 4wp0pj) tierra seca: mate; barro mojado: algo de brillo; tapa el metal de los tapones
    rough = lerp(rough, lerp(0.9, 0.4, bootWet), bootCov); o2.y *= 1.0 - (0.65 - 0.35 * bootWet) * bootCov; o2.z *= 1.0 - bootCov;
  }   // (2/11) el polvo y la tierra apagan el brillo
  o2.x = rough;
  o2.w = 0.0;
  // --- velocidad en pantalla ---
  r0 = v6 / float4(w2.x, w2.x, w2.y, w2.y);
  r0 = r0 * 0.5 + 0.5;
  r0 = r0.zwxy * float4(1.0, -1.0, 1.0, -1.0) + float4(0.0, 1.0, 0.0, 1.0);
  o3.xy = (r0.zw - r0.xy) * m_renderBuffer.xy;
  o3.zw = 0.0;
  o4 = float4(0.0, 0.0, 0.0, 1.0);
}
