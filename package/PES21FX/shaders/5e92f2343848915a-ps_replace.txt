  return g_tex_turfSlopTable.SampleLevel(g_samplerLinear_Clamp_s, uv, 0).y;
}

float bnoise(float2 p) { float2 i = floor(p), f = frac(p); f = f * f * (3 - 2 * f);

      c = lerp(c, dirt, wr.y);
      o0.xyz = saturate(c);
      o2.y = saturate(o2.y * (1.0 - wr.y * 0.6 - wr.x * 0.25));   // tierra y pasto seco: mate
      if (Q7.w > 1.5 && Q7.w < 2.5) o0.xyz = float3(wr.x, wr.y, 0.0);   // depuracion: desgaste
    }
  }
  return;
