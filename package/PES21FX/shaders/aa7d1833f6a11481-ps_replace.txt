void main(

  float4 v1 : COLOR0,
  float2 v2 : TEXCOORD0,
  float4 w2 : TEXCOORD9,   // (24/10 e) zw = posicion de reposo (y, x) del VS 1c02a92a
  float4 v3 : TEXCOORD4,
  float4 v4 : TEXCOORD5,
  float4 v5 : TEXCOORD6,
  float4 v6 : TEXCOORD7,
  float4 v7 : TEXCOORD8,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,
  out float4 o3 : SV_Target3,
  out float4 o4 : SV_Target4)
{
  float4 r0,r1,r2,r3;
  uint4 bitmask, uiDest;
  float4 fDest;
  // ---- PES21 piel del cuerpo v2 (16/10): brazos, piernas y cuello reciben las mismas mejoras que la cara ----
  // w = mejoras de piel (Ctrl+Alt+F7), w43 = on/off solo cuerpo [Alt+Shift+B]; x32/y32 nitidez, x35 cavidades, z = especular
  float4 P0b = IniParams.Load(int2(0,0));
  float4 P32b = IniParams.Load(int2(32,0));
  float4 P35b = IniParams.Load(int2(35,0));
  float4 P43b = IniParams.Load(int2(43,0));
  bool bodyOn = (P0b.w > 0.5) && (P43b.w > 0.5);
  float bBD = (bodyOn && P32b.w > 0.5) ? P32b.x : 0.0;
  float bBN = (bodyOn && P32b.w > 0.5) ? P32b.y : 0.0;

  // entre el pelo corto) quedaba brilloso y grisaceo con la pelicula de sudor y los poros de la cara. Peso hpK: cuello (v > 0.5

  float4 hpSM = headPiece ? pesSSSMap.SampleLevel(g_samplerLinear_Clamp_s, v2.xy, 0) : float4(0, 0, 0, 0);

  float bSpec = ((bodyOn || headPiece) && P0b.z > 0) ? P0b.z : 1.0;

  // poros de la cara (x94), igual que el shader de cara: derivadas fuera de las ramas
  float4 P94h = IniParams.Load(int2(94,0));

  r0.w = dot(v5.xyz, v5.xyz);

  r1.xyz = v5.xyz * r0.www;

      float2 mM = float2(nM.x * nM.z, nM.y) * 2.0 - 1.0;
      r2.xy += (defM - 1.0) * mM;
      float l2 = dot(r2.xy, r2.xy); if (l2 > 0.94) r2.xy *= rsqrt(l2 / 0.94);
    }

    if (near > 0.001) {
      float4 guide = pesFacePores.Sample(g_samplerLinear_Clamp_s, v2.xy);
      hpTZone = guide.b;
      float4 mn = pesSkinMicro.SampleGrad(g_samplerLinear_Clamp_s, frac(hpMuv), hpMdx, hpMdy);

      mn = lerp(mn, mn2, guide.g * 0.7);
      float3 tn = mn.xyz * 2.0 - 1.0;

      float str = clamp(P94h.x, 0.0, 2.0) * near * 1.1 * skinM * (0.25 + 1.0 * guide.r) * hpK;

      hpPoreOcc = 1.0 - hpPoreA * guide.a * 0.55 * saturate(P94h.x);
      float4 ma = pesSkinMicro.SampleGrad(g_samplerLinear_Clamp_s, frac(hpMuv), hpMdx * 6.0, hpMdy * 6.0);
      hpMicroVar = saturate((1.0 - length(ma.xyz * 2.0 - 1.0)) * 8.0) * saturate(P94h.x * 0.7) * skinM;

    bCav = 1.0 - saturate(P35b.x) * smoothstep(0.02, 0.35, 1.0 - bnz) * 0.55;

  r0.xyz = r0.xyz * r2.xxx + r1.xyz;

  r0.w = rsqrt(r0.w);
  r1.xyz = v6.xyz * r0.www;
  r0.w = -r2.x * r2.x + 1;
  r0.w = -r2.y * r2.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.w = rsqrt(r0.w);
  r0.w = r1.w * r0.w;
  r0.xyz = r1.xyz * r0.www + r0.xyz;
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r0.xyz = r0.xyz * r0.www;
  r0.w = dot(v3.xyz, v3.xyz);
  r0.w = rsqrt(r0.w);
  r1.xyz = v3.xyz * r0.www;
  r0.w = dot(r0.xyz, r1.xyz);
  r0.w = saturate(1 + -r0.w);
  r0.w = log2(r0.w);
  r0.w = g_psMaterial.m_materials[1].x * r0.w;
  r0.w = exp2(r0.w);
  r1.xy = g_tex_srm.SampleBias(g_samplerLinear_Clamp_s, v2.xy, bBD).xy;
  r1.z = g_psMaterial.m_materials[2].w * r1.x;
  r0.w = r1.z * r0.w;
  r2.xyz = g_tex_Dirty.Sample(g_sampler_Dirty_s, v2.xy).xyz;
  r1.z = saturate(dot(g_psMaterial.m_materials[1].yzw, r2.xyz));
  r2.xy = r1.yy * float2(255,255) + float2(-205,-170);
  r1.w = r2.x * -0.00800000038 + 0.930000007;
  r2.x = 0.0110000009 * r2.y;
  r2.x = saturate(r2.x);
  r2.x = min(0.660000026, r2.x);
  r2.x = r1.y * r2.x + -r1.y;
  r2.x = r1.z * r2.x + r1.y;
  r1.w = max(0.730000019, r1.w);
  r1.w = min(0.930000007, r1.w);
  r2.yzw = g_tex_diffuse.SampleBias(g_samplerLinear_Clamp_s, v2.xy, bBD).xyz;
  r3.xyz = r2.yzw * r1.www + -r2.yzw;
  r2.yzw = r1.zzz * r3.xyz + r2.yzw;

  float hpTrans = g_tex_translucent.Sample(g_samplerLinear_Clamp_s, v2.xy).y;

    float4 P62h = IniParams.Load(int2(62,0));

    if (P62h.x > 0.5) {   // mapa de SSS de la cara: orejas, nariz y labios finos; sangre
      float4 sm = pesSSSMap.Sample(g_samplerLinear_Clamp_s, v2.xy);
      float k = sm.w;
      hpTrans = lerp(hpTrans, max(hpTrans, sm.x * saturate(P62h.y)), k);
      hpTrans = saturate(hpTrans + 0.12 * sm.z * saturate(P62h.y) * k);
      hpBlood = sm.z * k * saturate(P62h.z);
    }
    float hpThin = hpTrans;
    float aSat = P0b.y > 0 ? P0b.y : 1.0;
    float lum = dot(r2.yzw, float3(0.2126, 0.7152, 0.0722));
    r2.yzw = max(0, lerp(lum.xxx, r2.yzw, aSat));

    r2.yzw = saturate(r2.yzw);
    float4 P9h = IniParams.Load(int2(9,0));
    float swP = saturate((IniParams.Load(int2(10,0)).x - P9h.x) / max(P9h.y - P9h.x, 0.1));
    float flushF = swP * IniParams.Load(int2(12,0)).x;
    r2.yzw *= float3(1.0 + flushF, 1.0 - flushF * 0.35, 1.0 - flushF * 0.45);
    float4 P72h = IniParams.Load(int2(72,0)), P73h = IniParams.Load(int2(73,0));
    if (P72h.w > 0.5) {   // pelicula de sudor por zonas (mismo mapa que la cara)
      float4 FZ = pesZones.Sample(g_samplerLinear_Clamp_s, v2.xy);
      float zoneSw = lerp(1.0, saturate(FZ.r * 1.25), saturate(P72h.y));
      float swPZ = saturate(swP * lerp(1.0, 0.35 + 1.1 * saturate(FZ.r * 1.25), saturate(P73h.z)));
      hpFilm = saturate(swPZ * zoneSw) * saturate(P72h.x);
    }
    r2.yzw *= (1.0 - 0.06 * hpFilm) * hpPoreOcc;
    float sssB = P0b.x > 0 ? P0b.x : 1.0;
    hpTrans = saturate(hpTrans * sssB);
    r2.yzw = lerp(hpAlb0, r2.yzw, hpK);
    hpTrans = lerp(hpTr0, hpTrans, hpK);
  }
  float3 stVeil = g_psMaterial.m_materials[2].xyz;
  bool stBody = v3.w > 1.999 && v3.w < 3.001;   // (24/10 e) VS 1c02a92a = cuerpo (la cabeza generica viene con faab87ee)
  int stE = PesSkinEntry(v6.w);
  float4 stPB = stE >= 0 ? IniParams.Load(int2(197 + 3 * stE, 0)) : float4(1, 1, 0, 0);
  bool stTorso = stPB.w > 1.5;                                   // (24/10 g) "solo el torso"
  float stTm = stBody ? PesSkinTorso(v6.w) : 0.0;                // cuanto de torso es este pixel (VS)
#ifdef PES_HL_ACC
  // (24/10 e) medicion: color del cuerpo en la base del cuello (posicion de reposo: y 1.47..1.60, |x| < 0.07) -> lugar 4.
  // (24/10 g) con "solo el torso", en el pecho alto (y 1.36..1.46, |x| < 0.10, solo piel del torso): el cuello no es torso

#endif
  {   // (24/10 c) tono de piel propio del jugador (skintone.cpp + pes_skintone.hlsli): Oklab, tambien el velo de borde;
      // (24/10 e) el cuerpo usa su fila (con la correccion "cuerpo como la cara"), la cabeza la de la cara; (24/10 g) con
      // "solo el torso" la fila del cuerpo va solo al torso (brazos, cuello y piernas quedan como el juego)
    if (stE >= 0) {
      float3 stP = stBody ? (stTorso ? lerp(float3(1, 1, 0), stPB.xyz, stTm) : stPB.xyz) : PesSkinParams(stE);

    }

  r3.xyz = stVeil + -r2.yzw;
  o0.xyz = r0.www * r3.xyz + r2.yzw;
  o0.xyz *= bCav;
  {   // (24/10) resaltado del musculo elegido en el menu Fisico (como el brillo que late en los menus de la cara): el VS
      // 1c02a92a deja el peso del vertice en v3.w (+2; otros VS no lo escriben) y la DLL manda el pulso en x174.
      // (24/10 b) y como en la cara, un aro y un punto blanco brillante sobre el musculo: posicion de reposo en v4.w..v6.w,
      // marcas en 187 + k (punto, radio), latido y desvanecido en x186 / y186
    bool hlOk = v3.w > 1.999 && v3.w < 3.001;
    float4 HL = IniParams.Load(int2(174,0));
    float hlW = hlOk ? saturate(v3.w - 2.0) : 0.0;
    float hlK = hlW * saturate(HL.x);
    if (hlK > 0.001) o0.xyz = lerp(o0.xyz, float3(0.93, 0.97, 1.0), hlK * 0.6);
#ifdef PES_HL_ACC
    // (24/10 c) donde quedo cada marca en la pantalla: suma de las posiciones de los pixeles cerca de la marca (por lado)
    if (hlOk && HL.x > 0.001) {
      float w0 = abs(v4.w), w1 = abs(v5.w);
      uint slot = 99;
      if (w0 > 0.5) slot = v4.w >= 0 ? 0 : 1;
      else if (w1 > 0.5) slot = v5.w >= 0 ? 2 : 3;
      if (slot < 4) {
        uint base = slot * 16, dmy;
        gHlAcc.InterlockedAdd(base + 0, (uint)v0.x, dmy);
        gHlAcc.InterlockedAdd(base + 4, (uint)v0.y, dmy);
        gHlAcc.InterlockedAdd(base + 8, 1u, dmy);
      }
    }
#endif
  }
  o0.w = 1;
  r0.w = abs(r0.x) + abs(r0.y);
  r0.w = r0.w + abs(r0.z);
  r0.xy = r0.xy / r0.ww;
  r0.z = cmp(0 < -r0.z);
  r1.zw = cmp(r0.xy >= float2(0,0));
  r1.zw = r1.zw ? float2(1,1) : float2(-1,-1);

  r0.xy = r0.xy * float2(0.5,0.5) + float2(0.5,0.5);
  r0.xy = float2(2047,2047) * r0.xy;
  r0.xy = (uint2)r0.xy;
  bitmask.z = ((~(-1 << 1)) << 1) & 0xffffffff;  r0.z = (((uint)r0.x << 1) & bitmask.z) | ((uint)0 & ~bitmask.z);
  bitmask.z = ((~(-1 << 1)) << 0) & 0xffffffff;  r0.z = (((uint)r0.y << 0) & bitmask.z) | ((uint)r0.z & ~bitmask.z);
  r0.xy = (uint2)r0.xy >> int2(1,1);
  r0.xyw = (uint3)r0.xyz;
  o1.xyw = float3(0.000977517106,0.000977517106,0.333333343) * r0.xyw;
  r0.x = hpTrans;
  r0.x = r0.x * 63 + 0.5;
  r0.x = (uint)r0.x;
  r0.x = (uint)r0.x << 4;
  r0.x = (uint)r0.x;
  o1.z = 0.000977517106 * r0.x;
  r0.x = min(0.56078434, r1.y);
  o2.y = saturate(r1.x * bSpec * lerp(1.0, bCav, 0.8));

  if (hpK > 0.001) {   // (20/10) brillo como la cara: pelicula de sudor, poros (Toksvig) y zona T

    float gS = IniParams.Load(int2(2,0)).y; gS = gS > 0 ? gS : 1.0;
    r0.x = saturate(r0.x * gS);
    r0.x = lerp(r0.x, r0.x * 0.45, hpFilm);
    r0.x = saturate(r0.x + hpMicroVar * 0.30 + hpPoreA * 0.20);
    r0.x = saturate(r0.x * lerp(1.06, 0.86, hpTZone * saturate(P94h.x)));
    o2.y = lerp(o2.y, max(o2.y, 0.5), hpFilm * 0.8);
    o2.y *= (1.0 - 0.45 * hpPoreA) * lerp(0.95, 1.12, hpTZone * saturate(P94h.x));
    r0.x = lerp(hpR0, r0.x, hpK); o2.y = lerp(hpS0, o2.y, hpK);
  }
  r0.y = r0.x * 0.0361702144 + 0.0904255286;
  r0.y = r0.y * r0.x + 0.180851057;
  r0.y = r0.y * r0.x + 0.271276593;
  r0.y = r0.y * r0.x + 0.271276593;
