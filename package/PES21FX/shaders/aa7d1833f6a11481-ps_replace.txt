void main(
  float4 v0 : SV_Position0,
  float4 v1 : COLOR0,
  float2 v2 : TEXCOORD0,
  float4 w2 : TEXCOORD9,   // (24/10 e) zw = posicion de reposo (y, x) del VS 1c02a92a
  float4 v3 : TEXCOORD4,
  float4 v4 : TEXCOORD5,
  float4 v5 : TEXCOORD6,
  float4 v6 : TEXCOORD7,
  float4 v7 : TEXCOORD8,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,

  float4 P0b = IniParams.Load(int2(0,0));
  float4 P32b = IniParams.Load(int2(32,0));
  float4 P35b = IniParams.Load(int2(35,0));
  float4 P43b = IniParams.Load(int2(43,0));
  bool bodyOn = (P0b.w > 0.5) && (P43b.w > 0.5);
  float bBD = (bodyOn && P32b.w > 0.5) ? P32b.x : 0.0;
  float bBN = (bodyOn && P32b.w > 0.5) ? P32b.y : 0.0;
  // cabeza de una cara generica: la DLL enlazo el mapa de SSS de la cara (1024x1024) en t22
  uint hpW = 0, hpH = 0; pesSSSMap.GetDimensions(hpW, hpH);
  bool headPiece = (P0b.w > 0.5) && IniParams.Load(int2(147,0)).x > 0.5 && hpW == 1024 && hpH == 1024;

  // entre el pelo corto) quedaba brilloso y grisaceo con la pelicula de sudor y los poros de la cara. Peso hpK: cuello (v > 0.5
  // del plano de la cabeza) u orejas (zonas finas del mapa de SSS), y solo donde el mapa dice piel (la nuca/pelo queda afuera)
  float4 hpSM = headPiece ? pesSSSMap.SampleLevel(g_samplerLinear_Clamp_s, v2.xy, 0) : float4(0, 0, 0, 0);
  float hpK = headPiece ? saturate(max(smoothstep(0.50, 0.62, v2.y), saturate((hpSM.x - 0.25) * 3.0)) * hpSM.w) : 0.0;
  float bSpec = ((bodyOn || headPiece) && P0b.z > 0) ? P0b.z : 1.0;

  float2 hpMdx = ddx(hpMuv), hpMdy = ddy(hpMuv);
  float hpFpx = max(fwidth(v2.x), fwidth(v2.y)) * 2048.0;
  float hpPoreA = 0.0, hpMicroVar = 0.0, hpPoreOcc = 1.0, hpTZone = 0.0;

  r0.xy = float2(0.125,0.125) * r0.xy;
  r0.yz = g_tex_mesh.Sample(g_samplerPoint_Wrap_s, r0.xy).xw;
  r0.x = g_psSystem.m_param.w * r0.z;
  r1.x = 1;
  r1.y = v1.w;
  r0.xy = r1.xy + -r0.xy;
  r0.xy = cmp(r0.xy < float2(0,0));
  r0.x = (int)r0.y | (int)r0.x;

  r0.x = dot(v4.xyz, v4.xyz);
  r0.x = rsqrt(r0.x);
  r0.xyz = v4.xyz * r0.xxx;
  r0.w = dot(v5.xyz, v5.xyz);
  r0.w = rsqrt(r0.w);
  r1.xyz = v5.xyz * r0.www;

  r2.x = r2.x * r2.z;
  r2.xy = r2.xy * float2(2,2) + float2(-1,-1);
  float2 nBase0 = r2.xy;   // (26/10 b) el normal map tal cual (poros para la piel real, union del brazo)
  // (26/10, v12.7) musculos con textura: el VS 1c02a92a (pes_limbs.hlsli) manda en v1.xyz = (definicion, venas, 7) cuanto marcar
  // el musculo que trabaja y las venas (brazos, piernas y manos de TorsoBody); en el resto del cuerpo v1.z no es 7 y no cambia nada
  float4 P305 = IniParams.Load(int2(305,0));
  bool lbOn = P305.x > 0.5 && v1.z > 6.5 && v1.z < 7.5;
  float lbDef = lbOn ? clamp(v1.x, 0.0, 2.0) : 0.0, lbVein = lbOn ? clamp(v1.y, 0.0, 2.0) : 0.0, lbVM = 0.0;

  // a mas de 20 cm del centro y arriba de la cadera; manos incluidas); x285 = (prendido, texturas, definicion)
  bool vrBody = v3.w > 1.999 && v3.w < 3.001;
  float4 P285 = IniParams.Load(int2(285,0));
  float vrT = (vrBody && P285.x > 0.5) ? PesVariantTCode(v6.w) : 0.0;
  float vrWa = max(vrT, 0.0), vrWr = max(-vrT, 0.0);
  float vrTm = vrBody ? PesSkinTorso(v6.w) : 0.0;
  float vrArm = vrBody ? saturate((abs(w2.w) - 0.20) / 0.05) * step(1.0, w2.z) * (1.0 - vrTm) : 0.0;
  float vrDefK = P285.z > 0.0 ? P285.z : 1.0, vrTexK = P285.y > 0.0 ? P285.y : 1.0;
  // (26/10) los normal maps propios de TorsoBody (4096: torsob_nrm, arm_nrm, thigh_nrm) tienen el verde al reves para el juego:
  // la vista del juego es de mano izquierda y B = cross(N, T) * w queda invertida respecto de la textura (el relieve se veia
  // iluminado desde abajo: biceps y espalda "hacia adentro", sobre todo con la definicion alta). x286 = 1 lo corrige al final
  // (base, definicion, mapas de las variantes y venas estan todos en la convencion de la textura)
  uint tbW = 0, tbH = 0; g_tex_normal.GetDimensions(tbW, tbH);
  bool tbFlip = vrBody && tbW >= 4096 && IniParams.Load(int2(286,0)).x > 0.5;
  // (26/10 c) pelvis "sin short" (SinCamiseta, tecla 4): la DLL enlaza t23 solo en ese dibujo; sus texturas (t20, t21, t23, t24)
  // cubren el cuadrante libre de abajo a la derecha de la textura del torso (u, v 0.5..1), reescalado a 0..1
  uint paW = 0, paH = 0; pesPelvisAlb.GetDimensions(paW, paH);
  bool pel = vrBody && paW > 0;
  float2 pelUV = saturate((v2.xy - 0.5) * 2.0);
  float4 P293 = IniParams.Load(int2(293,0));
  if (vrBody && tbW >= 4096 && vrTm < 0.5 && P293.w > 0.5) {
    // (26/10 b) union del brazo con el torso (w293): el normal map del torso es casi liso en los 2 cm de la costura y el del
    // brazo no (relieve del doble): cerca del anillo del hombro (del lado del brazo) el relieve del brazo baja al del torso
    float sA = dot(float3(abs(w2.w), w2.z, 0.02) - float3(0.2631, 1.3704, 0.0195), float3(0.7611, -0.6432, -0.0839));
    if (abs(w2.w) > 0.20 && w2.z > 1.15) r2.xy *= lerp(0.55, 1.0, smoothstep(0.0, 0.03, sA));
  }
  {   // (26/10 b) base esculpida (x289): el relieve que dibuja el normal map de TorsoBody ahora esta en la malla (filas 7..9 de los
      // datos del VS): esta textura le saca a la textura lo que la malla ya muestra (si no, se veria dos veces) y deja la espalda
      // suave (las lineas duras al 30 %). La DLL la enlaza solo en los torsos
    uint bfW = 0, bfH = 0; pesBaseFix.GetDimensions(bfW, bfH);
    bool bfArm = bfW > 0 && bfH == bfW;   // la del brazo cubre todo su atlas; la del torso, v 0..0.5 (2:1)
    if (vrBody && bfW > 0 && (pel || ((bfArm || v2.y < 0.5) && IniParams.Load(int2(289,0)).x > 0.5))) {
      r2.xy += pesBaseFix.SampleBias(g_samplerLinear_Clamp_s, pel ? pelUV : (bfArm ? v2.xy : float2(v2.x, v2.y * 2.0)), bBN).xy;
      float l2 = dot(r2.xy, r2.xy); if (l2 > 0.94) r2.xy *= rsqrt(l2 / 0.94);
    }
  }
  {   // (24/10) definicion muscular (x172, menu Musculatura): agranda o achica solo el relieve grande del normal map

    float defM = IniParams.Load(int2(172,0)).x; defM = defM > 0.01 ? clamp(defM, 0.0, 3.0) : 1.0;

      float4 stH = IniParams.Load(int2(194,0)); int stI = (int)round(v6.w) - 1000;
      if (stH.y > 0.5 && stI >= 0 && stI < min((int)(stH.x + 0.5), 20)) { float dP = IniParams.Load(int2(195 + 3 * stI, 0)).w; if (dP > 0.01) defM = clamp(dP, 0.0, 3.0); }
    }
    defM += lbDef;   // (26/10, v12.7) el musculo que trabaja: mas relieve solo ahi
    // (26/10) variantes: torso 1 + 0.15 atletico - 0.75 robusto; brazos 1 + 0.20 atletico - 0.50 robusto (el atletico ya marca
    // con la malla y los mapas delta; con mas definicion las sombras del normal map se veian manchadas)
    defM *= 1.0 + vrDefK * (vrTm * (0.15 * vrWa - 0.75 * vrWr) + vrArm * (0.20 * vrWa - 0.50 * vrWr));

      float3 nM = g_tex_normal.SampleBias(g_samplerLinear_Clamp_s, v2.xy, bBN + 3.5).xyw;
      float2 mM = float2(nM.x * nM.z, nM.y) * 2.0 - 1.0;
      r2.xy += (defM - 1.0) * mM;
      float l2 = dot(r2.xy, r2.xy); if (l2 > 0.94) r2.xy *= rsqrt(l2 / 0.94);
    }
  }
  if (abs(vrT) > 0.001) {   // (26/10) variantes del torso: el robusto un poco mas liso; mapas delta (pliegues, estrias / Adonis,
                            // serratos, trapecio...) en el espacio UV del torso (v 0..0.5 -> 0..1); venas de los brazos
    r2.xy *= 1.0 - 0.3 * vrWr * vrTm * vrDefK;

  }
  {   // (26/10) columna marcada (todos los torsos de TorsoBody, con o sin variante): surco de la columna y erectores (x288 = fuerza;
      // la DLL enlaza t19 solo en los dibujos del torso). El robusto la tapa un poco y el atletico la marca un poco mas
    uint spW = 0, spH = 0; pesSpine.GetDimensions(spW, spH);
    float spK = IniParams.Load(int2(288,0)).x;
    if (vrBody && spW > 0 && spK > 0.001 && v2.y < 0.5) {
      float2 sd = pesSpine.SampleBias(g_samplerLinear_Clamp_s, float2(v2.x, v2.y * 2.0), bBN).xy;
      r2.xy += sd * min(spK, 3.0) * (1.0 - 0.5 * vrWr) * (1.0 + 0.25 * vrWa);
      float l2 = dot(r2.xy, r2.xy); if (l2 > 0.94) r2.xy *= rsqrt(l2 / 0.94);
    }
  }
  if (lbVein > 0.002) {   // (26/10, v12.7) venas: relieve de la textura de venas (mismo espacio tangente que arm_nrm) + mascara para el tono
    uint lvW = 0, lvH = 0; pesVeins.GetDimensions(lvW, lvH);
    if (lvW > 0) {
      float4 lv = pesVeins.SampleBias(g_samplerLinear_Clamp_s, v2.xy, bBN);
      r2.xy += lbVein * (lv.xy * 2.0 - 1.0);
      float l2 = dot(r2.xy, r2.xy); if (l2 > 0.94) r2.xy *= rsqrt(l2 / 0.94);
      lbVM = saturate(lbVein * lv.z);
    }
  }
  if (hpK > 0.001 && P94h.w > 0.5) {   // (20/10) micro normal de poros, misma escala y mismo plano UV que la cara: sigue sin costura
    float near = saturate((5.0 - hpFpx) / 3.5);
    if (near > 0.001) {
      float4 guide = pesFacePores.Sample(g_samplerLinear_Clamp_s, v2.xy);
      hpTZone = guide.b;
      float4 mn = pesSkinMicro.SampleGrad(g_samplerLinear_Clamp_s, frac(hpMuv), hpMdx, hpMdy);
      float2 muv2 = hpMuv * 0.5 + 0.37;
      float4 mn2 = pesSkinMicro.SampleGrad(g_samplerLinear_Clamp_s, frac(muv2), hpMdx * 0.5, hpMdy * 0.5);
      mn = lerp(mn, mn2, guide.g * 0.7);
      float3 tn = mn.xyz * 2.0 - 1.0;
      float skinM = lerp(0.5, 1.0, pesZones.Sample(g_samplerLinear_Clamp_s, v2.xy).a);
      float str = clamp(P94h.x, 0.0, 2.0) * near * 1.1 * skinM * (0.25 + 1.0 * guide.r) * hpK;
      r2.xy += tn.xy * str;
      hpPoreA = mn.a * near * (0.3 + guide.r);
      hpPoreOcc = 1.0 - hpPoreA * guide.a * 0.55 * saturate(P94h.x);
      float4 ma = pesSkinMicro.SampleGrad(g_samplerLinear_Clamp_s, frac(hpMuv), hpMdx * 6.0, hpMdy * 6.0);
      hpMicroVar = saturate((1.0 - length(ma.xyz * 2.0 - 1.0)) * 8.0) * saturate(P94h.x * 0.7) * skinM;
    }
  }
  if ((bodyOn || headPiece) && P35b.w > 0.5) {   // cavidades desde el normal map (codo, rodilla, nudillos, tendones)
    float bnz = sqrt(saturate(1.0 - dot(r2.xy, r2.xy)));
    bCav = 1.0 - saturate(P35b.x) * smoothstep(0.02, 0.35, 1.0 - bnz) * 0.55;
  }
  if (tbFlip) r2.y = -r2.y;   // (26/10) TorsoBody: verde a la convencion del juego (x286)
  r1.xyz = r2.yyy * r1.xyz;
  r0.xyz = r0.xyz * r2.xxx + r1.xyz;
  r0.w = dot(v6.xyz, v6.xyz);
  r0.w = rsqrt(r0.w);
  r1.xyz = v6.xyz * r0.www;
  r0.w = -r2.x * r2.x + 1;
  r0.w = -r2.y * r2.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.w = rsqrt(r0.w);
  r0.w = r1.w * r0.w;
  r0.xyz = r1.xyz * r0.www + r0.xyz;
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r0.xyz = r0.xyz * r0.www;
  r0.w = dot(v3.xyz, v3.xyz);
  r0.w = rsqrt(r0.w);
  r1.xyz = v3.xyz * r0.www;
  r0.w = dot(r0.xyz, r1.xyz);
  r0.w = saturate(1 + -r0.w);
  r0.w = log2(r0.w);
  r0.w = g_psMaterial.m_materials[1].x * r0.w;
  r0.w = exp2(r0.w);
  r1.xy = g_tex_srm.SampleBias(g_samplerLinear_Clamp_s, v2.xy, bBD).xy;
