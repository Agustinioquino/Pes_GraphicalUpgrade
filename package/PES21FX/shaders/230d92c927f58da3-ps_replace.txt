// === PES21 SOL: color del sol segun su altura (calido al atardecer, blanco calido al mediodia) ===
float3 PESSunColor() {
  float3 c = g_psLight.m_lightColor0.xyz;
  float4 S = IniParams.Load(int2(4,0));      // z4 = realismo del sol (0 = original), w4 = intensidad
  bool sunReal = S.z > 0;
  float3 L = normalize(g_psLight.m_lightDir0.xyz);
  float elev = saturate(abs(dot(L, float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12))));
  float low = 1 - smoothstep(0.10, 0.60, elev);
  float3 tint = lerp(float3(1.0, 0.962, 0.905), float3(1.0, 0.80, 0.60), low);
  const float3 LW = float3(0.2126, 0.7152, 0.0722);
  float lum = dot(c, LW);
  float warmO = c.r / max(c.b, 1e-4);
  float k = (sunReal ? saturate(S.z) : 0.0) * saturate((warmO - 0.85) * 4.0);   // de noche (luz fria de estadio) casi no toca
  float3 res = lerp(c, lum * tint / dot(tint, LW), k);
  res *= (S.w > 0 ? lerp(1.0, S.w, k) : 1.0);
  // PES21 curva del dia (18/10): intensidad y calidez del sol segun su altura (solo sol real, no torres)
  { PESDay dy = PESDayCurve(elev, saturate((warmO - 0.85) * 4.0)); res *= dy.sun; res = PESWarmTint(res, dy.warm); }
  res *= PESDaySunMul(saturate((warmO - 0.85) * 4.0));   // PES21 contraste de dia (x109)
  // PES21 atmosfera (17/10): con el sol bajo el archivo de luz deja el sol muy apagado; y54 lo compensa (solo sol real,
  // no luz de torres), proporcional a lo bajo que esta.
  float4 ATM = IniParams.Load(int2(54,0));
  if (ATM.w > 0.5) { float lowS = 1.0 - smoothstep(0.05, 0.40, elev); float kSun = saturate((warmO - 0.85) * 4.0);
    res *= 1.0 + max(ATM.y, 0.0) * lowS * kSun; }
  return res;
}
// === FIN SOL ===
// PES21 (18/10): terminador suave SIN correr la luz. Igual a max(NdotL,0) cuando NdotL >= w (lado iluminado intacto,
// misma direccion que el sol); entre -w y w una curva suave (C1) que solo entra un poco en la sombra.
float3 PESSoftTerm(float nl, float3 w) {
  w = max(w, 1e-3);
  float3 x = nl.xxx;
  float3 s = (x + w) * (x + w) / (4.0 * w);
  return x >= w ? x : (x > -w ? s : 0.0);
}

  r1.xyw = (uint3)r1.xyw;
  r0.z = (uint)r1.x << 1;
  r0.w = (uint)r1.w >> 1;
  r0.z = (int)r0.w | (int)r0.z;
  bitmask.w = ((~(-1 << 31)) << 1) & 0xffffffff;  r0.w = (((uint)r1.y << 1) & bitmask.w) | ((uint)r1.w & ~bitmask.w);
  r1.xy = (uint2)r0.zw;
  r0.zw = r1.xy * float2(0.00097703957,0.00097703957) + float2(-1,-1);

  r6.xyzw = r2.wwww * r2.wwww;
  if (r1.x != 0) {

      float sheen = pow(1.0 - nvS, 4.0) * saturate(dot(nS, LS) * 0.6 + 0.4);

    }
  }
  if (isSkin) {
    // --- reconstruccion de normal, vista y luz ---
    float4 nb = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q = (uint3)(float3(1023,1023,3) * nb.xyw);
    uint ex = (q.x << 1) | (q.z >> 1);
    uint ey = (q.y << 1) | (q.z & 1);
    float2 e = float2(ex, ey) * 0.00097703957 - 1;
    float3 n = float3(e.xy, 1 - abs(e.x) - abs(e.y));
    if (n.z < 0) n.xy = (1 - abs(e.yx)) * (e.xy >= 0 ? 1 : -1);
    n = normalize(n); n.z = -n.z;
    float2 pp = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V = -normalize(float3(pp, 1));
    float3 L = normalize(pesLDir.xyz);
    float3 lightC = PESSunColor() * g_psScene.m_exposure.zzz;
    // --- transmision a contraluz ---
    float3 Lt = normalize(L + n * 0.3);
    float tv = pow(saturate(dot(V, -Lt)), 3.0);
    float thick = saturate(trans * 1.6);
    float4 P76 = IniParams.Load(int2(76,0));   // PES21 SSS v2: x76 intensidad | y76 zonas que transmiten
    float gateT = lerp(smoothstep(0.45, 0.85, thick), 1.0, saturate(P76.y));
    float3 transm = thick * gateT * tv * float3(1.0, 0.24, 0.10) * P5.x * (P76.x > 0.0 ? P76.x : 1.0);   // 18/10 v2
    o0.xyz += transm * lightC * sqrt(saturate(r6.x));
  }
  r0.xyz = r0.xyz * r6.xyz;
  if (isSkin) {
    r0.xyz *= skinSpec;
    float4 nb2 = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q2 = (uint3)(float3(1023,1023,3) * nb2.xyw);
    float2 e2 = float2((q2.x << 1) | (q2.z >> 1), (q2.y << 1) | (q2.z & 1)) * 0.00097703957 - 1;
    float3 n2 = float3(e2.xy, 1 - abs(e2.x) - abs(e2.y));
    if (n2.z < 0) n2.xy = (1 - abs(e2.yx)) * (e2.xy >= 0 ? 1 : -1);
    n2 = normalize(n2); n2.z = -n2.z;
    float2 pp2 = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V2 = -normalize(float3(pp2, 1));
    float3 L2 = normalize(pesLDir.xyz);
    float3 H2 = normalize(L2 + V2);
    float nl2 = saturate(dot(n2, L2));
    // === PES21 (19/10) brillo de piel fisico: x103 el brillo toma el color del sol (no el tinte del material -> menos cera
    // amarilla), y103 Fresnel de piel (F0 0.028: debil de frente, crece en la silueta), z103 fuerza total, w103 on/off
    { float4 Q103 = IniParams.Load(int2(103,0));
      if (Q103.w > 0.5) {
        float nv2 = saturate(dot(n2, V2));
        float rim = pow(1.0 - nv2, 4.0);
        float3 sunN = PESSunColor(); float sl = max(dot(sunN, float3(0.3, 0.59, 0.11)), 1e-4); sunN /= sl;
        float lumS = dot(r1.xyz, float3(0.3, 0.59, 0.11));
        r1.xyz = lerp(r1.xyz, lumS * sunN, saturate(Q103.x));
        r1.xyz *= lerp(1.0, lerp(0.55, 1.7, rim), saturate(Q103.y)) * (Q103.z > 0 ? Q103.z : 1.0);
      } }
    // lobulo fino (pelicula de grasa/sudor): GGX con rugosidad 0.18
    float a2 = 0.18 * 0.18; float a4 = a2 * a2;
    float nh = saturate(dot(n2, H2));
    float dd = nh * nh * (a4 - 1) + 1;
    float D2 = a4 / (3.14159 * dd * dd);
    float F2 = 0.028 + 0.972 * pow(1 - saturate(dot(V2, H2)), 5);
    float sharp = D2 * F2 * nl2 * 0.25 * P5.y;
    r0.xyz += sharp * PESSunColor() * g_psScene.m_exposure.zzz * shadowScalar * saturate(specInt * 2 + 0.3);
  }
  { float dbgW12 = IniParams.Load(int2(12,0)).w; if (dbgW12 > 3.5 && dbgW12 < 4.5) { float lum = dot(o0.xyz, float3(0.2126, 0.7152, 0.0722)); o0.xyz = lerp(o0.xyz, float3(10.0, 0.0, 0.0), saturate((1.0 - csDbg) * 3.0)); r0.xyz = 0; } }   // diagnostico 4: sombras de contacto
  if (IniParams.Load(int2(12,0)).w > 2.5 && IniParams.Load(int2(12,0)).w < 3.5 && abs(skinTag - 0.98039) < 0.003) { o0.xyz = float3(4.0, 2.0, 0.0); r0.xyz = 0; }   // diagnostico: naranja = esta variante (sin shadow map) ilumina la piel
  o1.xyz = min(float3(10000,10000,10000), r0.xyz);
  o0.w = 1;
  o1.w = 1;
  // PES21 inspector (19/10): LUZ PLANA (x89 = 1): irradiancia constante (y89) sin direccion, sombras ni brillos, para ver
  // texturas y materiales sin la iluminacion. z89 = 1 deja el brillo especular.
  { float4 P89 = IniParams.Load(int2(89,0));
    if (P89.x > 0.5) { float fl = P89.y > 0.0 ? P89.y : 1.0; float3 c89 = pesLCol.xyz; float m89 = max(max(c89.x, c89.y), c89.z); if (m89 < 1e-3) m89 = 1.0;
      o0.xyz = (c89 / m89) * fl * 0.6; if (P89.z < 0.5) o1.xyz = 0; } }
  o2.xyzw = r6.xyzw;
  return;
}
