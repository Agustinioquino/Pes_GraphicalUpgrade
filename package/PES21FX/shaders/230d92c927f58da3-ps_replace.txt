// Parametros: x3 ancho SSS, y3 sangrado en sombras, z3 on/off, w3 especular base piel,
//             x5 transmision, y5 lobulo fino, z5 suavizado de sombra en piel

// === PES21 SOL: color del sol segun su altura (calido al atardecer, blanco calido al mediodia) ===
float3 PESSunColor() {
  float3 c = g_psLight.m_lightColor0.xyz;
  float4 S = IniParams.Load(int2(4,0));      // z4 = realismo del sol (0 = original), w4 = intensidad
  if (S.z <= 0) return c;
  float3 L = normalize(g_psLight.m_lightDir0.xyz);
  float elev = saturate(abs(dot(L, float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12))));
  float low = 1 - smoothstep(0.10, 0.60, elev);
  float3 tint = lerp(float3(1.0, 0.962, 0.905), float3(1.0, 0.80, 0.60), low);
  const float3 LW = float3(0.2126, 0.7152, 0.0722);
  float lum = dot(c, LW);
  float warmO = c.r / max(c.b, 1e-4);
  float k = saturate(S.z) * saturate((warmO - 0.85) * 4.0);   // de noche (luz fria de estadio) casi no toca
  float3 res = lerp(c, lum * tint / dot(tint, LW), k);
  return res * (S.w > 0 ? lerp(1.0, S.w, k) : 1.0);
}
// === FIN SOL ===
void main(
  float4 v0 : TEXCOORD0,
  float4 v1 : TEXCOORD1,
  float4 v2 : SV_Position0,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2)
{
  float4 r0,r1,r2,r3,r4,r5,r6,r7,r8;
  uint4 bitmask, uiDest;
  float4 fDest;

  float4 P3 = IniParams.Load(int2(3,0));
  float4 P5 = IniParams.Load(int2(5,0));
  bool  sssOn    = P3.z > 0.5;
  float sssWidth = P3.x > 0 ? P3.x : 1.0;
  float sssBleed = P3.y;
  float skinSpec = P3.w > 0 ? P3.w : 1.0;
  float rawNdotL = 0, trans = 0, transFloor = 0, skinTag = 0, shadowScalar = 1;

  r0.xy = v1.xy / v1.ww;
  r1.xyzw = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, r0.xy).xyzw;
  r2.xyzw = g_SpecularTexture.Sample(g_samplerPoint_Wrap_s, r0.xy).zxyw;
  skinTag = r2.x;  float specInt = r2.z;  float2 uvG = r0.xy;
  r1.xyw = float3(1023,1023,3) * r1.xyw;
  r1.xyw = (uint3)r1.xyw;
  r0.z = (uint)r1.x << 1;
  r0.w = (uint)r1.w >> 1;
  r0.z = (int)r0.w | (int)r0.z;
  bitmask.w = ((~(-1 << 31)) << 1) & 0xffffffff;  r0.w = (((uint)r1.y << 1) & bitmask.w) | ((uint)r1.w & ~bitmask.w);
  r1.xy = (uint2)r0.zw;
  r0.zw = r1.xy * float2(0.00097703957,0.00097703957) + float2(-1,-1);
  r1.xyw = float3(1,1,1) + -abs(r0.zwz);
  r3.z = r1.x + -abs(r0.w);
  r1.x = cmp(r3.z >= 0);
  r4.xy = cmp(r0.zw >= float2(0,0));
  r4.xy = r4.xy ? float2(1,1) : float2(-1,-1);
  r1.yw = r4.xy * r1.yw;
  r3.xy = r1.xx ? r0.zw : r1.yw;
  r0.z = dot(r3.xyz, r3.xyz);
  r0.z = rsqrt(r0.z);
  r3.xyz = r3.xyz * r0.zzz;
  r0.z = r1.z * 1023 + 0.5;
  r0.z = (uint)r0.z;
  r0.w = (int)r0.z & 15;
  int matType = (int)r0.w;
  r1.x = cmp((int)r0.w == 1);
  if (r1.x != 0) {
    r4.x = 0;
    r5.y = 0;
  } else {
    r0.z = (uint)r0.z >> 4;
    r1.y = cmp((int)r0.w == 3);
    if (r1.y != 0) {
      r1.y = (uint)r0.z;
      r5.y = 0.0158730168 * r1.y;
      r4.x = 0;
    } else {
      r0.z = (uint)r0.z;
      r5.y = 0.0158730168 * r0.z;
      r2.x = 0;
      r4.x = r2.w;
    }
  }
  r0.x = g_DepthTexture.Sample(g_samplerPoint_Wrap_s, r0.xy).x;
  r0.yz = v0.xy / v0.ww;
  r0.yz = -g_psScene.m_cameraCenterOffset.xy + r0.yz;
  r0.yz = g_psScene.m_projectionParam.xy * r0.yz;
  r0.x = -g_psScene.m_projectionParam.w + r0.x;
  r6.z = g_psScene.m_projectionParam.z / r0.x;
  r6.xy = r6.zz * r0.yz;
  r0.x = dot(r6.xyz, r6.xyz);
  r0.x = rsqrt(r0.x);
  r0.xyz = r6.xyz * r0.xxx;
  r1.yzw = PESSunColor() * g_psScene.m_exposure.zzz;
  r2.w = -g_psObject.m_localParam[2].x + r6.z;
  r2.w = saturate(g_psObject.m_localParam[2].y * r2.w);
  r2.w = 1 + -r2.w;
  r6.xyzw = r2.wwww * r2.wwww;
  if (r1.x != 0) {
    r7.y = r2.x * 0.959999979 + 0.0399999991;
    r8.xyz = float3(1,1,1);
    r5.xy = float2(0,0);
  } else {
    r0.w = cmp((int)r0.w == 3);
    if (r0.w != 0) {
      r7.y = r2.x * 0.959999979 + 0.0399999991;
      r8.xyz = float3(0.899999976,1,0.800000012);
      r5.x = 1;
    } else {
      r4.yz = float2(0.75,0.25);
      r7.y = g_MaterialTexture.SampleLevel(g_samplerPoint_Clamp_s, r4.xy, 0).x;
      r8.xyzw = g_MaterialTexture.SampleLevel(g_samplerPoint_Clamp_s, r4.xz, 0).xyzw;
      r5.x = r8.w;
    }
  }
  r3.w = -r3.z;
  r0.w = dot(g_psLight.m_lightDir0.xyz, r3.xyw);
  r1.x = dot(-r0.xyz, r3.xyw);
  r2.x = r1.x + r1.x;
  r4.xyz = r3.xyw * -r2.xxx + -r0.xyz;
  r4.xyz = -g_psLight.m_lightDir0.xyz + -r4.xyz;
  r2.x = dot(r4.xyz, r4.xyz);
  r2.x = sqrt(r2.x);
  r2.x = saturate(g_psLight.m_lightColor0.w / r2.x);
  r4.xyz = r2.xxx * r4.xyz + g_psLight.m_lightDir0.xyz;
  r2.x = dot(r4.xyz, r4.xyz);
  r2.x = rsqrt(r2.x);
  r0.xyz = r4.xyz * r2.xxx + -r0.xyz;
  r2.x = dot(r0.xyz, r0.xyz);
  r2.x = rsqrt(r2.x);
  r0.xyz = r2.xxx * r0.xyz;
  r2.x = saturate(dot(r0.xyz, r3.xyw));
  r2.w = -r0.w * r0.w + 1;
  r2.w = r2.w * r5.y + r0.w;
  rawNdotL = r0.w; trans = r5.y;
  r3.x = r5.y * r5.x;
  transFloor = r3.x;
  r2.w = saturate(max(r3.x, r2.w));
  r0.w = saturate(r0.w);
  r3.xyz = r2.www * r1.yzw;
  bool isSkin = sssOn && (matType == 0) && (trans > 0.004) && (abs(skinTag - 0.98039) < 0.003);
  if (isSkin) {
    float3 w = saturate(trans) * sssWidth * float3(1.0, 0.42, 0.28);
    float3 d = saturate((rawNdotL + w) / (1.0 + w));
    d = max(d, transFloor.xxx);
    r3.xyz = d * r1.yzw;
  }
  r4.xyz = r8.xyz * r2.zzz;
  r7.x = dot(r0.xyz, g_psLight.m_lightDir0.xyz);
  r0.x = g_lut1.Sample(g_samplerLinear_Clamp_s, r7.xy).x;
  r1.x = saturate(r1.x);
  r0.y = max(0.0199999996, r2.y);
  r0.y = r0.y * r0.y;
  r0.z = 0.5 * r0.y;
  r2.y = r0.y * r0.y;
  r2.x = r2.x * r2.x;
  r2.z = r0.y * r0.y + -1;
  r2.x = r2.x * r2.z + 1;
  r2.x = r2.x * r2.x;
  r2.x = r2.y / r2.x;
  r0.y = -r0.y * 0.5 + 1;
  r2.y = r0.w * r0.y + r0.z;
  r0.y = r1.x * r0.y + r0.z;
  r0.y = r2.y * r0.y;
  r0.y = 1 / r0.y;
  r0.x = r2.x * r0.x;
  r0.x = r0.x * r0.y;
  r0.x = r0.x * r0.w;
  r0.x = 0.25 * r0.x;
  r0.xyz = r0.xxx * r1.yzw;
  r0.xyz = r0.xyz * r4.xyz;
  float csDbg = 1.0;
  // === PES21 sombras de contacto (trazado en pantalla hacia el sol) ===
  {
    float2 sC = v0.xy / v0.ww;
    float2 uvC = v1.xy / v1.ww;
    float2 dS = float2(ddx(sC.x), ddy(sC.y));
    float2 dU = float2(ddx(uvC.x), ddy(uvC.y));
    dS = (abs(dS) < 1e-9) ? 1e-9 : dS;
    float2 Am = dU / dS;
    float2 Bm = uvC - Am * sC;
    float4 P12c = IniParams.Load(int2(12,0));
    float csK = P12c.y < 0 ? 0.0 : (P12c.y > 0 ? P12c.y : 0.6);
    float csLen = P12c.z > 0 ? P12c.z : 0.35;
    float dC = g_DepthTexture.SampleLevel(g_samplerPoint_Wrap_s, uvC, 0).x;
    float zC = g_psScene.m_projectionParam.z / (dC - g_psScene.m_projectionParam.w);
    if (csK > 0 && zC > 0.1 && zC < 150.0) {
      float3 Pv = float3((sC - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy * zC, zC);
      float3 Lc = normalize(g_psLight.m_lightDir0.xyz);
      float ign = frac(52.9829189 * frac(dot(v2.xy, float2(0.06711056, 0.00583715))));
      float occ = 0;
      float3 Q = Pv + Lc * (0.015 + 0.0025 * zC);
      [loop] for (int ci = 0; ci < 16; ci++) {
        float t = ((ci + ign) / 16.0) * csLen;
        float3 R = Q + Lc * t;
        if (R.z < 0.1) break;
        float2 s = R.xy / (R.z * g_psScene.m_projectionParam.xy) + g_psScene.m_cameraCenterOffset.xy;
        float2 uv = Am * s + Bm;
        if (any(uv < 0.0) || any(uv > 1.0)) break;
        float dS2 = g_DepthTexture.SampleLevel(g_samplerPoint_Wrap_s, uv, 0).x;
        float zS = g_psScene.m_projectionParam.z / (dS2 - g_psScene.m_projectionParam.w);
        float dz = R.z - zS;
        if (dz > 0.012 + 0.0012 * R.z && dz < 0.06 + 0.0015 * R.z && (zC - zS) > 0.04 + 0.002 * zC) { occ = 1.0 - 0.5 * (ci / 16.0); break; }
      }
      float fade = saturate((40.0 - zC) / 15.0);
      float cs = 1.0 - saturate(occ * fade * csK);
      csDbg = cs;
      if (isSkin) cs = lerp(1.0, cs, 0.4);
      r6.xyz *= cs;
    }
  }
  float3 shadowRGB = r6.xyz;
  shadowScalar = saturate(r6.x);
  if (isSkin) {
    float sh = saturate(r6.x);
    // sombra mas suave sobre la piel (la luz se difunde dentro del tejido)
    sh = lerp(sh, smoothstep(0.0, 1.0, sqrt(sh)), saturate(P5.z));
    float3 shC = pow(max(sh, 1e-4), float3(0.55, 1.0, 1.12));
    shadowRGB = lerp(sh.xxx, shC, saturate(trans * 2.0) * sssBleed);
  }
  o0.xyz = r3.xyz * shadowRGB;
  // === PES21 tela: brillo aterciopelado en los bordes (sheen) para ropa/uniformes ===
  if (!isSkin && matType == 0) {
    float4 spc = g_SpecularTexture.Sample(g_samplerPoint_Wrap_s, uvG).zxyw;
    float4 P5s = IniParams.Load(int2(5,0));
    float shK = 0.0;   // PES21 v2: brillo de tela apagado (titilaba en los bordes)
    if (shK > 0 && spc.y > 0.42 && spc.x > 0.99) {
      float4 nbS = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
      uint3 qS = (uint3)(float3(1023,1023,3) * nbS.xyw);
      float2 eS = float2((qS.x << 1) | (qS.z >> 1), (qS.y << 1) | (qS.z & 1)) * 0.00097703957 - 1;
      float3 nS = float3(eS.xy, 1 - abs(eS.x) - abs(eS.y));
      if (nS.z < 0) nS.xy = (1 - abs(eS.yx)) * (eS.xy >= 0 ? 1 : -1);
      nS = normalize(nS); nS.z = -nS.z;
      float2 ppS = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
      float3 VS = -normalize(float3(ppS, 1));
      float3 LS = normalize(g_psLight.m_lightDir0.xyz);
      float nvS = saturate(dot(nS, VS));
      float sheen = pow(1.0 - nvS, 4.0) * saturate(dot(nS, LS) * 0.6 + 0.4);
      float clothW = saturate((spc.y - 0.42) * 4.0);
      o0.xyz += sheen * 0.45 * shK * clothW * PESSunColor() * g_psScene.m_exposure.zzz * shadowScalar;
    }
  }
  if (isSkin) {
    // --- reconstruccion de normal, vista y luz ---
    float4 nb = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q = (uint3)(float3(1023,1023,3) * nb.xyw);
    uint ex = (q.x << 1) | (q.z >> 1);
    uint ey = (q.y << 1) | (q.z & 1);
    float2 e = float2(ex, ey) * 0.00097703957 - 1;
    float3 n = float3(e.xy, 1 - abs(e.x) - abs(e.y));
    if (n.z < 0) n.xy = (1 - abs(e.yx)) * (e.xy >= 0 ? 1 : -1);
    n = normalize(n); n.z = -n.z;
    float2 pp = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V = -normalize(float3(pp, 1));
    float3 L = normalize(g_psLight.m_lightDir0.xyz);
    float3 lightC = PESSunColor() * g_psScene.m_exposure.zzz;
    // --- transmision a contraluz ---
    float3 Lt = normalize(L + n * 0.3);
    float tv = pow(saturate(dot(V, -Lt)), 3.0);
    float thick = saturate(trans * 1.6);
    float3 transm = thick * tv * float3(1.0, 0.24, 0.10) * P5.x;
    o0.xyz += transm * lightC * sqrt(saturate(r6.x));
  }
  r0.xyz = r0.xyz * r6.xyz;
  if (isSkin) {
    r0.xyz *= skinSpec;
    float4 nb2 = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q2 = (uint3)(float3(1023,1023,3) * nb2.xyw);
    float2 e2 = float2((q2.x << 1) | (q2.z >> 1), (q2.y << 1) | (q2.z & 1)) * 0.00097703957 - 1;
    float3 n2 = float3(e2.xy, 1 - abs(e2.x) - abs(e2.y));
    if (n2.z < 0) n2.xy = (1 - abs(e2.yx)) * (e2.xy >= 0 ? 1 : -1);
    n2 = normalize(n2); n2.z = -n2.z;
    float2 pp2 = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V2 = -normalize(float3(pp2, 1));
    float3 L2 = normalize(g_psLight.m_lightDir0.xyz);
    float3 H2 = normalize(L2 + V2);
    float nl2 = saturate(dot(n2, L2));
    // lobulo fino (pelicula de grasa/sudor): GGX con rugosidad 0.18
    float a2 = 0.18 * 0.18; float a4 = a2 * a2;
    float nh = saturate(dot(n2, H2));
    float dd = nh * nh * (a4 - 1) + 1;
    float D2 = a4 / (3.14159 * dd * dd);
    float F2 = 0.028 + 0.972 * pow(1 - saturate(dot(V2, H2)), 5);
    float sharp = D2 * F2 * nl2 * 0.25 * P5.y;
    r0.xyz += sharp * PESSunColor() * g_psScene.m_exposure.zzz * shadowScalar * saturate(specInt * 2 + 0.3);
  }
  o1.xyz = min(float3(10000,10000,10000), r0.xyz);
  o0.w = 1;
  o1.w = 1;
  o2.xyzw = r6.xyzw;
  return;
}
