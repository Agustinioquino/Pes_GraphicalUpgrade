// (12/11) pasto aplastado: el mayor de pisadas (R) y rastro de la pelota (G) en una uv de la cancha
// (13/11 tanda 2) z448 pisadas que se borran despacio: B = canal plastico de las pisadas (la DLL, solo con z448 > 0; en el mapa de 2 canales
// se lee 0). Para doblar y hundir R pasa a max(R, 0.55 B'), B' = (B - 0.012) / 0.988. x = como antes (R / G: los rebotes), y = con B
float2 PESFlatAt2(float2 uv) {
  float3 f = PESFlatMap.SampleLevel(g_samplerLinear_Clamp_s, uv, 0).xyz; float rP = f.x;
  if (IniParams.Load(int2(448, 0)).z > 0.0) rP = max(f.x, 0.55 * saturate((f.z - 0.012) / 0.988));
  return float2(max(f.x, f.y), max(rP, f.y));
}
float PESFlatAt(float2 uv) { return PESFlatAt2(uv).y; }
// (11/11 o) gradiente analitico del ruido de valor de PESGnoise (las mismas 4 esquinas): para correr los macollos de las hojas 3D
float2 PESGnGrad(float2 p) {
  float2 i = floor(p), f = frac(p), u = f * f * (3.0 - 2.0 * f), du = 6.0 * f * (1.0 - f);
  float a = PESGh21(i), b = PESGh21(i + float2(1, 0)), c = PESGh21(i + float2(0, 1)), d = PESGh21(i + 1.0);
  return du * float2(b - a + (a - b - c + d) * u.y, c - a + (a - b - c + d) * u.x);
}
// (11/11 f) azar entero (PCG) y Halton base 3 (raiz de cada tarjeta en la malla del mod, como BuildDense de la DLL): ver pesHv y pesCardRoot
uint PESBh(uint v) { uint s = v * 747796405u + 2891336453u; uint w = ((s >> ((s >> 28u) + 4u)) ^ s) * 277803737u; return (w >> 22u) ^ w; }
float PESHal3(uint n) { float f = 1.0, r = 0.0; [loop] for (int k = 0; k < 12; k++) { if (n == 0u) break; f *= (1.0 / 3.0); r += f * (float)(n % 3u); n /= 3u; } return r; }

// (11/11) hojas 3D de primerisimo plano: azar entero (PCG) por celda y por hoja, el mismo en todos los vertices de la hoja
float PESBu(uint s, uint k) { return (float)(PESBh(s ^ (k * 0x9E3779B9u)) & 0xFFFFFFu) * (1.0 / 16777216.0); }
// (13/11 tanda 2) ruido de valor periodico a lo largo de x cada nx celdas (D da la vuelta a los 8192 m y las frecuencias son n / 8192: sin
// costura). El indice x se corre 4 nx antes del modulo: a >= -1.01 x 8192 m, asi que nunca es negativo (modulo sin signo, bien definido)
float PESWv(float2 p, uint nx, uint seed) {
  float2 i = floor(p), f = p - i; f = f * f * (3.0 - 2.0 * f);
  uint x0 = (uint)((int)i.x + 4 * (int)nx) % nx;
  uint x1 = (x0 + 1u == nx) ? 0u : x0 + 1u;
  uint y0 = (uint)(int)i.y, y1 = y0 + 1u;
  return lerp(lerp(PESBu(x0 + y0 * 8209u, seed), PESBu(x1 + y0 * 8209u, seed), f.x),
              lerp(PESBu(x0 + y1 * 8209u, seed), PESBu(x1 + y1 * 8209u, seed), f.x), f.y);
}
#define PESWF(n) ((float)(n) / 8192.0)
float PESWg(float a, float c, float tg) { return 0.65 * PESWv(float2(a * PESWF(901), 0.035 * c + 0.03 * tg), 901u, 14u) + 0.35 * PESWv(float2(a * PESWF(3441), 0.16 * c + 0.08 * tg), 3441u, 15u); }
#define PESFR(f) (6.2831853 * frac(tg * (f) / 6.2831853))   // fase de (f) rad/s, precisa para cualquier tg
// (13/11 tanda 2) el viento unico (filas 494 / 495, wind.cpp): D (recorrido del aire, m mod 8192) y tg (reloj del juego, s mod 8192) vienen
// de la DLL. Mismo dibujo que V6 (frentes de racha, temblor al llegar, turbulencia de costado) pero el campo viaja con D: en camara lenta va
// mas lento y en la pausa queda quieto. Sin las rachas propias de x408 (la velocidad de 494.z ya trae la racha); y408 sigue girando el empuje
float2 PESGrassWindU(float2 root, float2 wd, float2 wc, float hb, float D, float tg, float W, out float gustOut) {
  float a = dot(root, wd) - D, c = dot(root, wc);   // = V6 con q = root - wd * D
  float dA = radians(clamp(IniParams.Load(int2(408, 0)).y, 0.0, 60.0)) * (PESWv(float2(a * PESWF(91), c / 90.0 + 0.01 * tg), 91u, 13u) * 2.0 - 1.0);
  float2 pd = wd, pc = wc;   // solo giran los vectores del empuje
  if (abs(dA) > 1e-4) { float cA = cos(dA), sA = sin(dA); pd = float2(cA * wd.x - sA * wd.y, sA * wd.x + cA * wd.y); pc = float2(-pd.y, pd.x); }
  float gN = PESWg(a, c, tg), gB = PESWg(a - 0.7, c, tg);   // gB = lo que llega en ~0.2 s
  float gust = smoothstep(0.38, 0.80, gN), rise = saturate((smoothstep(0.38, 0.80, gB) - gust) * 4.0);
  float aB = 0.55 + 0.9 * frac(hb * 7.13 + 0.31);
  float osc  = sin(PESFR(5.0 + 4.0 * hb) + hb * 6.2832 + 6.2831853 * frac(a * PESWF(1696)));   // = V6 a * 1.3
  float osc2 = sin(PESFR(1.1 + 0.9 * hb) + hb * 17.0 + c * 0.4);
  float push = gust * (0.34 + 0.10 * osc2), trem = (0.22 * gust + 0.78 * rise) * 0.11;
  float turb = (PESWv(float2(a * PESWF(7373), 0.9 * c), 7373u, 16u) - 0.5) * 0.30 * (0.3 + 0.7 * gust);
  gustOut = gust * saturate(W);
  return (pd * (0.03 + push + trem * osc) + pc * (turb + trem * 0.7 * cos(PESFR(4.1 + 3.3 * hb) + hb * 11.0))) * W * aB;
}
#undef PESFR
#undef PESWF

// (2/11) viento de las hojas en funcion del tiempo, para calcularlo dos veces: ahora (z23) y en el cuadro anterior (w23). Con la
// camara quieta el TAA del juego solo ve el movimiento que le dice el vector de velocidad (o6.zw / o7.y): antes se calculaba con la
// MISMA posicion (solo camara), asi que las hojas se movian con el viento pero el TAA las creia quietas y las promediaba (borroso).
// (12/11, Agustin: "ademas de rehacer el viento") VIENTO NUEVO (w433 = 1; 0 = el de antes). El anterior dejaba las hojas inclinadas todo el
// tiempo hacia el viento (hasta 0.55 del alto) con un aleteo parejo y rapido (2-3.5 Hz) que con pocos FPS se veia como ruido. Ahora:
//  - frentes de racha anchos y largos (~9 m por ~28 m, cruzados al viento) con ondas chicas adentro (~2.4 m), que viajan mas rapido cuanto
//    mas viento (1.6 .. 4.8 m/s); entre frentes el pasto queda casi quieto y parado (vuelve solo: no queda peinado)
//  - la hoja se dobla CON la racha y tiembla al LLEGAR el frente (sacudida que se apaga), no todo el tiempo; el temblor es lento (0.8-1.4 Hz)
//    y cada hoja tiene su fuerza (unas duras, otras flojas), su fase y su respiracion lenta
//  - turbulencia de costado que viaja con el viento (remolinos de ~1 m): las hojas no se mueven todas en el mismo plano
float2 PESGrassWindV6(float2 pesRoot, float2 wd, float2 wc, float hb, float tw, float W, out float gustOut) {
  // (13/11 tanda 2) la velocidad de los frentes sale del W del menu, no del W con las rachas de x408: con la racha el campo se corria
  // tw x dspd en cada cuadro (0.25-0.75 mm de ruido por cuadro que el TAA borroneaba). Las rachas cambian solo la fuerza
  float W0 = W;
  float4 P408 = IniParams.Load(int2(408, 0));
  float kG = clamp(P408.x, 0.0, 2.0);
  if (kG > 0.001) {
    float e1 = PESGnoise(float2(tw * 0.07, 3.7)) * 2.0 - 1.0, e2 = PESGnoise(float2(tw * 0.23 + dot(pesRoot, wd) * 0.015, 8.1)) * 2.0 - 1.0;
    W *= max(1.0 + kG * (0.55 * e1 + 0.30 * e2), 0.10);
  }
  float dA = radians(clamp(P408.y, 0.0, 60.0)) * (PESGnoise(float2(tw * 0.06, 1.3) + pesRoot * 0.012) * 2.0 - 1.0);
  if (abs(dA) > 1e-4) { float cA = cos(dA), sA = sin(dA); wd = float2(cA * wd.x - sA * wd.y, sA * wd.x + cA * wd.y); wc = float2(-wd.y, wd.x); }
  float spd = 1.6 + 3.2 * saturate(W0);
  float2 q = pesRoot - wd * tw * spd;
  float a = dot(q, wd), c = dot(q, wc);
  float gN = PESGnoise(float2(a * 0.11, c * 0.035) + 1.7) * 0.65 + PESGnoise(float2(a * 0.42, c * 0.16) + 7.3) * 0.35;
  float gB = PESGnoise(float2((a - 0.7) * 0.11, c * 0.035) + 1.7) * 0.65 + PESGnoise(float2((a - 0.7) * 0.42, c * 0.16) + 7.3) * 0.35;   // lo que llega en ~0.2 s
  float gust = smoothstep(0.38, 0.80, gN);
  float rise = saturate((smoothstep(0.38, 0.80, gB) - gust) * 4.0);   // el frente esta llegando
  float aB = 0.55 + 0.9 * frac(hb * 7.13 + 0.31);                     // hoja dura .. floja
  float osc = sin(tw * (5.0 + 4.0 * hb) + hb * 6.2832 + a * 1.3);
  float osc2 = sin(tw * (1.1 + 0.9 * hb) + hb * 17.0 + c * 0.4);
  float push = gust * (0.34 + 0.10 * osc2);
  float trem = (0.22 * gust + 0.78 * rise) * 0.11;
  float turb = (PESGnoise(q * 0.9 + 3.1) - 0.5) * 0.30 * (0.3 + 0.7 * gust);
  gustOut = gust * W;
  return (wd * (0.03 + push + trem * osc) + wc * (turb + trem * 0.7 * cos(tw * (4.1 + 3.3 * hb) + hb * 11.0))) * W * aB;
}
float2 PESGrassWindV5(float2 pesRoot, float2 wd, float2 wc, float hb, float tw, float W, out float gustOut) {
  if (IniParams.Load(int2(433, 0)).w > 0.5) return PESGrassWindV6(pesRoot, wd, wc, hb, tw, W, gustOut);
  // (08/11 y, Agustin: "el movimiento del viento no siempre es igual") x408 rachas y calmas en el tiempo: el viento de toda la cancha sube y
  // baja lento (calmas de varios segundos, rachas que llegan) y las rachas pegan mas fuerte; y408 la direccion oscila hasta +- y408 grados
  // (lento, y distinta en cada zona de la cancha). 0 / 0 = como antes. Todo depende solo del tiempo y la posicion (igual en los dos llamados)
  float4 P408 = IniParams.Load(int2(408, 0));
  float kG = clamp(P408.x, 0.0, 2.0);
  if (kG > 0.001) {
    float e1 = PESGnoise(float2(tw * 0.09, 3.7)) * 2.0 - 1.0, e2 = PESGnoise(float2(tw * 0.31 + dot(pesRoot, wd) * 0.015, 8.1)) * 2.0 - 1.0;
    W *= max(1.0 + kG * (0.55 * e1 + 0.30 * e2), 0.12);
  }
  float dA = radians(clamp(P408.y, 0.0, 60.0)) * (PESGnoise(float2(tw * 0.06, 1.3) + pesRoot * 0.012) * 2.0 - 1.0);
  if (abs(dA) > 1e-4) { float cA = cos(dA), sA = sin(dA); wd = float2(cA * wd.x - sA * wd.y, sA * wd.x + cA * wd.y); wc = float2(-wd.y, wd.x); }
  float2 q = pesRoot - wd * tw * 3.0;
  float g = PESGnoise(q * 0.08) * 0.6 + PESGnoise(float2(dot(q, wd) * 0.30, dot(q, wc) * 0.07) + 5.1) * 0.4;
  float gust = smoothstep(0.30 - 0.08 * saturate(kG), 0.85, g) * (1.0 + 0.35 * kG * saturate(g - 0.6));
  float fl = sin(tw * (7.0 + 5.0 * hb) + hb * 6.2832);
  float sw = sin(tw * (1.6 + 0.8 * hb) + hb * 3.1 + dot(pesRoot, wd) * 0.9);
  gustOut = gust * W;
  // (08/11, Agustin: "viento entre hebras") aleteo propio de cada hoja: frecuencia y fase por hoja (hb), mas fuerte con la racha, en
  // las dos direcciones; de lejos se promedia solo (el TAA) y de cerca cada hoja tiembla distinto a la de al lado
  float fl2 = sin(tw * (13.0 + 9.0 * hb) + hb * 21.0) * sin(tw * (3.1 + 1.7 * hb) + hb * 9.0);
  return wd * ((0.12 + 0.88 * gust) * 0.55 + sw * 0.07) * W + wc * fl * (0.05 + 0.10 * gust) * W + (wd * 0.4 + wc) * fl2 * (0.025 + 0.05 * gust) * W;
}
float2 PESGrassWindV3(float2 pesRoot, float2 wd, float tw, float W) {
  float phw = dot(pesRoot, wd) * 0.6 - tw * 2.2;
  float gust = sin(phw) * 0.6 + sin(phw * 2.7 + tw * 0.9) * 0.25 + sin(tw * 0.7 + pesRoot.x * 0.13) * 0.15;
  return wd * (0.25 + 0.75 * saturate(gust * 0.5 + 0.5)) * 0.45 * W;
}

// (13/11 tanda 2) pelota v2 (y446): movimiento de la pelota en s de JUEGO. bA / bB = pelota ahora / cuadro anterior, dtB = s de juego del
// cuadro. x = recorrido del cuadro (m), y = m/s de juego (una copia repetida con recorrido 0 da 0: sin saltos), z = rodando (0..1)
float3 PESBallKin(float3 bA, float3 bB, float dtB, out float2 dirB) {
  float2 vB = bA.xz - bB.xz; float dF = length(vB); dirB = dF > 1e-4 ? vB / dF : float2(0.0, 0.0);
  float vG = dF > 0.001 ? dF / clamp(dtB, 1.0 / 1000.0, 0.1) : 0.0;
  float kRoll = (dF < 1.5 ? 1.0 : 0.0) * smoothstep(0.30, 0.15, bA.y) * smoothstep(0.15, 0.6, vG);
  return float3(dF, vG, kRoll);
}
// capsula de la pelota entre este cuadro (bA) y el anterior (bB): punto mas cercano a p y alto del centro ahi (12/11, igual que antes)
void PESBallCap(float2 p, float3 bA, float3 bB, out float2 cC, out float yC) {
  float2 sA = bA.xz, sAB = bB.xz - bA.xz;
  float lAB = dot(sAB, sAB);
  float uC = (lAB > 1e-6 && lAB < 4.0 && bB.y < 0.35) ? saturate(dot(p - sA, sAB) / lAB) : 0.0;
  cC = sA + sAB * uC; yC = lerp(bA.y, bB.y, uC);
}
// (13/11 tanda 2) y446 CORONA: la hoja cuya raiz esta a dBr del centro se abre hasta que la punta (alto hRef) queda 1.5 cm afuera de la
// pelota (radio de la esfera a la altura de la punta); las raices mas cerca (debajo de la pelota) se hunden como antes. need = cuanto corre
// la punta, maxL = tope (2/3 del alto). Devuelve el peso de la corona (0..kV2)
float PESCrownK(float dBr, float yC, float hRef, float kV2, float w177, out float need, out float maxL) {
  const float RbS = 0.115;
  float dyT = yC - hRef, rSec = sqrt(max(RbS * RbS - dyT * dyT, 0.0));   // radio de la pelota a la altura de la punta
  float rTgt = rSec + 0.015, onG = smoothstep(0.17, 0.125, yC) * saturate(w177);
  maxL = 0.66 * hRef;
  float kClear = smoothstep(rTgt - maxL - 0.010, rTgt - maxL + 0.005, dBr);   // las raices mas cerca se hunden como antes
  need = max(rTgt - dBr, 0.0);
  return kV2 * onG * kClear * saturate(need / 0.004);
}
// (13/11 tanda 2) todo lo que la pelota corre la hoja (huella, nido o corona y fila 442), como funcion para poder calcularlo dos veces: con la
// pelota de este cuadro y con la del anterior (w446: el TAA recibe el movimiento de las hojas que empuja la pelota). Es la MISMA cuenta de
// antes (12/11) con y446 = 0, salvo hTopB (alto real de la punta con el tipo y104). p = vertice (xz) antes de la pelota, hW0 = su alto (v0.y), rootB = raiz de la hoja (igual en toda la hoja),
// hRefE = pesHref, kV2 = y446, sZ = zona x446 (1 sin y446), dtB = s de juego del cuadro. Devuelve el corrimiento; sinkF = hundimiento
float2 PESBallDeform(float2 p, float hW0, float2 rootB, float3 bA, float3 bB, float hRefE, bool blade, bool natV, bool isExt, float natCut,
                     float hv, float kV2, float sZ, float dtB, inout float sinkF, out float crK) {
  crK = 0.0;
  float2 q = p;
  float iS = 1.0 / sZ;
  float4 W177 = IniParams.Load(int2(177, 0));
  float2 dirK; float3 kin = PESBallKin(bA, bB, dtB, dirK);
  if ((isExt || natV) && W177.w > 0.001 && bA.y < 0.35) {
    float dB = length(q - bA.xz);
    if (natV) {
      // (11/11 h / l, 12/11) natural: huella de la esfera R 0.115 en la capsula de la pelota, nido y w439 pelota asentada (ver main)
      const float RbS = 0.115;
      float2 cC; float yC; PESBallCap(q, bA, bB, cC, yC);
      dB = length(q - cC);
      float ySB = dB < RbS ? yC - sqrt(max(RbS * RbS - dB * dB, 0.0)) : 1000.0;
      float hRef = max(hRefE * (blade ? natCut : 1.0), 0.005);
      float kB = saturate(1.0 - max(ySB, 0.0) / hRef) * saturate(W177.w);   // alto real de la hoja
      float kNest = smoothstep(0.17, 0.085, dB) * smoothstep(0.17, 0.125, yC) * saturate(W177.w);
      float kSeat = saturate(IniParams.Load(int2(439, 0)).w);
      kNest *= 1.0 - 0.75 * kSeat;
      float kLeanIn = kSeat * smoothstep(0.145, 0.095, dB) * (1.0 - kB) * smoothstep(0.17, 0.125, yC) * saturate(W177.w);
      float2 dO = dB > 1e-4 ? (q - cC) / dB : float2(1.0, 0.0);
      // (13/11 tanda 2, Agustin: con el original la pelota quieta queda rodeada de hojas paradas) y446 CORONA en vez del nido / asiento: el nido
      // empujaba y hundia en proporcion al alto, el disco del mapa la hundia otra vez y quedaba un pozo. Ahora la hoja se abre ENTERA desde la
      // raiz (viga empotrada: la raiz firme, la punta afuera de la pelota) y no se hunde. Medido en la RAIZ: toda la hoja igual. Rodando, sin
      // pliegue hacia atras (la estela la lleva hacia adelante)
      if (kV2 > 0.001) {
        float2 cCr; float yCr; PESBallCap(rootB, bA, bB, cCr, yCr);
        float dBr = length(rootB - cCr); float2 dOr = dBr > 1e-4 ? (rootB - cCr) / dBr : dO;
        dOr -= dirK * min(dot(dOr, dirK), 0.0) * kin.z;   // sin pliegue hacia atras
        float need, maxL; crK = PESCrownK(dBr, yCr, hRef, kV2, W177.w, need, maxL);
        float tB = saturate(max(hW0, 0.0) / hRef);
        q += dOr * (min(need, maxL) * crK * tB * tB * (3.0 - tB) * 0.5);   // viga: la punta despeja la pelota
        kB *= 1.0 - crK; kNest *= 1.0 - kV2; kLeanIn *= 1.0 - kV2;
        sinkF *= 1.0 - 0.7 * crK;   // el disco del mapa no hunde la corona
      }
      sinkF = max(sinkF, max(0.9 * kB, 0.45 * kNest));

        }
        float2 dsp = normalize(dU + dirB * 0.8) * (0.85 * kBow) + dirB * (0.80 * kWake * cWk + 0.55 * kAir) + latB * (sign(lt) * 0.25 * kAir);
        dsp = float2(dsp.x * csJ.x - dsp.y * csJ.y, dsp.x * csJ.y + dsp.y * csJ.x);   // desvio propio de la hoja
        q += dsp * (kBall * hW * gBb * stB * 1.1);
        sinkF = max(sinkF, saturate((0.30 * kBow + 0.60 * kWake) * kBall * wetB));
      }
      float kPq = clamp(P442.w, 0.0, 2.0) * smoothstep(0.010, 0.035, vY) * smoothstep(0.65, 0.20, bA.y);
      if (kPq > 0.001 && dL < 0.8 * sZ) {
        float rr = 0.12 + 0.9 * max(bA.y - 0.11, 0.0);
        float ring = exp(-((dLs - rr) * (dLs - rr)) / 0.005);
        q += dU * (ring * kPq * 1.2 * hW * gBb * stB);
        sinkF = max(sinkF, saturate(0.40 * ring * kPq));
      }
    }
  }
  return q - p;
}

  r0.x = dot(r5, float4(1.0, 0.016671, 0.001178, -0.000208));
  r0.x = saturate(r0.x + 0.5);
  r2.y = r10.y + r0.x;
  r5.x = -0.069282;
  r0.x = dot(r5, float4(1.0, 0.048773, -0.004529, 0.000109));
  r0.x = saturate(r0.x + 0.5);
  r2.x = r10.x + r0.x;
  r0.x = r2.x - 0.5;
  r0.z = r2.y - 0.5;
  r0.x = r1.x * r0.x;
  o5.y = r0.z;
  o5.x = r0.x * r1.z;
  o5.z = r0.y * r1.y;
  o5.w = (pesIsExt && cb5[9].x > 0.5) ? 1.0 : -pesPaint;   // PES21: depuracion, el PS pinta magenta (1); (27/10 h) negativo = hoja pintada por la linea
  // (2/11) w139: posicion de este vertice en el CUADRO ANTERIOR con el viento de entonces (misma hoja, misma curva): el TAA recibe el
  // movimiento real de la hoja y deja de promediarla con la camara quieta (nitidez). Sin viento o con w139 = 0, la misma posicion de antes
  // (13/11 tanda 2) w446: + lo que la pelota corria esta parte de la hoja en el cuadro anterior (pesBallD = antes - ahora; 0 sin w446)
  float4 r6p = r6;
  if (abs(pesWindD.x) + abs(pesWindD.y) > 1e-7 || abs(pesBallD.x) + abs(pesBallD.y) > 1e-9) {
    float2 dxz = pesWindD * (pesV5 ? pesHref * pesTB * pesTB : v0.y * saturate(v0.y * 25.0)) + pesBallD;
    r6p.xz += dxz;
    if (pesV5) { float2 bp = pesBend + dxz;
      if (pesLimB >= 0.0) {   // (13/11) el mismo tope de costado en el cuadro anterior: sin esto el TAA veia una hoja que se estiraba de mas
        float2 bpU = pesBendU + dxz; float blP = length(bpU);
        float2 bpC = (blP > pesLimB && blP > 1e-6) ? bpU * (pesLimB / blP) : bpU;
        r6p.xz += bpC - bp; bp = bpC; }
      float bb2 = min(dot(bp, bp), pesYY * pesYY * 0.9375); r6p.y += sqrt(max(pesYY * pesYY - bb2, 0.0)) - pesBentY; }
  }
  o6.z = dot(r6p, cb2[16]);
  o6.w = dot(r6p, cb2[17]);
  o7.y = dot(r6p, cb2[19]);
  // (04/11) x291 / y330: el color del suelo se toma en la RAIZ de la hoja, no en su pixel. En el pixel de la hoja, una hoja delante de un
  // botin o de la pelota copiaba el color del botin (la copia de RT0 ya tiene al jugador): el pasto se veia transparente y el botin entero
  {
    float4 rw = float4(pesRoot.x, 0.0, pesRoot.y, 1.0);
    float4 rv = float4(dot(rw, cb2[8]), dot(rw, cb2[9]), dot(rw, cb2[10]), dot(rw, cb2[11]));
    float rcw = dot(rv, cb2[7]);
    o7.zw = rcw > 0.01 ? float2(dot(rv, cb2[4]), dot(rv, cb2[5])) / rcw : float2(9.0, 9.0);   // 9 = fuera de pantalla: el PS usa su pixel
  }
  // (27/10 m) y = oclusion de la mata (tanda 10, 0.5..1; antes el pelado, que el PS solo usaba en la depuracion w107 = 2)
  // (11/11 j, Agustin: "falta tambien la sombra en el pasto de la pelota") natural: la pelota no esta en el mapa de sombras del sol (su sombra
  // es la sombra plana del juego, g_tex_planarShadow) y a las hojas dibujadas con el stencil de la base se les apaga la sombra plana (para no
  // oscurecer dos veces a los jugadores): sobre el pasto no quedaba la sombra de la pelota. Se calcula aca: el rayo de esta parte de la hoja
  // hacia el sol pasa por la pelota (esfera de 11.2 cm, borde suave que se abre con la distancia) -> sombra; mas la oscuridad de contacto
  // debajo y alrededor cuando apoya. Solo donde la sombra plana esta apagada; mas fuerte con sol fuerte (w414)
  // (12/11, Agustin: "si podes hacer que la pelota tambien reaccione mejor al pasto en cuanto a la sombra") con w439 tambien con la sombra
  // plana prendida (x290 = 0, "Como el pasto del juego"): antes este bloque no corria y sobre las hojas quedaba solo la sombra plana del
  // juego, leida en la raiz (borrosa, sin la oscuridad de contacto). Se toma la MAS OSCURA de las dos (no se oscurece dos veces)
  bool pesBallSh = pesPlanOff || (pesIsExt && IniParams.Load(int2(439, 0)).w > 0.001);
  if (pesNatV && pesBallSh && pesBLive && pesBA.y > 0.06 && pesBA.y < 40.0) {   // (13/11 tanda 2) pesBA: la pelota de este cuadro (o extrapolada); vieja: nada
    const float Rb = 0.112;
    float3 dB = pesBA - r6.xyz;
    float3 Ls = PESSunSky();
    float shB = 0.0;
    float tB = dot(dB, Ls);
    if (tB > 0.0 && Ls.y > 0.05) {
      float dd = sqrt(max(dot(dB, dB) - tB * tB, 0.0));
      shB = 1.0 - smoothstep(Rb * 0.70 - 0.004 * tB, Rb * 1.10 + 0.02 + 0.03 * tB, dd);
    }
    float sunB = saturate(IniParams.Load(int2(414, 0)).w);
    float dl = length(dB);
    float aoB = (1.0 - smoothstep(Rb * 1.05, Rb * (2.4 + 1.35 * (pesSZ - 1.0)), dl)) * saturate(0.25 + 0.75 * saturate(dB.y / max(dl, 1e-3)));   // (13/11 tanda 2) x446: borde 1.05 + 1.35 S
    // (11/11 m, capturas 19.22 de Agustin: hojas lima dentro de la sombra negra de la pelota) mas fuerte, como la sombra plana del suelo:
    // 0.62 x (0.35..1 segun el sol) -> 0.88 x (0.6..1); con el sol tapado o de noche igual queda la sombra de contacto
    float kB = saturate(1.0 - 0.88 * shB * lerp(0.6, 1.0, sunB) - 0.40 * aoB);
    if (pesPlanOff) o3.xyz *= kB;
    else o1.z = min(o1.z, kB);   // (12/11) con la sombra plana: la mas oscura de las dos (o1.z multiplica el color de la hoja en el PS)
  }
  o8 = float4(pesWear.x * (pesWShort ? 0.6 : 1.0), pesIsExt ? pesClumpAO : 1.0, (min(floor(pesYaw * 32.0), 31.0) + frac(pesHv * 37.1 + 0.13) * 0.999) / 32.0, pesV3 ? 1.0 + pesBand + ((pesFl & 2) != 0 ? 8.0 : 0.0) + (pesBlade ? (pesNarV ? 48.0 : 16.0) : 0.0) : 0.0);   // z (27/10 h) valor al azar por hoja (antes pesH, el PS no lo leia); (04/11) z = (sector de 32 del giro + azar) / 32; w = 1 + banda del atlas (0 = sin v3) + 8 si se mezcla con el suelo (igual en todo el dibujo) + 16 hojas 3D (11/11)
  if (pesKill) o0 = float4(0.0, 0.0, -2.0, 1.0);   // todos los vertices iguales: triangulos nulos, no se rasteriza
}
