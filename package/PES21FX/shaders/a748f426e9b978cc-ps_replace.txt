// === PES21 SOL: color del sol segun su altura (calido al atardecer, blanco calido al mediodia) ===
float3 PESSunColor() {
  float3 c = g_psLight.m_lightColor0.xyz;
  float4 S = IniParams.Load(int2(4,0));      // z4 = realismo del sol (0 = original), w4 = intensidad
  bool sunReal = S.z > 0;
  float3 L = normalize(g_psLight.m_lightDir0.xyz);
  float elev = saturate(abs(dot(L, float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12))));
  float low = 1 - smoothstep(0.10, 0.60, elev);
  float3 tint = lerp(float3(1.0, 0.962, 0.905), float3(1.0, 0.80, 0.60), low);
  const float3 LW = float3(0.2126, 0.7152, 0.0722);
  float lum = dot(c, LW);
  float warmO = c.r / max(c.b, 1e-4);
  float k = (sunReal ? saturate(S.z) : 0.0) * saturate((warmO - 0.85) * 4.0);   // de noche (luz fria de estadio) casi no toca
  float3 res = lerp(c, lum * tint / dot(tint, LW), k);
  res *= (S.w > 0 ? lerp(1.0, S.w, k) : 1.0);
  // PES21 curva del dia (18/10): intensidad y calidez del sol segun su altura (solo sol real, no torres)
  { PESDay dy = PESDayCurve(elev, saturate((warmO - 0.85) * 4.0)); res *= dy.sun; res = PESWarmTint(res, dy.warm); }
  // PES21 atmosfera (17/10): con el sol bajo el archivo de luz deja el sol muy apagado; y54 lo compensa (solo sol real,
  // no luz de torres), proporcional a lo bajo que esta.
  float4 ATM = IniParams.Load(int2(54,0));
  if (ATM.w > 0.5) { float lowS = 1.0 - smoothstep(0.05, 0.40, elev); float kSun = saturate((warmO - 0.85) * 4.0);
    res *= 1.0 + max(ATM.y, 0.0) * lowS * kSun; }
  return res;
}

static float g_pesDynMinR = 0.0;   // PES21 sol dinamico: radio minimo del filtro (lo fija main)
float PESShadowFilter(float2 uv, float ref, float cas, float2 pix, float orig, float rscale, float2 dzduv)
{
  float4 SH = IniParams.Load(int2(20,0));
  float4 SQ = IniParams.Load(int2(21,0));
  float res = g_psScene.m_shadowMapResolutions.x;
  if (SH.y < 0.5 || cas < -0.5 || res < 1.0) return orig;
  float2 tx = float2(1.0 / res, 1.0 / (3.0 * res));           // atlas de 3 cascadas apiladas en vertical
  float2 lo = float2(tx.x, cas / 3.0 + tx.y);                   // no salir de la cascada actual
  float2 hi = float2(1.0 - tx.x, (cas + 1.0) / 3.0 - tx.y);
  float2 p = pix;
  if (SQ.w > 0.5) p += frac(g_psSystem.m_timeParam.z * 61.8034) * 97.0;
  float rot = 6.2831853 * frac(52.9829189 * frac(dot(p, float2(0.06711056, 0.00583715))));
  float radius = max(max(SH.z, g_pesDynMinR), 0.5) * rscale;
  // PES21 sombras fisicas (18/10): x87 sesgo por plano receptor | z87 penumbra fisica por tamano del sol | w87 tope del ajuste (x0.001 por texel)
  float4 PH = IniParams.Load(int2(87,0));
  float limT = (PH.w > 0.0 ? PH.w : 10.0) * 0.001;
  bool rpb = PH.x > 0.5;
  float casS = cas < 0.5 ? 1.0 : (cas < 1.5 ? g_psScene.m_shadowProjection2._m03 : g_psScene.m_shadowProjection2._m13);
  if (SH.y > 1.5) {
    float searchR = max(SQ.z, 1.0);
    float bsum = 0, bcnt = 0;
    [unroll] for (int j = 0; j < 8; j++) {
      float rj = sqrt((j + 0.5) / 8.0);
      float aj = j * 2.39996323 + rot;
      float2 offj = float2(cos(aj), sin(aj)) * (rj * searchR) * tx;
      float2 suj = clamp(uv + offj, lo, hi);
      float refj = ref + (rpb ? clamp(dot(offj, dzduv), -limT * rj * searchR, limT * rj * searchR) : 0.0);
      float d = g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, suj, 0).x;
      if (d > refj) { bsum += d - refj; bcnt += 1; }
    }
    if (bcnt < 0.5) return 1.0;
    float avgB = bsum / bcnt;
    float4 P64s = IniParams.Load(int2(64,0));
    if (PH.z > 0.5 && P64s.x > 0.5 && P64s.y > 0.0) {
      // penumbra fisica: ancho = distancia receptor-oclusor x tan(diametro angular del sol), pasado a texeles de esta cascada
      float refPerM = max(length(g_psScene.m_shadowProjection._m02_m12_m22) * abs(g_psScene.m_shadowProjection2._m30) * abs(casS), 1e-6);
      float texPerM = length(g_psScene.m_shadowProjection._m00_m10_m20) * abs(casS) * res * 0.5;
      float distM = avgB / refPerM;
      float penTex = distM * tan(radians(P64s.y)) * texPerM * (SQ.x > 0 ? SQ.x / 200.0 : 1.0);   // x21 = 200 -> 1:1 fisico
      radius = clamp(penTex, max(0.5, g_pesDynMinR), max(SQ.y, 1.0));
    } else radius = clamp(avgB * SQ.x, max(0.5, g_pesDynMinR), max(SQ.y, 1.0));
  }
  float sum = 0;
  [unroll] for (int i = 0; i < 16; i++) {
    float ri = sqrt((i + 0.5) / 16.0);
    float ai = i * 2.39996323 + rot;
    float2 offi = float2(cos(ai), sin(ai)) * (ri * radius) * tx;
    float2 sui = clamp(uv + offi, lo, hi);
    // plano receptor: cada muestra se compara con la profundidad que el plano de la superficie tiene en ESE texel
    float refi = ref + (rpb ? clamp(dot(offi, dzduv), -limT * ri * radius, limT * ri * radius) : 0.0);
    sum += g_ShadowTexture.SampleCmpLevelZero(g_samplerComparisonLess_Linear_Clmap_s, sui, refi).x;
  }
  return 1.0 - sum / 16.0;
}

// PES21 (18/10): terminador suave SIN correr la luz. Igual a max(NdotL,0) cuando NdotL >= w (lado iluminado intacto,
// misma direccion que el sol); entre -w y w una curva suave (C1) que solo entra un poco en la sombra.
float3 PESSoftTerm(float nl, float3 w) {
  w = max(w, 1e-3);
  float3 x = nl.xxx;
  float3 s = (x + w) * (x + w) / (4.0 * w);
  return x >= w ? x : (x > -w ? s : 0.0);
}

  { PESCloudParams CPs = PESCloudLoad(g_psSystem.m_timeParam.z); float cK = saturate(IniParams.Load(int2(85,0)).z);

      float dC0 = g_DepthTexture.SampleLevel(g_samplerPoint_Wrap_s, uvG, 0).x;

      float cT = PESCloudShadow(pWc, sunWc, CPs);
      r0.x *= lerp(1.0, cT, cK);
    } }
  r0.y = r0.x * r0.x;
  r9.xy = float2(-0,-0);
  r9.zw = -g_psLight.m_lightSourceParam3.xy;
  r6.xyw = r9.xyz + r6.xyz;
  r0.z = dot(r6.xyw, r6.xyw);
  r0.z = sqrt(r0.z);
  r0.z = r0.z + r9.w;
  r0.z = max(0, r0.z);
  r0.z = saturate(g_psLight.m_lightSourceParam1.x * r0.z);
  r0.w = cmp(r6.w >= 0);
  r0.w = r0.w ? 1.000000 : 0;
  r0.z = r0.z * r0.w;
  r0.x = -r0.x * r0.x + 1;
  r0.x = r0.z * r0.x + r0.y;
  r0.x = saturate(-g_psLight.m_lightSourceParam1.w + r0.x);
  r0.y = cmp(r11.x == -1.000000);
  r0.x = r0.y ? 1 : r0.x;
  r0.y = -g_psObject.m_localParam[2].x + r6.z;
  r0.y = saturate(g_psObject.m_localParam[2].y * r0.y);
  r0.y = 1 + -r0.y;
  r0.y = r0.y * r0.y;
  r0.xyzw = r0.xxxx * r0.yyyy;
  r1.x = max(g_psLight.m_lightSourceParam2.x, r0.w);
  if (r1.z != 0) {
    r6.y = r2.x * 0.959999979 + 0.0399999991;

    r5.xy = float2(0,0);
  } else {

    }

  r3.w = -r3.z;
  r1.y = dot(pesLDir.xyz, r3.xyw);
  r1.z = dot(-r7.xyz, r3.xyw);
  r1.w = r1.z + r1.z;
  r4.xyz = r3.xyw * -r1.www + -r7.xyz;

  r2.x = r2.x * r5.y + r1.y;

  r3.xyz = r2.xxx * r8.xyz;

  float pesCurv = 0.0;
  { float3 nC = normalize(r3.xyw);
    float dzC = g_DepthTexture.SampleLevel(g_samplerPoint_Wrap_s, uvG, 0).x;
    float zLC = abs(g_psScene.m_projectionParam.z / (dzC - g_psScene.m_projectionParam.w));
    float mpp = max(zLC * g_psScene.m_projectionParam.x * 2.0 / max(g_psSystem.m_renderBuffer.x, 1.0), 1e-6);

  // apagado, la formula del motor envolvia la cara con la translucidez (que el mapa de SSS sube) y la luz no seguia al sol.
  if (isSkinGeo) {

  {

Con 4 cm NUNCA habia impacto: una cabeza o un brazo miden 10-25 cm y el rayo pasa por detras de su cara visible (verificado con una simulacion numerica)               // grosor aceptado: crece con el largo del rayo

      float3 HK = normalize(LK + VK);
      float nhK = saturate(dot(nK, HK));
      float aK = clamp(KC.z, 0.2, 1.0);
      float invA = 1.0 / aK;
      float sin2 = max(1.0 - nhK * nhK, 1e-4);
      float Dch = (2.0 + invA) * pow(sin2, 0.5 * invA) / 6.2831853;
      float nlS = saturate(nlK);
      float Vch = 1.0 / (4.0 * max(nlS + nvK - nlS * nvK, 0.1));
      float sheenK = Dch * Vch * nlS * saturate(nlS * 4.0) * saturate(KC.x) * shadowScalar;
      sheenK = min(sheenK, 0.6);
      // 70 % va al canal difuso (toma el color del kit en la composicion), 30 % al especular (blanco)
      o0.xyz += sheenK * 0.7 * lightK;
      r1.yzw += sheenK * 0.3 * lightK;
      // 3) la tela no tiene el brillo plastico del GGX: lo baja a la mitad
      r1.yzw *= 0.5;
      // === PES21 kits v4 (18/10): brillo anisotropico de la trama + translucidez de la tela a contraluz ===
      // x71 brillo anisotropico (banda a lo largo del tejido, vertical en la camiseta), y71 translucidez,
      // z71 cuanto se estira la banda, w71 on/off
      { float4 KA = IniParams.Load(int2(71,0));
        if (KA.w > 0.5) {
          float3 upK = normalize(float3(g_psScene.m_view._m10, g_psScene.m_view._m11, g_psScene.m_view._m12));
          float3 TK = upK - nK * dot(upK, nK);
          float tl = length(TK);
          if (tl > 1e-3) {
            TK /= tl;
            // Kajiya-Kay: el brillo se estira perpendicular a la trama (a lo largo de la fibra horizontal del punto)
            float3 HK2 = normalize(LK + VK);
            float th = dot(TK, HK2);
            float sinTH = sqrt(saturate(1.0 - th * th));
            float expo = lerp(14.0, 60.0, saturate(KA.z));
            float an = pow(sinTH, expo) * saturate(nlK) * saturate(nlK * 4.0);
            float3 anC = an * KA.x * 0.18 * lightK * shadowScalar;
            o0.xyz += anC * 0.6; r1.yzw += anC * 0.4;
          }
          // translucidez: con el sol detras de la camiseta (y de frente a la camara), la tela se "enciende"
          float backK = saturate(dot(-nK, LK));
          float fwdK = pow(saturate(dot(VK, -LK)), 2.0);
          float thinK = 0.5;   // tela fina de futbol
          o0.xyz += lightK * backK * (0.5 + 0.5 * fwdK) * thinK * KA.y * 0.35;
        } }
      // diagnostico: w26 = 2 pinta de magenta todo lo que se detecta como kit
      if (KC.w > 1.5) { o0.xyz = float3(1.0, 0.0, 1.0) * dot(lightK, float3(0.3, 0.3, 0.3)); r1.yzw = 0; }
    }
  }
  if (isSkin) {
    // --- reconstruccion de normal, vista y luz ---
    float4 nb = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q = (uint3)(float3(1023,1023,3) * nb.xyw);
    uint ex = (q.x << 1) | (q.z >> 1);
    uint ey = (q.y << 1) | (q.z & 1);
    float2 e = float2(ex, ey) * 0.00097703957 - 1;
    float3 n = float3(e.xy, 1 - abs(e.x) - abs(e.y));
    if (n.z < 0) n.xy = (1 - abs(e.yx)) * (e.xy >= 0 ? 1 : -1);
    n = normalize(n); n.z = -n.z;
    float2 pp = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V = -normalize(float3(pp, 1));
    float3 L = normalize(pesLDir.xyz);
    float3 lightC = PESSunColor() * g_psScene.m_exposure.zzz;
    // --- transmision a contraluz ---
    // 16/10: transmision fisica (Beer-Lambert): la luz que atraviesa el tejido pierde primero el azul y el verde,
    // el rojo llega mas lejos. y34 = densidad del tejido (1 = piel real), P5.x = intensidad general.
    float3 Lt = normalize(L + n * 0.3);
    float fwd = pow(saturate(dot(V, -Lt)), 4.0);
    float back = saturate(dot(-n, L));
    float thin = saturate(trans * 1.6);
    float3 sigT = float3(1.0, 3.6, 6.5) * (P34.y > 0 ? P34.y : 1.0);
    float3 Tr = exp(-(1.15 - thin) * sigT) * thin;
    float visT = lerp(0.25, 1.0, saturate(r0.x));   // la luz entra por el lado de atras aunque el frente este en sombra
    // 18/10 v2: solo donde la piel es fina de verdad (orejas, aletas, labios) y el sol esta detras; sin el
    // termino 'de atras' amplio que encendia de naranja todo el lado en sombra de la cara
    float thinGate = lerp(smoothstep(0.45, 0.85, thin), 1.0, saturate(P76.y));   // y76: 0 = solo orejas/nariz/labios, 1 = toda la piel (como antes)
    float3 transm = Tr * fwd * (0.6 + 0.4 * back) * thinGate * P5.x * 1.6 * sssV2;
    // PES21 transmision por shadow map (18/10, Jimenez 2010): el grosor real que atraviesa la luz = distancia entre lo
    // que ve el sol (shadow map) y este pixel. Orejas y nariz se encienden solo cuando el sol esta realmente detras.
    if (P64.x > 0.5 && P64.z > 0.0 && pesShCas > -0.5 && g_psScene.m_shadowMapResolutions.x > 1.0) {
      tookSM = true;
      float casS = pesShCas < 0.5 ? 1.0 : (pesShCas < 1.5 ? g_psScene.m_shadowProjection2._m03 : g_psScene.m_shadowProjection2._m13);
      float refPerM = max(length(g_psScene.m_shadowProjection._m02_m12_m22) * abs(g_psScene.m_shadowProjection2._m30) * abs(casS), 1e-5);
      float resS = g_psScene.m_shadowMapResolutions.x;
      float2 txS = float2(1.0 / resS, 1.0 / (3.0 * resS));
      float dS = g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, pesShUV, 0).x;
      dS += g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, pesShUV + float2( txS.x, 0), 0).x;
      dS += g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, pesShUV + float2(-txS.x, 0), 0).x;
      dS += g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, pesShUV + float2(0,  txS.y), 0).x;
      dS += g_ShadowTexture.SampleLevel(g_samplerPoint_Clamp_s, pesShUV + float2(0, -txS.y), 0).x;
      dS *= 0.2;
      float thickM = max(abs(dS - pesShRef) - 0.5 / refPerM * 0.004, 0.0) / refPerM;   // 18/10: distancia sin asumir el sentido del eje Z del shadow map
      float dens = (P64.w > 0 ? P64.w : 30.0) * (1.25 - thin);   // el mapa de SSS afina: orejas/nariz dejan pasar mas luz
      float3 Tt = PESSkinTransmittance(thickM * dens);
      float backL = saturate(0.3 + dot(L, -n)) * (1.0 - saturate(rawNdotL * 3.0));
      float3 transSM = Tt * backL * P64.z * 0.8;
      o0.xyz += max(transSM, transm * visT * 0.5) * lightC;   // sin perder el contraluz direccional cuando el shadow map no alcanza
      // diagnostico w12 = 3: rojo = grosor medido (10 cm = 1), verde = de espaldas al sol, azul = en sombra segun el shadow map
      if (dbgT) o0.xyz = float3(saturate(thickM / 0.10), saturate(dot(L, -n)), r0.x < 0.5 ? 0.6 : 0.0) * 4.0;
    } else if (P34.w > 0.5) o0.xyz += transm * lightC * visT;
    else {
      float3 transmO = thin * smoothstep(0.45, 0.85, thin) * pow(saturate(dot(V, -Lt)), 3.0) * float3(1.0, 0.24, 0.10) * P5.x;
      o0.xyz += transmO * lightC * sqrt(saturate(r0.x));
    }
  }
  r1.xyz = r1.yzw * r1.xxx;
  if (isSkin) {
    r1.xyz *= skinSpec;
    float4 nb2 = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uvG);
    uint3 q2 = (uint3)(float3(1023,1023,3) * nb2.xyw);
    float2 e2 = float2((q2.x << 1) | (q2.z >> 1), (q2.y << 1) | (q2.z & 1)) * 0.00097703957 - 1;
    float3 n2 = float3(e2.xy, 1 - abs(e2.x) - abs(e2.y));
    if (n2.z < 0) n2.xy = (1 - abs(e2.yx)) * (e2.xy >= 0 ? 1 : -1);
    n2 = normalize(n2); n2.z = -n2.z;
    float2 pp2 = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
    float3 V2 = -normalize(float3(pp2, 1));
    float3 L2 = normalize(pesLDir.xyz);
    float3 H2 = normalize(L2 + V2);
    float nl2 = saturate(dot(n2, L2));
    // lobulo fino (pelicula de grasa/sudor): GGX con rugosidad 0.18
    float a2 = 0.18 * 0.18; float eF = 1.0;
    if (P64.x > 0.5 && P64.y > 0.0) a2 = PESSunWidenAlpha(a2, P64.y, eF);   // PES21 sol de calidad
    float a4 = a2 * a2;
    float nh = saturate(dot(n2, H2));
    float dd = nh * nh * (a4 - 1) + 1;
    float D2 = a4 / (3.14159 * dd * dd);
    float F2 = 0.028 + 0.972 * pow(1 - saturate(dot(V2, H2)), 5);
    float sharp = D2 * F2 * nl2 * 0.25 * P5.y * eF;
    r1.xyz += sharp * PESSunColor() * g_psScene.m_exposure.zzz * shadowScalar * saturate(specInt * 2 + 0.3);
  }
  // diagnostico w12 = 3 sobre piel: magenta = es piel pero no se midio el grosor (sin cascada / transmision apagada);
  // cian = marcada como piel pero el SSS esta apagado o sin translucidez (isSkin falso)
  if (dbgT && abs(skinTag - 0.98039) < 0.003 && !tookSM) { o0.xyz = isSkin ? float3(4.0, 0.0, 4.0) : float3(0.0, 4.0, 4.0); r1.xyz = 0; }
  // diagnostico w12 = 4: la escena en gris y las sombras de contacto en rojo (mas rojo = mas oscura)
  if (dbgC) { o0.xyz = lerp(o0.xyz, float3(10.0, 0.0, 0.0), saturate((1.0 - csDbg) * 3.0)); r1.xyz = 0; }   // 18/10 d: rojo donde hay sombra de contacto (el gris no se veia: esta salida es irradiancia, el color lo pone el albedo despues)
  o1.xyz = min(float3(10000,10000,10000), r1.xyz);
  o0.w = 1;
  o1.w = 1;
  // PES21 inspector (19/10): LUZ PLANA (x89 = 1): irradiancia constante (y89) sin direccion, sombras ni brillos, para ver
  // texturas y materiales sin la iluminacion. z89 = 1 deja el brillo especular.
  { float4 P89 = IniParams.Load(int2(89,0));
    if (P89.x > 0.5) { float fl = P89.y > 0.0 ? P89.y : 1.0; float3 c89 = pesLCol.xyz; float m89 = max(max(c89.x, c89.y), c89.z); if (m89 < 1e-3) m89 = 1.0;
      o0.xyz = (c89 / m89) * fl * 0.6; if (P89.z < 0.5) o1.xyz = 0; } }
  o2.xyzw = r0.xyzw;
  return;
}
