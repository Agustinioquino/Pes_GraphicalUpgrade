// === PES21 pelo con hebras (Claude, 16/10) ===
// x33 sesgo de mip (negativo = hebras mas finas y nitidas), y33 variacion entre hebras (brillo + inclinacion),
// z33 hebras por texel de la textura, w33 on/off [Alt+Shift+F12]
float PESHash21(float2 p) { p = frac(p * float2(123.34, 456.21)); p += dot(p, p + 45.32); return frac(p.x * p.y); }
// x = factor de brillo de la hebra, y = inclinacion de la normal, z = fade por distancia, w = 1 si las hebras corren a lo largo de v
float4 PESStrand(Texture2D<float4> tex, float2 uv, float4 HP)
{
  uint tw, th; tex.GetDimensions(tw, th);
  float2 ts = float2(max(tw, 1u), max(th, 1u));
  float2 o = 2.0 / ts;
  float lu = abs(tex.SampleLevel(g_samplerLinear_Clamp_s, uv + float2(o.x, 0), 2.0).w - tex.SampleLevel(g_samplerLinear_Clamp_s, uv - float2(o.x, 0), 2.0).w);
  float lv = abs(tex.SampleLevel(g_samplerLinear_Clamp_s, uv + float2(0, o.y), 2.0).w - tex.SampleLevel(g_samplerLinear_Clamp_s, uv - float2(0, o.y), 2.0).w);
  bool acrossU = lu >= lv;                       // el alfa cambia a lo ancho de u -> las hebras corren a lo largo de v
  float s = acrossU ? uv.x * ts.x : uv.y * ts.y; // coordenada a traves de las hebras (en texeles)
  float a = acrossU ? uv.y * ts.y : uv.x * ts.x; // coordenada a lo largo
  float k = s * max(HP.z, 0.1);
  float id = floor(k);
  float r1 = PESHash21(float2(id, 3.1));
  float r2 = PESHash21(float2(id, 7.7 + floor(a * 0.015)));
  float fade = saturate(1.5 - fwidth(k) * 1.5);  // si una hebra ocupa menos de un pixel, se apaga (sin titileo)
  float v = saturate(HP.y);
  return float4(1.0 + (r1 - 0.5) * 2.0 * v * fade, (r2 - 0.5) * 1.5 * v * fade, fade, acrossU ? 1.0 : 0.0);
}
// === PES21 brillo de pelo anisotropico (17/10) ===
// En el pelo real el brillo es una banda perpendicular a las hebras (Kajiya-Kay / Marschner), no un punto.
// El motor ilumina en diferido con GGX sobre la normal: si en el G-buffer se escribe la normal que, para esta hebra,
// refleja la luz hacia la camara (N = H sin su parte a lo largo de la hebra), GGX da exactamente la banda de Kajiya-Kay.
// x50 fuerza de la banda, y50 desplazamiento (raiz/punta), z50 espacio de la luz (0 mundo, 1 vista), w50 on/off
// x51 segundo brillo (TRT: color del pelo, hacia la punta), y51 dispersion entre hebras
float3 PESLightV(float4 HK)
{
  float3 L = g_psSystem.m_dominantLightDir.xyz;
  if (HK.z < 0.5) L = float3(dot(L, g_psScene.m_view._m00_m10_m20), dot(L, g_psScene.m_view._m01_m11_m21), dot(L, g_psScene.m_view._m02_m12_m22));
  return normalize(L + float3(1e-5, 2e-5, 1e-5));
}
void PESHairAniso(inout float3 N, inout float3 alb, float3 T, float3 V, float jitter, float fade)
{
  float4 HK = IniParams.Load(int2(50,0));
  float4 HK2 = IniParams.Load(int2(51,0));
  if (HK.w < 0.5) return;
  float3 L = PESLightV(HK);
  float3 H = normalize(L + V);
  T = normalize(T);
  float sh1 = HK.y + jitter * HK2.y;
  float3 T1 = normalize(T + N * sh1);
  float3 Nk = H - T1 * dot(H, T1);
  float nkl = length(Nk);
  Nk = nkl > 1e-4 ? Nk / nkl : N;
  if (dot(Nk, V) < 0.05) Nk = N;
  N = normalize(lerp(N, Nk, saturate(HK.x) * fade));
  float3 T2 = normalize(T - N * (sh1 * 1.5 + 0.1));
  float c2 = dot(T2, H);
  float s2 = sqrt(saturate(1.0 - c2 * c2));
  float lit = saturate(dot(N, L) * 0.5 + 0.5);
  alb += alb * pow(s2, 80.0) * max(HK2.x, 0.0) * lit * fade;
}
