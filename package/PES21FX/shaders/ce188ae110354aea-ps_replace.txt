//
// Este shader NO ilumina: escribe el G-buffer que despues usa la pasada de luz.
//   o0.rgb = albedo (+ "velo" de borde por fresnel)
//   o1     = normal octaedrica (2x11 bits) + translucidez (6 bits, la usa el SSS del motor)
//   o2.x   = rugosidad codificada, o2.y = intensidad especular, o2.w = tipo de material
//   o3.xy  = vectores de movimiento
//
// Parametros en vivo (d3dx.ini, seccion [Constants]):
//   x  = multiplicador de translucidez / SSS   (1.0 = original)
//   y  = saturacion del albedo de la piel        (1.0 = original)
//   z  = multiplicador de especular              (1.0 = original)
//   w  = 1 activa las mejoras, 0 = shader original
//   x2 = calidez de las zonas finas (orejas, nariz, labios) usando el mapa de translucidez
//   y2 = multiplicador de rugosidad -> <1 piel mas brillosa, >1 mas mate
//   x8..w8 = gotas de sudor: cantidad, tamano, velocidad, on/off
//   x9/y9 = minuto de inicio / maximo del sudor, z9 = realismo del vello facial
//   x32/y32 = sesgo de mip del color / normal de la cara (negativo = mas nitido), w32 on/off  (16/10)
//
// ARRUGAS DE EXPRESION (independiente de w; w13 = 1 las activa, 0 = arrugas originales)
//   x7  = ganancia global de las arrugas animadas      (1.0 = original)
//   x13 = curva de respuesta: >0 las expresiones suaves ya marcan arrugas (0 = lineal)
//   y13 = sombreado de pliegues (oscurece y quita brillo dentro del surco)  (0 = nada)
//   z13 = sonrisa -> patas de gallo: refuerza la zona de ojos cuando boca y ojos
//         se activan juntos (sonrisa real / Duchenne)                      (0 = nada)
//   x14/y14/z14/w14 = ganancia por zona: frente+sienes / entrecejo / ojos / boca
//   w15 = signo del eje Y del mapa de arrugas para detectar surcos (-1 = correcto para PES)
//   x15 = 1 usa las texturas remasterizadas (t20/t21), 0 = las del juego
//   y15 = zona muerta: por debajo de este peso no aparecen arrugas (cara en reposo lisa)
//
// FISICA DE LA EXPRESION (no sube las arrugas: cambia como reacciona la piel)  w16 = on/off
//   x16 = sangre: pliegue mas oscuro/rojizo, piel estirada mas palida   (0 = nada)
//   y16 = brillo: piel estirada mas lisa y brillante, pliegue mas mate   (0 = nada)
//   z16 = translucidez: el pliegue deja pasar menos luz, la piel tensa mas (0 = nada)

// ===================== gotas de sudor procedurales =====================
float h21(float2 p) { p = frac(p * float2(123.34, 456.21)); p += dot(p, p + 45.32); return frac(p.x * p.y); }

// Devuelve: xy = gradiente para la normal, z = mascara mojada, w = altura
float vn21(float2 p) { float2 i = floor(p), f = frac(p); f = f * f * (3 - 2 * f);
  return lerp(lerp(h21(i), h21(i + float2(1,0)), f.x), lerp(h21(i + float2(0,1)), h21(i + 1), f.x), f.y); }
float2 PoreGrad(float2 uv, float F) {
  float e = 0.35;
  float2 q = uv * F;
  return float2(vn21(q + float2(e,0)) - vn21(q - float2(e,0)), vn21(q + float2(0,e)) - vn21(q - float2(0,e)));
}
float4 SweatLayer(float2 uv, float cells, float amount, float size, float t, bool moving, float fp)
{
  float2 g = uv * float2(cells, cells);
  float2 c = floor(g);
  float2 f = frac(g);
  float r0 = h21(c);
  if (r0 > amount) return 0;
  float r1 = h21(c + 17.1);
  float r2 = h21(c + 41.7);
  float r3 = h21(c + 73.3);
  float2 pos = float2(0.3 + 0.4 * r1, 0.5);
  float slide = 0;
  if (moving) {
    // la gota se queda quieta un rato y despues cae (ciclo)
    float cyc = frac(r2 + t * (0.05 + 0.08 * r3));
    slide = smoothstep(0.55, 1.0, cyc);
    pos.y = lerp(0.15, 0.95, slide);
    pos.x += sin(cyc * 18.0 + r1 * 6.28) * 0.025 * slide;   // zigzag leve
  }
  float sz = size * (0.55 + 0.6 * r3);
  float2 rad = float2(0.16, 0.20 + 0.10 * slide) * sz;     // se alarga al caer
  float2 d = (f - pos) / rad;
  float d2 = dot(d, d);
  float h = saturate(1 - d2);
  float2 grad = (d2 < 1) ? (-2 * d / rad) * 0.045 : 0;
  float wet = smoothstep(0.0, 0.2, h);                        // PES21 (20/10): borde suave (antes 0/1 duro)
  if (moving) {
    // estela mojada por encima de la gota
    float dx = abs(f.x - pos.x);
    float trail = (f.y < pos.y && f.y > 0.05) ? saturate(1 - dx / (0.035 * sz)) * slide : 0;
    wet = max(wet, trail * 0.8);
    grad.x += (f.x > pos.x ? -1 : 1) * trail * 0.25;
  }
  // PES21 (20/10): antialiasing de las gotas. Una gota de menos de ~1 pixel de radio era un espejo diminuto: titilaba como
  // brillantina en los planos medios. Se apaga la gota y el que llama suma la humedad promedio pareja (pelicula fina)
  float px = rad.x / max(fp, 1e-5);
  float k = (fp > 0.0) ? saturate((px - 0.8) / 1.6) : 1.0;
  return float4(grad * k, wet * k, h * k);
}

void main(
  float4 v0 : SV_Position0,
  float4 v1 : COLOR0,
  float4 v2in : TEXCOORD0,   // ce188: UV de la cara en .zw (en 5ce3 era float2 v2); .xy = segundo UV
  float4 v3 : TEXCOORD4,
  float4 v4 : TEXCOORD5,
  float4 v5 : TEXCOORD6,
  float4 v6 : TEXCOORD7,
  float4 v7 : TEXCOORD8,
  float2 w2 : TEXCOORD9,   // PES21 (19/10 c): AL FINAL. En ce188 el VS (455eb97bc2c3a958) saca TEXCOORD9 en su propio registro (v8),
                           // despues de TEXCOORD8; declarado antes corria todos los registros uno y la cara salia rayada. (27/10 j: float2 como el VS)
  float4 pR : TEXCOORD10,   // (27/10 j) posicion de reposo (x, y, z) del VS de la cabeza, w = 11
  float4 pD : TEXCOORD11,   // (01/11 j) gesto respecto de la cabeza rigida (marco de reposo), w = 12
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2,
  out float4 o3 : SV_Target3,
  out float4 o4 : SV_Target4)
{
  float2 v2 = v2in.zw;   // PES21 (19/10): variante de la cara con animacion facial (primeros planos)
  float2 uvD = v2in.xy;  // PES21 (19/10 b): en el original la DIFUSA (t0) se lee con TEXCOORD0.xy y el resto (normal, srm, suciedad,
                         // animacion, mascaras) con .zw. Las caras genericas de Konami (VS 455eb97bc2c3a958) traen dos UV distintos:
                         // con un solo UV la difusa salia como un barrido diagonal.
  float4 r0,r1,r2,r3,r4;
  uint4 bitmask, uiDest;
  float4 fDest;

  // ---- parametros en vivo ----
  float4 P0 = IniParams.Load(int2(0,0));   // x y z w
  float4 P2 = IniParams.Load(int2(2,0));   // x2 y2 z2 w2
  float4 P8 = IniParams.Load(int2(8,0));   // sudor: x8 cantidad, y8 tamano, z8 velocidad, w8 on
  float4 P9 = IniParams.Load(int2(9,0));   // x9 minuto inicio sudor, y9 minuto maximo, z9 vello facial
  // detalle de textura de la cara (16/10): x32 sesgo del color, y32 sesgo del normal map (poros), w32 on/off
  float4 P32 = IniParams.Load(int2(32,0));
  float faceBD = P32.w > 0.5 ? P32.x : 0.0;
  float faceBN = P32.w > 0.5 ? P32.y : 0.0;
  // (08/11, Agustin: "que lo maximo sean texturas en 2k") z32 = tope de resolucion de las texturas de la cara en px (0 = sin tope).
  // Devuelve el mip minimo para que una textura de 4096 nunca se lea por encima de 2048 (las de 2048 o menos no cambian).
  float faceCap = P32.z;
  float tmin = IniParams.Load(int2(10,0)).x;   // minutos reales de partido (los calcula d3dx.ini)
  float sweatProg = saturate((tmin - P9.x) / max(P9.y - P9.x, 0.1));
  bool  enhance      = P0.w > 0.5;
  float sssBoost     = enhance ? P0.x : 1.0;
  float albedoSat    = enhance ? P0.y : 1.0;
  float specScale    = enhance ? P0.z : 1.0;
  float thinWarmth   = enhance ? P2.x : 0.0;
  float glossScale   = enhance ? P2.y : 1.0;
  // ---- arrugas de expresion ----
  float4 P7  = IniParams.Load(int2(7,0));
  float4 P13 = IniParams.Load(int2(13,0));
  float4 P14 = IniParams.Load(int2(14,0));
  float4 P15 = IniParams.Load(int2(15,0));
  bool  wrOn      = P13.w > 0.5;
  float wrGain    = (P7.x  > 0) ? P7.x  : 1.0;
  float wrCurve   = max(P13.x, 0.0);
  float wrCrease  = max(P13.y, 0.0);
  float wrSmile   = max(P13.z, 0.0);
  float4 wrZone   = float4(P14.x > 0 ? P14.x : 1.0, P14.y > 0 ? P14.y : 1.0,
                           P14.z > 0 ? P14.z : 1.0, P14.w > 0 ? P14.w : 1.0);
  float wrYSign   = (P15.w > 0) ? 1.0 : -1.0;
  bool  wrHi      = wrOn && P15.x > 0.5;
  float wrWeight  = 0;     // peso final de arrugas (para el sombreado de pliegues)
  float wrFold    = 0;     // 0..1: cuanto "surco" hay en este pixel
  float wrRidgeF  = 0;     // (04/11 h) 0..1: cresta fina entre arrugas (piel estirada)
  float shFold = 0, shRidge = 0;   // (04/11 h) surco / cresta de las arrugas esculpidas de la sonrisa
  float wrFoldW = 0, shFoldW = 0, wrRidgeW = 0;  // (05/11) los mismos surcos medidos 4 veces mas ancho: zonas suaves para la luz de frente
  float4 P16 = IniParams.Load(int2(16,0));
  bool  exOn      = wrOn && P16.w > 0.5;
  float exBlood   = max(P16.x, 0.0);
  float exGloss   = max(P16.y, 0.0);
  float exTrans   = max(P16.z, 0.0);
  float exFoldZ   = 0;     // zona de pliegue (ancha y suave) * peso de la expresion
  float exRidgeZ  = 0;     // zona de piel estirada
  // valores seguros si el ini no los define (IniParams vale 0)
  if (sssBoost   <= 0) sssBoost   = 1.0;
  if (albedoSat  <= 0) albedoSat  = 1.0;
  if (specScale  <= 0) specScale  = 1.0;
  if (glossScale <= 0) glossScale = 1.0;

  // ---- vectores de movimiento (arreglado: el decompilador generaba w2.xzww) ----
  r0.xy = float2(-0.5,-0.5) + v0.xy;
  r1.xyzw = v7.xyzw / w2.xxyy;
  r1.xyzw = r1.xyzw * float4(0.5,0.5,0.5,0.5) + float4(0.5,0.5,0.5,0.5);
  r1.xyzw = r1.zwxy * float4(1,-1,1,-1) + float4(0,1,0,1);
  r0.zw = r1.zw + -r1.xy;
  o3.xy = g_psSystem.m_renderBuffer.xy * r0.zw;
  float2 pesMotionRaw = r0.zw;   // PES21 cejas: movimiento sin escalar (para las hebras)

  // ---- (01/11 j, hilo expresiones) ARRUGAS POR COMPRESION REAL DE LA PIEL (348.w) ----
  // El VS manda el gesto D (posicion con huesos, amplificador y musculos virtuales menos la cabeza rigida) en el marco de reposo
  // y la posicion de reposo R. Con las derivadas de pantalla se arma el jacobiano de D sobre la superficie de reposo (2 x 2 en la
  // base tangente): su parte simetrica es la deformacion de la piel y su autovalor minimo, si es negativo, cuanto se comprime la
  // piel en este pixel (12 % = arrugas enteras). Eso entra como peso de arrugas donde la mascara del juego las permite, ademas
  // de los pesos por zona del motor: las arrugas aparecen donde la piel realmente se arruga. Derivadas fuera de toda rama (fxc).
  float pesCmp = 0.0;
  {
    float3 dRx = ddx(pR.xyz), dRy = ddy(pR.xyz), dDx = ddx(pD.xyz), dDy = ddy(pD.xyz);
    float pesCmpW = IniParams.Load(int2(348, 0)).w;
    float lx = length(dRx);
    float3 e1 = lx > 1e-9 ? dRx / lx : float3(1.0, 0.0, 0.0);
    float3 e2 = dRy - e1 * dot(dRy, e1); float l2 = length(e2); e2 = l2 > 1e-9 ? e2 / l2 : float3(0.0, 1.0, 0.0);
    float a11 = dot(dRx, e1), a12 = dot(dRx, e2), a21 = dot(dRy, e1), a22 = dot(dRy, e2);
    float b11 = dot(dDx, e1), b12 = dot(dDx, e2), b21 = dot(dDy, e1), b22 = dot(dDy, e2);
    float det = a11 * a22 - a12 * a21;
    if (pesCmpW > 0.0 && pD.w > 9.0 && abs(det) > 1e-14) {
      // J = A^-1 B
      float j11 = ( a22 * b11 - a12 * b21) / det, j12 = ( a22 * b12 - a12 * b22) / det;
      float j21 = (-a21 * b11 + a11 * b21) / det, j22 = (-a21 * b12 + a11 * b22) / det;
      float exx = j11, eyy = j22, exy = 0.5 * (j12 + j21);
      float m = 0.5 * (exx + eyy), rr = sqrt(max(0.25 * (exx - eyy) * (exx - eyy) + exy * exy, 0.0));
      float lmin = m - rr;
      // (01/11 l, capturas: rayas finas en la mandibula con la boca abierta) 20 % = entero y como mucho 0.6 de peso
      pesCmp = saturate(-lmin / 0.20 * pesCmpW) * 0.6;
      // (01/11 k, Agustin: "las arrugas debajo del ojo estan muy fuertes") bajo el ojo (y 167.0..169.6, |x| 1.2..4.8 cm) a un cuarto
      float ueY = smoothstep(1.664, 1.672, pR.y) * (1.0 - smoothstep(1.694, 1.700, pR.y)), ueX = smoothstep(0.009, 0.014, abs(pR.x)) * (1.0 - smoothstep(0.046, 0.052, abs(pR.x)));
      pesCmp *= 1.0 - 0.75 * ueY * ueX;
    }
  }
  // ---- arrugas animadas ----
  r0.z = cmp(v2.x < 0.5);
  if (r0.z != 0) {
    r1.x = g_psMaterial.m_materials[0].y;
  } else {
    r1.x = g_psMaterial.m_materials[4].x;
  }
  r0.zw = float2(1,0.5) * v2.xy;
  r2.xyzw = wrHi ? pesWrinkleM.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw : g_tex_wrinkle_mask.Sample(g_samplerAnisotropic_Clamp_s, r0.zw).xyzw;
  r0.zw = v2.xy * float2(1,0.5) + float2(0,0.5);

  // que metieron las arrugas y se bajan igual los surcos / crestas (brillo, SSS, sombra del surco). Las barbas 3D (pes_beard) no

  if (fhMask > 0.001) { float fhK = 1.0 - 0.75 * fhMask; r1.xyz = normalize(r1.xyz - wrDN * (1.0 - fhK));
    wrFold *= fhK; wrFoldW *= fhK; wrRidgeF *= fhK; wrRidgeW *= fhK; shFold *= fhK; shFoldW *= fhK; shRidge *= fhK; }
  float poreAmt = 0;
  if (enhance) {
    // poros y microrrelieve de la piel (se apaga a distancia para no titilar)

    poreAmt = saturate(1.5 - fw) * (1.0 - fhMask);
    float2 pg = PoreGrad(v2.xy, 1400.0) * 0.10 + PoreGrad(v2.xy + 0.13, 520.0) * 0.06;
    r1.xy += pg * poreAmt;
    r1.xyz = normalize(r1.xyz);
  }
  float pesSwFp = max(length(ddx(v2.xy)), length(ddy(v2.xy)));   // PES21 (20/10): tamano del pixel en uv (fuera de la rama)
  if (enhance && P8.w > 0.5 && (sweatProg > 0.001 || rainK > 0.001)) {
    float t = g_psSystem.m_timeParam.z * max(P8.z, 0.01);
    float4 sA = SweatLayer(v2.xy, 70.0, P8.x * sweatProgZ, max(P8.y, 0.1), t, true, pesSwFp * 70.0);          // PES21 zonas: gotas donde se suda
    float4 sB = SweatLayer(v2.xy + 0.37, 150.0, P8.x * 1.3 * sweatProgZ, max(P8.y, 0.1) * 0.8, t, false, pesSwFp * 150.0);
    float2 sg = sA.xy + sB.xy * 0.7;
    // gotas que ya no se resuelven (menos de ~1 pixel): su humedad promedio, pareja (sin el mosaico de celdas)
    float kA = saturate((0.16 * max(P8.y, 0.1) / max(pesSwFp * 70.0, 1e-5) - 0.8) / 1.6);
    float kB = saturate((0.128 * max(P8.y, 0.1) / max(pesSwFp * 150.0, 1e-5) - 0.8) / 1.6);
    float farW = saturate(P8.x * sweatProgZ * (0.10 * (1.0 - kA) + 0.08 * (1.0 - kB)));
    sweatWet = saturate(max(max(sA.z, sB.z * 0.9), farW));
    r1.xy += sg;
    r1.xyz = normalize(r1.xyz);
  }

  // ---- espacio tangente -> mundo ----
  r0.z = dot(v4.xyz, v4.xyz);
  r0.z = rsqrt(r0.z);
  r2.xyz = v4.xyz * r0.zzz;
  r0.z = dot(v5.xyz, v5.xyz);
  r0.z = rsqrt(r0.z);
  r3.xyz = v5.xyz * r0.zzz;
  r3.xyz = r3.xyz * r1.yyy;
  r1.xyw = r2.xyz * r1.xxx + r3.xyz;
  r0.z = dot(v6.xyz, v6.xyz);
  r0.z = rsqrt(r0.z);
  r2.xyz = v6.xyz * r0.zzz;
  r1.xyz = r2.xyz * r1.zzz + r1.xyw;

  // ---- mascara de rotacion (ojos) ----
  r2.xyz = g_tex_rotation_mask.Sample(g_samplerAnisotropic_Clamp_s, v2.xy).xyz;
  r0.zw = float2(-10,-10) + g_psMaterial.m_materials[5].yz;
  r0.zw = min(float2(0,0), r0.zw);
  r2.x = g_psMaterial.m_materials[5].x * r2.x;
  r0.zw = r0.zw * r2.zy;
  r0.z = -0.0375000015 * r0.z;
  r0.z = r2.x * -0.00999999978 + r0.z;
  r0.z = r0.w * -0.0375000015 + r0.z;
  r1.w = r1.y + r0.z;
  r0.z = dot(r1.xzw, r1.xzw);
  r0.z = rsqrt(r0.z);
  r1.xyz = r1.xwz * r0.zzz;

  // ---- especular / brillo / albedo / suciedad ----
  float faceBS = faceBD, capD = 0.0;
  { uint cW, cH, cL; g_tex_srm.GetDimensions(0, cW, cH, cL); float capS = faceCap > 1.0 ? max(0.0, log2((float)max(cW, cH) / faceCap)) : 0.0; faceBS = max(faceBD, capS - g_tex_srm.CalculateLevelOfDetail(g_sampler_srm_s, v2.xy)); g_tex_diffuse.GetDimensions(0, cW, cH, cL); capD = faceCap > 1.0 ? max(0.0, log2((float)max(cW, cH) / faceCap)) : 0.0; }
  r0.zw = g_tex_srm.SampleBias(g_sampler_srm_s, v2.xy, faceBS).xy;
  float pesSpec0 = r0.z, pesSrmY = r0.w;   // (20/10) valores del juego, para el borde de las genericas
  r1.w = g_tex_diffuse.CalculateLevelOfDetail(g_samplerAnisotropic_Clamp_s, uvD);
  r1.w = clamp(r1.w + faceBD, capD, max(5.0, capD));
  r2.xyz = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, uvD, r1.w).xyz;

  // dibujan, la ceja PINTADA se borra del albedo y se rellena con la piel de alrededor (la mas clara de 8 puntos a ~1.2 cm, mip 3,
  // como el mapa de la ceja de pes_browmap_ps) mezclada con el promedio local; de lejos (sin hebras) la pintura vuelve sola, con el
  // mismo corte de distancia que el GS (y75). x382 = cuanto se borra (0 = como siempre), y382 = corrimiento del corte en mips
  { float4 PE = IniParams.Load(int2(382, 0));
    if (fgOk && PE.x > 0.001 && IniParams.Load(int2(74, 0)).x > 0.5) {
      float4 bm = pesBrowMap.SampleLevel(g_samplerLinear_Clamp_s, uvD, 0);

        float twB, thB; g_tex_diffuse.GetDimensions(twB, thB);
        float texelM = 0.3 / max(twB, 1.0);                                    // ~30 cm de cabeza por vuelta de textura

            float3 cB = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, uvD + float2(cos(aB2) * 0.034, sin(aB2) * 0.026), 3.0).xyz;

        // se borra, y de lejos) toma tambien el color elegido para las hebras de todos (fila 373: x mezcla, yzw rgb), con el tono
        // de cada trazo (claro / oscuro contra su zona): antes cambiaban solo las hebras y la ceja pintada seguia negra debajo
        { float4 G3 = IniParams.Load(int2(373, 0));

            float lP = dot(r2.xyz, float3(0.2126, 0.7152, 0.0722)), lZ = dot(g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, v2.xy, 3.0).xyz, float3(0.2126, 0.7152, 0.0722));

  // pierden (la textura es chica y se ve lavada). Se realza el detalle fino del albedo: la luminancia de este punto contra el

  // No inventa marcas: solo marca mas las que la cara ya tiene. 0 = como el juego
  { float mk = saturate(IniParams.Load(int2(124,0)).y) * (1.0 - fhMask) * saturate(3.5 - r1.w);

      float3 aB = g_tex_diffuse.SampleLevel(g_samplerAnisotropic_Clamp_s, uvD, r1.w + 2.5).xyz;

  r2.w = r3.x * -0.00800000038 + 0.930000007;
  r2.w = max(0.730000019, r2.w);
  r2.w = min(0.930000007, r2.w);
  r4.x = saturate(r4.x);
  r3.x = min(0.660000026, r4.x);

  r2.xyz = r1.www * r3.yzw + r2.xyz;
  r2.w = r0.w * r3.x + -r0.w;
  r0.w = r1.w * r2.w + r0.w;

  r1.w = g_tex_translucent.Sample(g_samplerLinear_Clamp_s, v2.xy).y;

  float4 P62 = IniParams.Load(int2(62,0));

  if (fgOk) r1.w = max(r1.w, FA.b * saturate(P68.z));   // 18/10: grosor de la quinta textura tal cual (la version con umbral no gusto)

  r3.x = g_psSystem.m_param.w * r3.z;

  r0.xy = r0.xy + -r3.xy;
  r0.xy = cmp(r0.xy < float2(0,0));

  //    se ve a traves de la piel: orejas, aletas de la nariz, labios).
  {

    r2.xyz = max(0, lerp(lum.xxx, r2.xyz, albedoSat));

    r2.xyz *= lerp(float3(1,1,1), bloodTint, saturate(thin * thinWarmth));
    r2.xyz *= lerp(float3(1,1,1), float3(1.0, 0.95, 0.94), saturate(mapBlood));   // 21/10: mas sutil   // PES21 mapa de SSS: zonas irrigadas
    // PES21 zonas (18/10): color de labios propio (rosado natural .. rojo), conservando la textura del juego
    { float3 lipTint = lerp(float3(1.10, 0.84, 0.86), float3(1.16, 0.70, 0.72), saturate(P73.y));
      r2.xyz = lerp(r2.xyz, r2.xyz * lipTint, lipM * saturate(P72.z)); }
    if (fgOk) {   // PES21 quinta textura: cavidades (pliegues, alas de nariz, comisuras) mas oscuras y rojizas, y mas mates
      float cavK = FA.r * saturate(P68.y);
      r2.xyz *= lerp(float3(1,1,1), float3(0.80, 0.76, 0.75), cavK);   // (04/11) mas definicion en los pliegues (antes 0.88 0.83 0.81)   // 21/10: casi neutro (con el rojo-marron fuerte la cara quedaba naranja y manchada; el AO ya oscurece el hueco)
      r0.w = lerp(r0.w, max(r0.w, 0.7), cavK * 0.4);

      // en el fondo del surco en proporcion a lo que se hundio (oclusion especular del relieve; hasta -45 % con 0.8 mm o mas)
      { float4 R359 = IniParams.Load(int2(359,0));
        if (IniParams.Load(int2(357,0)).x > 0.5 && R359.x > 0.001 && R359.y > 0.001)
          r0.z *= 1.0 - 0.45 * saturate(FA.r * saturate(R359.x) * R359.y / 0.8); }
    }
    r2.xyz = saturate(r2.xyz);
    // rubor progresivo por el esfuerzo (sube con el reloj del partido)
    float flushF = sweatProg * P12.x;
    // (25/10) sangre por zona (y12): la sangre sube donde la piel esta mas irrigada (mejillas, nariz, orejas: canal de sangre
    // del mapa de SSS y lo fino), no pareja en toda la cara; y con mas melanina se ve mucho menos (piel oscura: casi nada).
    // 0 = rubor parejo como antes
    float flushZ = 1.0;
    if (P12.y > 0.001) {
      float bl = sssMap.w > 0.01 ? sssMap.z * sssMap.w : 0.5;               // sin mapa de SSS: zona media
      float zoneB = saturate(0.30 + 1.10 * bl + 0.45 * saturate(thin));
      float lA = dot(r2.xyz, float3(0.2126, 0.7152, 0.0722));
      float melA = saturate(log2(0.47 / max(lA, 1e-3)) / 2.85);             // indice de melanina (PESMelanin)
      flushZ = lerp(1.0, 1.6 * zoneB * (1.0 - 0.7 * melA), saturate(P12.y));
    }
    flushF *= flushZ;
    r2.xyz *= float3(1.0 + flushF, 1.0 - flushF * 0.35, 1.0 - flushF * 0.45);
  }
  // cavidades (x35): menos luz difusa y menos brillo dentro de los huecos; la translucidez se mantiene (las orejas siguen rojas a contraluz)
  r2.xyz *= pesCav;
  r0.z  *= lerp(1.0, pesCav, 0.8);
  // Pliegues de expresion: dentro del surco llega menos luz y la piel brilla menos
  if (wrOn && wrFold > 0) {
    float fold = saturate(wrFold * wrCrease);
    r2.xyz *= (1.0 - 0.20 * fold);   // (09/11, Agustin: "las arrugas se ven oscuras a veces") 0.35 -> 0.20
    r0.z  *= (1.0 - 0.5 * fold);
