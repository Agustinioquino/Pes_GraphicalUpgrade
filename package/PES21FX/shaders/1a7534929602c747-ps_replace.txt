// === PES21 FX (27/10 n), fila 332: el pasto y la pelota se borraban con la camara quieta ===
// Como funciona el del juego: caja de vecindad = media +- gamma * desvio de 5 muestras del cuadro actual, con gamma = 2.25 por
// abajo y hasta 5.25 por arriba cuando no hay movimiento (muy ancha); la historia se recorta contra esa caja y el peso hacia
// el cuadro actual (alfa) sale solo del recorte y de la velocidad: con la camara quieta y la historia adentro de la caja, alfa = 0
// y la historia acumula para siempre: promedia los jitters y todo lo mas fino que un pixel (hojas, costuras de la pelota) se
// vuelve blando. Cambios (w332 on):
//   x332 alfa minimo hacia el cuadro actual (0.10 = la historia vale ~10 cuadros, como un TAA comun),
//   y332 ancho de la caja (1 = la del juego; 0.75 = mas angosta: menos fantasmas y menos promedio),
//   z332 alfa extra donde el vecindario tiene mucho contraste (pasto, lineas, costuras): ahi se promedia menos todavia.
//
// === (12/11, Agustin: "en cuanto opciones del taa, deberia hacer que el pasto se vea mas nitido al estar quieto") fila 454 (con w332 = 1;
// todo 454 = 0 = como antes). En el juego la camara casi nunca esta quieta del todo: panea despacio (0..0.5 px por cuadro) y ahi la historia
// se vuelve a muestrear corrida una fraccion de pixel cada cuadro y se ablanda (el pasto queda borroso aunque "este quieto"):
//   x454 detalle en movimiento: el realce del cuadro actual sale de cuanto ablanda ese remuestreo (fraccion de pixel y alfa de este pixel),
//        con un margen de 0.2 x el rango del vecindario (~3 % de halo); reemplaza el realce de w440. Con el pixel quieto (< 0.15 px) nunca
//        realza menos que antes,
//   y454 menos titileo en los paneos: en un paneo parejo (la velocidad del tile igual a la propia) de 0.5 px por cuadro o mas baja el
//        alfa (acumula mas cuadros; el paneo lento no se toca), solo cerca de velocidades de pixel entero (con fraccion de pixel bajar
//        el alfa solo ablanda);
//        detras de jugadores y pelota (el tile se mueve mas que el pixel) queda como el juego; nunca por debajo del recorte de la caja,
//   z454 caja pareja con la camara quieta: el lado de arriba de la caja (+5.25 desvios) se acerca al de abajo (-2.25): menos rastro claro,
//   w454 (en la pasada final 0f5c2b99): la nitidez extra de la camara quieta (x330) solo donde el pixel esta quieto o en un paneo lento
//        (completa debajo de 0.5 px por cuadro, nada desde 1.2 px), no en los paneos rapidos.
// Con x / y / z > 0 ademas la mezcla pesada por luminancia nunca queda por debajo del recorte de la caja (antes podia dejar fantasmas),
// y el realce no vuelve a serruchar los bordes que el AA por bordes (y331) acaba de suavizar mientras la camara se mueve.

// === PES21 (04/11, Agustin: "se puede mejorar el antialiasing por fuera del taa") antialiasing por bordes del cuadro ACTUAL antes de
// mezclarlo con la historia (tipo FXAA / SMAA 1x): donde el cuadro trae un borde serruchado (pelo, contorno de la cara, hojas, lineas)
// se promedia a lo largo del borde, asi el alfa hacia el cuadro actual ya no mete dientes de sierra en la historia.
// Fila 331 (x = SSSS de la piel, no se toca): y331 fuerza (0 = apagado, 1 = completo), z331 contraste minimo del borde (0.05..0.3).
// Solo corre en los pixeles de borde (el resto sale por el primer if): ~8 muestras ahi.
float PesLumaAA(float3 c) { return dot(c, float3(0.299, 0.587, 0.114)); }
float3 PesEdgeAA(float2 uv, float3 c, float4 P, out float wE)
{
  wE = 0.0;   // (12/11) x454: cuanto se suavizo este pixel (el realce de abajo no lo vuelve a serruchar)
  float w, h; t0.GetDimensions(w, h);
  float2 px = 1.0 / float2(w, h);
  float3 cNW = t0.SampleLevel(s11, uv + float2(-px.x, -px.y), 0).xyz, cNE = t0.SampleLevel(s11, uv + float2(px.x, -px.y), 0).xyz;
  float3 cSW = t0.SampleLevel(s11, uv + float2(-px.x, px.y), 0).xyz, cSE = t0.SampleLevel(s11, uv + float2(px.x, px.y), 0).xyz;
  float lNW = PesLumaAA(cNW), lNE = PesLumaAA(cNE), lSW = PesLumaAA(cSW), lSE = PesLumaAA(cSE), lM = PesLumaAA(c);
  float lMin = min(lM, min(min(lNW, lNE), min(lSW, lSE)));
  float lMax = max(lM, max(max(lNW, lNE), max(lSW, lSE)));
  if (lMax - lMin < max(clamp(P.z, 0.02, 0.5), lMax * 0.125)) return c;   // sin borde: igual que antes
  float2 dir = float2(-((lNW + lNE) - (lSW + lSE)), (lNW + lSW) - (lNE + lSE));
  float red = max((lNW + lNE + lSW + lSE) * 0.03125, 1.0 / 128.0);
  // (11/11, Agustin: "el taa hace que se pierda detalle") solo en bordes DE VERDAD (contorno, linea, pelo contra el fondo). En textura fina
  // (hojas de 1-2 px, red del arco, costuras) las 4 diagonales no forman un borde: el gradiente se cancela y el FXAA solo borroneaba la hoja.
  // Coherencia = gradiente / rango: ~1 borde recto, ~0.5 diagonal, ~0 una hoja fina o ruido -> sin AA
  float kE = smoothstep(0.25, 0.6, max(abs(dir.x), abs(dir.y)) / (2.0 * (lMax - lMin) + 1e-4));
  if (kE <= 0.0) return c;
  dir = clamp(dir / (min(abs(dir.x), abs(dir.y)) + red), -8.0, 8.0) * px;
  float3 a = 0.5 * (t0.SampleLevel(s11, uv + dir * (1.0 / 3.0 - 0.5), 0).xyz + t0.SampleLevel(s11, uv + dir * (2.0 / 3.0 - 0.5), 0).xyz);
  float3 b = a * 0.5 + 0.25 * (t0.SampleLevel(s11, uv - dir * 0.5, 0).xyz + t0.SampleLevel(s11, uv + dir * 0.5, 0).xyz);
  float lB = PesLumaAA(b);
  float3 r = (lB < lMin || lB > lMax) ? a : b;
  wE = saturate(P.y) * kE;
  return lerp(c, r, saturate(P.y) * kE);
}
