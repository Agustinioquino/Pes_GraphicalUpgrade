  r0.y = 1 + -r0.x;

  r3.z = 2 * g_psMaterial.m_materials[1].z;
  r3.xy = g_psMaterial.m_materials[1].xy * v2.zw;

  r0.w = 0.850000024 + -r2.y;
  r0.w = r2.x * r0.w + r2.y;
  o0.xyz = r0.xyz * r0.www;
  o0.w = 1;
  // PES21 camiseta transpirada (Claude, 16/10): se oscurece con el reloj del partido (x10 = minutos reales)
  // en pecho, espalda y axilas. Zonas en UV del kit (plantilla Konami). x27 fuerza, y27 minutos hasta el maximo,
  // z27 = 1 muestra las zonas en rojo / 2 muestra las UV (u rojo, v verde) para calibrar, w27 on/off
  { float4 SW = IniParams.Load(int2(27,0));
    if (SW.w > 0.5) {
      float2 uvK = v2.xy;
      float tMin = IniParams.Load(int2(10,0)).x;
      float prog = saturate(tMin / max(SW.y, 1.0));
      // zonas (centro, radio) en UV: espalda-columna, pecho, axila izq/der
      float m = 0;
      m = max(m, 1.0 - saturate(length((uvK - float2(0.75, 0.30)) / float2(0.07, 0.20))));
      m = max(m, 0.8 * (1.0 - saturate(length((uvK - float2(0.25, 0.24)) / float2(0.06, 0.12)))));
      m = max(m, 0.9 * (1.0 - saturate(length((uvK - float2(0.08, 0.16)) / float2(0.05, 0.06)))));
      m = max(m, 0.9 * (1.0 - saturate(length((uvK - float2(0.42, 0.16)) / float2(0.05, 0.06)))));
      m = smoothstep(0.0, 1.0, m);
      float sweat = m * prog * saturate(SW.x);
      if (SW.z > 1.5)    { o0.xyz = float3(uvK, 0); if (any(frac(uvK * 8.0) < 0.03)) o0.xyz = 1; }   // rejilla de 8x8
      else if (SW.z > 0.5) o0.xyz = lerp(o0.xyz, float3(1, 0, 0), m * 0.8);
      else {
        o0.xyz *= 1.0 - sweat * 0.38;
        o0.xyz = lerp(o0.xyz, dot(o0.xyz, float3(0.3, 0.59, 0.11)).xxx, sweat * 0.15);
      }
    } }
  r0.xy = v2.zw;
  r0.z = 1 + g_psMaterial.m_materials[0].z;
  r0.xy = g_tex_normal_array.SampleBias(g_sampler_normal_s, r0.xyz, pesBN).yw;
  r0.xy = r0.yx * float2(2,2) + float2(-1,-1);
  r0.w = -r0.x * r0.x + 1;
  r0.w = -r0.y * r0.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.x = rsqrt(r0.w);
  r0.z = r1.x * r0.w;
  r4.xy = v2.zw;
  r4.z = 0;
  r1.xz = g_tex_normal_array.SampleBias(g_sampler_normal_s, r4.xyz, pesBN).yw;
  r4.xy = r1.zx * float2(2,2) + float2(-1,-1);
  r0.w = -r4.x * r4.x + 1;
  r0.w = -r4.y * r4.y + r0.w;
  r0.w = max(0, r0.w);
  r0.w = 9.99999975e-005 + r0.w;
  r1.x = rsqrt(r0.w);
  r4.z = r1.x * r0.w;
  r0.xyz = -r4.xyz + r0.xyz;
  r0.xyz = g_psMaterial.m_materials[0].yyy * r0.xyz + r4.xyz;
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r4.xy = v2.zw;
  r4.z = 1;
  r4.xyzw = g_tex_normal_array.SampleBias(g_sampler_normal_s, r4.xyz, pesBN).xyzw;
  r5.xy = r4.wy * float2(2,2) + float2(-1,-1);
  r1.x = -r5.x * r5.x + 1;
  r1.x = -r5.y * r5.y + r1.x;
  r1.x = max(0, r1.x);
  r1.x = 9.99999975e-005 + r1.x;
  r1.z = rsqrt(r1.x);
  r5.z = r1.x * r1.z;
  r5.xyz = -r0.xyz * r0.www + r5.xyz;
  r0.xyz = r0.xyz * r0.www;
  r6.x = g_psMaterial.m_materials[3].w;
  r6.y = g_psMaterial.m_materials[4].w;
  r0.w = dot(r6.xy, r4.xz);
  r0.xyz = r0.www * r5.xyz + r0.xyz;
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r2.y = r2.y * r2.w;
  r4.xy = r2.yz * float2(2,2) + float2(-1,-1);
  r1.x = -r4.x * r4.x + 1;
  r1.x = -r4.y * r4.y + r1.x;
  r1.x = max(0, r1.x);
  r1.x = 9.99999975e-005 + r1.x;
  r1.z = rsqrt(r1.x);
  r4.z = r1.x * r1.z;
  r2.yzw = r0.xyz * r0.www + -r4.xyz;
  r0.xyz = r0.xyz * r0.www;
  r2.yzw = r2.xxx * r2.yzw + r4.xyz;
  r0.w = dot(r2.yzw, r2.yzw);
  r0.w = rsqrt(r0.w);
  r2.yzw = r2.yzw * r0.www;

  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r0.xyz = r0.xyz * r0.www;

  float pesKitCav = 1;
  { float4 KS = IniParams.Load(int2(22,0));
    if (KS.w > 0.5) {
      float nz = saturate(r0.z);
      pesKitCav = lerp(1.0, nz * nz, saturate(KS.y));
      r0.xy = r0.xy * max(KS.x, 0.0);
      r0.xyz = normalize(float3(r0.xy, max(r0.z, 0.001)));
    } }
  // PES21 estampados (Claude, 16/10): nombre, numero, escudo y sponsors con relieve propio sobre la tela.
  // Borde levantado (vinilo termoaplicado), superficie mas lisa que la trama y un grano fino propio.
  // x38 altura del borde, y38 grano, z38 cuanto aplana la trama debajo, w38 on/off [Alt+Shift+5]
  float2 dxu = ddx(v2.xy), dyu = ddy(v2.xy);   // derivadas fuera del if
  float pmx = ddx(pesPrint), pmy = ddy(pesPrint);
  { float4 PR = IniParams.Load(int2(38,0));
    if (PR.w > 0.5) {
      float m = pesPrint;
      float mx = pmx, my = pmy;
      float det = dxu.x * dyu.y - dxu.y * dyu.x;
      float2 g = abs(det) > 1e-12 ? float2(mx * dyu.y - my * dxu.y, my * dxu.x - mx * dyu.x) / det : float2(0, 0);

      float2 gdir = dot(g, g) > 1e-12 ? normalize(g) : float2(0, 0);
      float2 tilt = -gdir * edge * saturate(PR.x) * 0.55 * farFade;   // bisel del borde, visible a cualquier distancia
      float nearP = saturate(3.0 - fp);                          // grano solo de cerca
      float2 qc = floor(v2.xy * 1800.0);
      float2 hn = float2(frac(sin(dot(qc, float2(12.9898, 78.233))) * 43758.5453), frac(sin(dot(qc, float2(39.346, 11.135))) * 43758.5453)) - 0.5;

      r0.xyz = normalize(float3(r0.xy * flatK + tilt + grain, max(r0.z, 0.001)));

    float fpK = max(length(kdx), length(kdy)) * 512.0;          // texeles del tejido por pixel
    float kf = saturate(2.0 - fpK / 48.0);                       // muy lejos se apaga del todo
    float2 kxy = (kn.xy * 2.0 - 1.0) * saturate(KN.y) * kf * (1.0 - 0.7 * pesPrint);   // el estampado deja ver un poco la trama

    pesKitCav *= lerp(1.0, 0.8 + 0.2 * kn.w, saturate(KN.z) * kf * (1.0 - 0.8 * pesPrint));
  }
  r0.w = dot(v5.xyz, v5.xyz);
  r0.w = rsqrt(r0.w);
  r2.yzw = v5.xyz * r0.www;
  r2.yzw = r2.yzw * r0.yyy;
  r0.y = dot(v4.xyz, v4.xyz);
  r0.y = rsqrt(r0.y);
  r4.xyz = v4.xyz * r0.yyy;
  r0.xyw = r4.xyz * r0.xxx + r2.yzw;
  r1.x = v9.x ? 1 : -1;
  r2.yzw = v6.xyz * r1.xxx;
  r1.x = dot(r2.yzw, r2.yzw);
  r1.x = rsqrt(r1.x);
  r2.yzw = r2.yzw * r1.xxx;
  r0.xyz = r2.yzw * r0.zzz + r0.xyw;
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r0.xyz = r0.xyz * r0.www;
  r0.w = abs(r0.x) + abs(r0.y);
  r0.w = r0.w + abs(r0.z);

  r1.xz = r2.yz * r1.xz;
  r0.xy = r0.zz ? r0.xy : r1.xz;
  r0.xy = r0.xy * float2(0.5,0.5) + float2(0.5,0.5);
  r0.xy = float2(2047,2047) * r0.xy;
  r0.xy = (uint2)r0.xy;

  bitmask.z = ((~(-1 << 1)) << 0) & 0xffffffff;  r0.z = (((uint)r0.y << 0) & bitmask.z) | ((uint)r0.z & ~bitmask.z);

  r0.xyw = (uint3)r0.xyz;

  o1.z = 0.0938416421;
  r0.xy = g_tex_srm.SampleBias(g_sampler_srm_s, v2.xy, pesBD).xy;

  r0.xy = r1.ww * r0.zw + r0.xy;

  r0.y = -r1.z * r1.y + r0.y;
  r0.x = -r1.x + r0.x;
  r0.z = r1.z * r1.y;
  o2.x = r2.x * r0.y + r0.z;
  r0.y = max(g_psMaterial.m_materials[0].w, r1.w);
  r0.z = r2.x * r1.y + r0.y;
  r0.w = r2.x * r1.y;
  o2.z = -r0.w * r0.y + r0.z;
  r0.x = r0.w * r0.x + r1.x;
  r0.y = (int)v9.x & 0x3f800000;
  o2.y = r0.x * r0.y;
  o2.w = g_psMaterial.m_materials[0].x;
  r0.xyzw = v7.xyzw / v8.xxyy;
  r0.xyzw = r0.xyzw * float4(0.5,0.5,0.5,0.5) + float4(0.5,0.5,0.5,0.5);
  r0.xyzw = r0.zwxy * float4(1,-1,1,-1) + float4(0,1,0,1);
  r0.xy = r0.zw + -r0.xy;
  o3.xy = g_psSystem.m_renderBuffer.xy * r0.xy;
  o3.zw = float2(0,0);
  o4.xyzw = float4(0,0,0,1);
  o0.xyz = o0.xyz * pesKitCav;   // PES21 kits: oclusion de pliegues
  o2.z = 0.9607843;   // PES21: marca de kit para la iluminacion de tela (los shaders de luz no usan o2.z en este material)
  return;
