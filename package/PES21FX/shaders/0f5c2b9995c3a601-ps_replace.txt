float3 PESTap(float2 uv) { return inColorTexture.SampleLevel(g_samplerPoint_Clamp_s, uv, 0).xyz; }
float3 PESCas(float2 uv, float2 px, float sharp)
{
  float3 a = PESTap(uv + float2(-px.x, -px.y)), b = PESTap(uv + float2(0, -px.y)), c = PESTap(uv + float2(px.x, -px.y));
  float3 d = PESTap(uv + float2(-px.x, 0)),     e = PESTap(uv),                    f = PESTap(uv + float2(px.x, 0));
  float3 g = PESTap(uv + float2(-px.x, px.y)),  h = PESTap(uv + float2(0, px.y)),  i = PESTap(uv + float2(px.x, px.y));
  float3 mn = min(min(min(d, e), min(f, b)), h);
  float3 mn2 = min(mn, min(min(a, c), min(g, i)));
  mn += mn2;
  float3 mx = max(max(max(d, e), max(f, b)), h);
  float3 mx2 = max(mx, max(max(a, c), max(g, i)));
  mx += mx2;
  float3 amp = saturate(min(mn, 2.0 - mx) / max(mx, 1e-4));
  amp = sqrt(amp);
  float peak = -1.0 / lerp(8.0, 5.0, saturate(sharp));
  float3 w = amp * peak;
  float3 rw = 1.0 / (1.0 + 4.0 * w);
  return saturate((b * w + d * w + f * w + h * w + e) * rw);
}
