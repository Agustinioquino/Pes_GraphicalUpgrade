cbuffer cPSScene : register(b2)
{
  struct
  {
    float4x4 m_projectionView;
    float4x4 m_projection;
    float4x4 m_view;
    float4x4 m_shadowProjection;
    float4x4 m_shadowProjection2;
    float4 m_eyepos;
    float4 m_projectionParam;
    float4 m_viewportSize;
    float4 m_exposure;
    float4 m_fogParam[3];
    float4 m_fogColor;
    float4 m_cameraCenterOffset;
    float4 m_shadowMapResolutions;
  } g_psScene : packoffset(c0);
}

cbuffer cPSLight : register(b3)
{
  struct
  {
    float4 m_lightDir0;
    float4 m_lightColor0;
    float4 m_lightColor1;
    float4 m_lightSourcePos;
    float4 m_lightSourceColor;
    float4 m_lightSourceDir;
    float4 m_lightSourceParam1;
    float4 m_lightSourceParam2;
    float4 m_lightSourceParam3;
    float4 m_lightPos0;
    float4 m_lightParam2;
  } g_psLight : packoffset(c0);
}

cbuffer cPSMaterial : register(b4)
{
  struct
  {
    float4 m_materials[8];
  } g_psMaterial : packoffset(c0);
}

SamplerState g_samplerPoint_Wrap_s : register(s8);
SamplerState g_samplerPoint_Clamp_s : register(s9);
SamplerState g_samplerLinear_Clamp_s : register(s11);
Texture2D<float4> g_NormalTexture : register(t0);
Texture2D<float4> g_SpecularTexture : register(t1);
Texture2D<float4> g_DepthTexture : register(t2);
Texture2D<float4> g_lut1 : register(t9);
Texture2D<float4> g_MaterialTexture : register(t10);

// 3Dmigoto declarations
#define cmp -
Texture1D<float4> IniParams : register(t120);
Texture2D<float4> StereoParams : register(t125);
#include "pes_common.hlsli"
float3 PESSunColor() { return g_psLight.m_lightSourceColor.xyz; }   // para pes_skinlight: el tinte del brillo sigue al color del foco
#include "pes_skinlight.hlsli"

void main(
  float4 v0 : TEXCOORD0,
  float4 v1 : TEXCOORD1,
  float4 v2 : SV_Position0,
  out float4 o0 : SV_Target0,
  out float4 o1 : SV_Target1,
  out float4 o2 : SV_Target2)
{
  float2 uv = v1.xy / v1.ww;
  float4 nt = g_NormalTexture.Sample(g_samplerPoint_Wrap_s, uv);
  float4 st = g_SpecularTexture.Sample(g_samplerPoint_Wrap_s, uv);   // z = marca de piel / F0, x = aspereza, y = intensidad, w = material
  uint3 q = (uint3)(nt.xyw * float3(1023, 1023, 3));
  uint ux = (q.x << 1) | (q.z >> 1);
  uint uy = (q.y << 1) | (q.z & 1);
  float2 e = float2((float)ux, (float)uy) * asfloat(0x3a801002) - 1.0;
  float3 n;
  n.z = 1.0 - abs(e.x) - abs(e.y);
  float2 sg = float2(e.x >= 0 ? 1.0 : -1.0, e.y >= 0 ? 1.0 : -1.0);
  n.xy = n.z >= 0 ? e : float2((1.0 - abs(e.y)) * sg.x, (1.0 - abs(e.x)) * sg.y);
  n = n * rsqrt(dot(n, n));
  uint mv = (uint)(nt.z * 1023 + 0.5);
  int mat = (int)(mv & 15);
  float trans = 0, matU = 0;
  if (mat == 1) { trans = 0; matU = 0; }
  else if (mat == 3) { trans = (float)(mv >> 4) * asfloat(0x3c820821); matU = 0; }
  else { trans = (float)(mv >> 4) * asfloat(0x3c820821); matU = st.w; }
  float d = g_DepthTexture.Sample(g_samplerPoint_Wrap_s, uv).x;
  float2 pp = (v0.xy / v0.ww - g_psScene.m_cameraCenterOffset.xy) * g_psScene.m_projectionParam.xy;
  float3 P; P.z = g_psScene.m_projectionParam.z / (d - g_psScene.m_projectionParam.w); P.xy = pp * P.z;
  float3 Ve = P * rsqrt(dot(P, P));            // de la camara al pixel (= -V)
  float3 Lv = g_psLight.m_lightSourcePos.xyz - P;
  float dist2 = dot(Lv, Lv), invD = rsqrt(dist2);
  float dist = max(dist2 * invD, g_psMaterial.m_materials[7].x);
  float3 L = Lv * invD;
  float att = invD * invD - dist2 * g_psLight.m_lightSourceParam1.w;
  att = min(att, g_psMaterial.m_materials[7].w);
  float cone = exp2(log2(saturate((dot(g_psLight.m_lightSourceDir.xyz, -L) - g_psLight.m_lightSourceParam2.x) * g_psLight.m_lightSourceParam2.y)) * g_psLight.m_lightSourceParam2.z);
  att = max(att, 0);
  float3 col = cone * (att * g_psLight.m_lightSourceColor.xyz) * g_psScene.m_exposure.z;
  float srcR = saturate(g_psLight.m_lightSourceColor.w / dist * 0.9);   // radio aparente de la luz (luz de esfera)
  float f0 = 0, floorK = 0; float3 specC = 1;
  if (mat == 1) { f0 = st.z * 0.96 + 0.04; specC = float3(1, 1, 1); floorK = 0; trans = 0; }
  else if (mat == 3) { f0 = st.z * 0.96 + 0.04; specC = float3(0.9, 1, 0.8); floorK = 1; }
  else {
    f0 = g_MaterialTexture.SampleLevel(g_samplerPoint_Clamp_s, float2(matU, 0.75), 0).x;
    float4 m9 = g_MaterialTexture.SampleLevel(g_samplerPoint_Clamp_s, float2(matU, 0.25), 0);
    specC = m9.xyz; floorK = m9.w;
  }
  float3 nn = float3(n.x, n.y, -n.z);
  float nl = dot(L, nn);
  float nv = dot(-Ve, nn);
  float3 Rn = nn * -(nv + nv) - Ve;             // = -reflejo
  float3 c2r = -L * 1.0 - Rn;                   // del centro de la luz al rayo reflejado
  float3 Lr = min(srcR / sqrt(dot(c2r, c2r)), 1.0) * c2r + L;
  float3 H = Lr * rsqrt(dot(Lr, Lr)) - Ve;
  H = H * rsqrt(dot(H, H));
  float nh = saturate(dot(H, nn));
  float dGame = saturate(max(floorK * trans, (1 - nl * nl) * trans + nl));
  float nlS = saturate(nl);
  float3 diff = col * dGame;
  float3 specI = st.y * specC;
  float F = g_lut1.Sample(g_samplerLinear_Clamp_s, float2(dot(H, L), f0)).x;
  float nvS = saturate(nv);
  float al = max(st.x, 0.02); al = al * al;
  float k = al * 0.5;
  float dd = nh * nh * (al * al - 1) + 1; dd = dd * dd;
  float D = al * al / dd;
  float vis = 1.0 / ((nlS * (1 - k) + k) * (nvS * (1 - k) + k));
  float sp = F * D * vis * nlS * 0.25;
  float3 spec = col * sp * specI;
  // piel (pes_skinlight): curvatura fuera de los if (usa derivadas)
  float3 skN = normalize(nn);
  float skCurv = PesSkCurv(skN, P.z);
  bool skOn = PesSkIsSkin(mat, trans, st.z);
  if (skOn) {
    diff = (PesSkDiffuse(nl, trans, floorK * trans, P.z, skCurv) + PesSkTrans(skN, -Ve, L, trans)) * col;
    spec = PesSkSpec(spec, skN, -Ve, L, st.x, st.y, col, 1.0);
  }
  o0.xyz = diff * g_psLight.m_lightSourceParam3.y;
  o1.xyz = min(spec * g_psLight.m_lightSourceParam3.x, 10000.0.xxx);
  o0.w = 1;
  o1.w = 1;
  o2.xyzw = float4(1, 1, 1, 1);
  return;
}
