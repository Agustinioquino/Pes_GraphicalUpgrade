// PES 2021 - vertex shader de la piel del cuerpo (brazos, piernas, cuello, torso; malla animada por huesos, PS aa7d1833)
// PES21 FX (24/10): igual al del juego (volcado por 3Dmigoto) mas el resaltado del musculo elegido en el menu Fisico: un
// peso por vertice segun su posicion de REPOSO (v0, espacio del modelo: X = izquierda, Y = arriba, Z = adelante; con |x|
// cada capsula sirve para los dos lados). Las capsulas las arma la DLL (fisico.cpp) en x174..182; el PS lee el peso en
// o3.w (+2, para distinguirlo de los VS que no lo escriben: o3.w queda en [2, 3]).
// (25/10) manos: si el dibujo es una mano de TorsoBody, el skinning lo hace pes_hands.hlsli (t126 enlazado por manos.cpp).
// (24/10 c) o4.w / o5.w = cercania a las marcas 0 / 1 del musculo (187 / 188; signo = lado: + izquierdo, - derecho): con eso el
// PS de la piel (variante pes_skin_hl_ps.hlsl, solo en el menu Fisico) junta en u5 donde quedo cada marca en la pantalla y la
// DLL dibuja encima el aro y el punto blanco, como una capa de interfaz. o6.w = jugador de la tabla de tonos de piel
// (pes_skintone.hlsli: 1000 + entrada, 999 = ninguno).
// ---- Created with 3Dmigoto v1.3.16 on Thu Oct 08 21:08:23 2026

cbuffer cVSScene : register(b2)
{

  struct
  {
    float4x4 m_projectionView;
    float4x4 m_projection;
    float4x4 m_view;
    float4x4 m_shadowProjection;
    float4x4 m_shadowProjection2;
    float4 m_eyepos;
    float4 m_projectionParam;
    float4 m_viewportSize;
    float4 m_exposure;
    float4 m_fogParam[3];
    float4 m_fogColor;
    float4 m_cameraCenterOffset;
    float4 m_shadowMapResolutions;
  } g_vsScene : packoffset(c0);

}

cbuffer cVSObject : register(b5)
{

  struct
  {
    float4x4 m_viewWorld;
    float4x4 m_world;

    float4 m_localParam[9];
  } g_vsObject : packoffset(c0);

}

cbuffer VSBones : register(b6)
{

  struct
  {
    float4x3 m_boneMatrices[32];
  } g_vsBone : packoffset(c0);


  struct
  {
    float4x3 m_boneMatrices[32];
  } g_vsPrevBone : packoffset(c96);

}



// 3Dmigoto declarations
#define cmp -
Texture1D<float4> IniParams : register(t120);
Texture2D<float4> StereoParams : register(t125);
#include "pes_skintone.hlsli"
#include "pes_hands.hlsli"   // (25/10) manos: cuaterniones duales + correctivas por pose (solo en los dibujos de la mano)


void main(
  float4 v0 : POSITION0,
  float4 v1 : NORMAL0,
  float4 v2 : COLOR0,
  float4 v3 : TANGENT0,
  uint4 v4 : BLENDINDICES0,
  float4 v5 : BLENDWEIGHT0,
  float4 v6 : TEXCOORD0,
  float4 v7 : TEXCOORD1,
  float4 v8 : TEXCOORD2,
  float2 v9 : TEXCOORD3,
  uint vid : SV_VertexID,
  out float4 o0 : SV_Position0,
  out float4 o1 : COLOR0,
  out float2 o2 : TEXCOORD0,
  out float4 p2 : TEXCOORD9,
  out float4 o3 : TEXCOORD4,
  out float4 o4 : TEXCOORD5,
  out float4 o5 : TEXCOORD6,
  out float4 o6 : TEXCOORD7,
  out float4 o7 : TEXCOORD8)
{
  float4 r0,r1,r2,r3,r4,r5,r6;
  uint4 bitmask, uiDest;
  float4 fDest;

  r0.x = cmp(g_vsObject.m_useWeightCount.x != 0.000000);
  float3 hP, hN, hT, hPP; float hDbg;
  bool handDone = (r0.x != 0) && PesHandSkin(vid, v0.xyz, v1.xyz, v3.xyz, v4, v5, hP, hN, hT, hPP, hDbg);
  if (handDone) {
    r4.xyz = hP; r5.xyz = hN; r1.xyz = hT; r2.xyz = hPP;
  } else if (r0.x != 0) {
    r0.xyzw = (int4)v4.xyzw * int4(3,3,3,3);
    r1.xyzw = g_vsBone.m_boneMatrices[v4.y]._m00_m10_m20_m30 * v5.yyyy;
    r2.xyzw = g_vsBone.m_boneMatrices[v4.y]._m01_m11_m21_m31 * v5.yyyy;
    r3.xyzw = g_vsBone.m_boneMatrices[v4.y]._m02_m12_m22_m32 * v5.yyyy;
    r1.xyzw = g_vsBone.m_boneMatrices[v4.x]._m00_m10_m20_m30 * v5.xxxx + r1.xyzw;
    r2.xyzw = g_vsBone.m_boneMatrices[v4.x]._m01_m11_m21_m31 * v5.xxxx + r2.xyzw;
    r3.xyzw = g_vsBone.m_boneMatrices[v4.x]._m02_m12_m22_m32 * v5.xxxx + r3.xyzw;
    r1.xyzw = g_vsBone.m_boneMatrices[v4.z]._m00_m10_m20_m30 * v5.zzzz + r1.xyzw;
    r2.xyzw = g_vsBone.m_boneMatrices[v4.z]._m01_m11_m21_m31 * v5.zzzz + r2.xyzw;
    r3.xyzw = g_vsBone.m_boneMatrices[v4.z]._m02_m12_m22_m32 * v5.zzzz + r3.xyzw;
    r1.xyzw = g_vsBone.m_boneMatrices[v4.w]._m00_m10_m20_m30 * v5.wwww + r1.xyzw;
    r2.xyzw = g_vsBone.m_boneMatrices[v4.w]._m01_m11_m21_m31 * v5.wwww + r2.xyzw;
    r3.xyzw = g_vsBone.m_boneMatrices[v4.w]._m02_m12_m22_m32 * v5.wwww + r3.xyzw;
    r4.x = dot(v0.xyzw, r1.xyzw);
    r4.y = dot(v0.xyzw, r2.xyzw);
    r4.z = dot(v0.xyzw, r3.xyzw);
    r5.x = dot(v1.xyz, r1.xyz);
    r5.y = dot(v1.xyz, r2.xyz);
    r5.z = dot(v1.xyz, r3.xyz);
    r1.x = dot(v3.xyz, r1.xyz);
    r1.y = dot(v3.xyz, r2.xyz);
    r1.z = dot(v3.xyz, r3.xyz);
    r2.xyzw = g_vsPrevBone.m_boneMatrices[v4.y]._m00_m10_m20_m30 * v5.yyyy;
    r3.xyzw = g_vsPrevBone.m_boneMatrices[v4.y]._m01_m11_m21_m31 * v5.yyyy;
    r6.xyzw = g_vsPrevBone.m_boneMatrices[v4.y]._m02_m12_m22_m32 * v5.yyyy;
    r2.xyzw = g_vsPrevBone.m_boneMatrices[v4.x]._m00_m10_m20_m30 * v5.xxxx + r2.xyzw;
    r3.xyzw = g_vsPrevBone.m_boneMatrices[v4.x]._m01_m11_m21_m31 * v5.xxxx + r3.xyzw;
    r6.xyzw = g_vsPrevBone.m_boneMatrices[v4.x]._m02_m12_m22_m32 * v5.xxxx + r6.xyzw;
    r2.xyzw = g_vsPrevBone.m_boneMatrices[v4.z]._m00_m10_m20_m30 * v5.zzzz + r2.xyzw;
    r3.xyzw = g_vsPrevBone.m_boneMatrices[v4.z]._m01_m11_m21_m31 * v5.zzzz + r3.xyzw;
    r6.xyzw = g_vsPrevBone.m_boneMatrices[v4.z]._m02_m12_m22_m32 * v5.zzzz + r6.xyzw;
    r2.xyzw = g_vsPrevBone.m_boneMatrices[v4.w]._m00_m10_m20_m30 * v5.wwww + r2.xyzw;
    r3.xyzw = g_vsPrevBone.m_boneMatrices[v4.w]._m01_m11_m21_m31 * v5.wwww + r3.xyzw;
    r0.xyzw = g_vsPrevBone.m_boneMatrices[v4.w]._m02_m12_m22_m32 * v5.wwww + r6.xyzw;
    r2.x = dot(v0.xyzw, r2.xyzw);
    r2.y = dot(v0.xyzw, r3.xyzw);
    r2.z = dot(v0.xyzw, r0.xyzw);
  } else {
    r5.xyz = v1.xyz;
    r1.xyz = v3.xyz;
    r4.xyz = v0.xyz;
    r2.xyz = v0.xyz;
  }
  r4.w = v0.w;
  r0.x = dot(r4.xyzw, g_vsObject.m_viewWorld._m00_m10_m20_m30);
  r0.y = dot(r4.xyzw, g_vsObject.m_viewWorld._m01_m11_m21_m31);
  r0.z = dot(r4.xyzw, g_vsObject.m_viewWorld._m02_m12_m22_m32);
  r0.w = dot(r4.xyzw, g_vsObject.m_viewWorld._m03_m13_m23_m33);
  r3.x = dot(r0.xyzw, g_vsScene.m_projection._m00_m10_m20_m30);
  r3.y = dot(r0.xyzw, g_vsScene.m_projection._m01_m11_m21_m31);
  o0.z = dot(r0.xyzw, g_vsScene.m_projection._m02_m12_m22_m32);
  r3.w = dot(r0.xyzw, g_vsScene.m_projection._m03_m13_m23_m33);
  r2.w = v0.w;
  r4.x = dot(r2.xyzw, g_vsScene.m_shadowProjection._m00_m10_m20_m30);
  r4.y = dot(r2.xyzw, g_vsScene.m_shadowProjection._m01_m11_m21_m31);
  r4.z = dot(r2.xyzw, g_vsScene.m_shadowProjection._m02_m12_m22_m32);
  r4.w = dot(r2.xyzw, g_vsScene.m_shadowProjection._m03_m13_m23_m33);
  o7.z = dot(r4.xyzw, g_vsScene.m_shadowProjection2._m00_m10_m20_m30);
  o7.w = dot(r4.xyzw, g_vsScene.m_shadowProjection2._m01_m11_m21_m31);
  p2.y = dot(r4.xyzw, g_vsScene.m_shadowProjection2._m03_m13_m23_m33);
  r0.w = dot(r0.xyz, r0.xyz);
  r0.w = rsqrt(r0.w);
  r0.xyz = r0.xyz * r0.www;
  r2.x = dot(r5.xyz, g_vsObject.m_viewWorld._m00_m10_m20);
  r2.y = dot(r5.xyz, g_vsObject.m_viewWorld._m01_m11_m21);
  r2.z = dot(r5.xyz, g_vsObject.m_viewWorld._m02_m12_m22);
  r0.w = dot(r2.xyz, r2.xyz);
  r0.w = rsqrt(r0.w);
  r2.xyz = r2.xyz * r0.www;
  r4.x = dot(r1.xyz, g_vsObject.m_viewWorld._m00_m10_m20);
  r4.y = dot(r1.xyz, g_vsObject.m_viewWorld._m01_m11_m21);
  r4.z = dot(r1.xyz, g_vsObject.m_viewWorld._m02_m12_m22);
  r0.w = dot(r4.xyz, r4.xyz);
  r0.w = rsqrt(r0.w);
  r1.xyz = r4.xyz * r0.www;
  r4.xyz = r2.zxy * r1.yzx;
  r4.xyz = r2.yzx * r1.zxy + -r4.xyz;
  r0.w = dot(r4.xyz, r4.xyz);
  r0.w = rsqrt(r0.w);
  r4.xyz = r4.xyz * r0.www;
  r4.xyz = v3.www * r4.xyz;
  o3.xyz = -r0.xyz;
  o4.xyz = r1.xyz;
  o5.xyz = r4.xyz;
  o6.xyz = r2.xyz;
  o0.xyw = r3.xyw;
  o1.xyz = v2.xyz;
  o1.w = g_vsObject.m_useWeightCount.w;
  o7.xy = r3.xy;
  o2.xy = v6.xy;
  p2.x = r3.w;
  p2.zw = v0.yx;   // (24/10 e) posicion de reposo (y, x) del vertice: el PS mide el color del cuello con esto
  {   // ---- PES21 (24/10): resaltado del musculo elegido (x174 pulso, y174 capsulas, z174 todo el cuerpo; 175 + 2k = A y radio,
      // 176 + 2k = B): 1 adentro de la capsula, baja suave hasta 0 en el borde
    float4 HL = IniParams.Load(int2(174,0));
    float hw = 0;
    if (HL.x > 0.001) {
      if (HL.z > 0.5) hw = 1;
      else {
        float3 hp = float3(abs(v0.x), v0.y, v0.z);
        int hn = (int)(HL.y + 0.5);
        [loop] for (int hk = 0; hk < 4; hk++) {
          if (hk >= hn) break;
          float4 HA = IniParams.Load(int2(175 + 2 * hk, 0));
          float4 HB = IniParams.Load(int2(176 + 2 * hk, 0));
          float3 hab = HB.xyz - HA.xyz;
          float ht = saturate(dot(hp - HA.xyz, hab) / max(dot(hab, hab), 1e-6));
          float hd = length(hp - (HA.xyz + hab * ht)) / max(HA.w, 1e-3);
          hw = max(hw, 1 - smoothstep(0.45, 1.0, hd));
        }
      }
    }
    o3.w = 2.0 + hw;
    // (24/10 c) cercania a las marcas del musculo (para el aro de la capa de arriba); el signo dice el lado
    float pm0 = 0, pm1 = 0;
    int nm = (int)(HL.w + 0.5);
    if (HL.x > 0.001 && nm > 0) {
      float3 hp = float3(abs(v0.x), v0.y, v0.z);
      float sideS = v0.x >= 0 ? 1.0 : -1.0;
      float4 M0 = IniParams.Load(int2(187, 0));
      pm0 = sideS * (1.0 - smoothstep(0.55, 1.0, length(hp - M0.xyz) / max(M0.w, 1e-3)));
      if (nm > 1) { float4 M1 = IniParams.Load(int2(188, 0)); pm1 = sideS * (1.0 - smoothstep(0.55, 1.0, length(hp - M1.xyz) / max(M1.w, 1e-3))); }
    }
    o4.w = pm0; o5.w = pm1;
  }
  {   // (24/10 c) tono de piel por jugador: un punto del dibujo en el mundo (pes_skintone.hlsli) -> entrada mas cercana
    bool sk = g_vsObject.m_useWeightCount.x != 0;
    float3 b0 = float3(g_vsBone.m_boneMatrices[0]._m30, g_vsBone.m_boneMatrices[0]._m31, g_vsBone.m_boneMatrices[0]._m32);
    float3 b0b = float3(g_vsPrevBone.m_boneMatrices[0]._m30, g_vsPrevBone.m_boneMatrices[0]._m31, g_vsPrevBone.m_boneMatrices[0]._m32);
    float3 sw = PesSkinPoint(g_vsObject.m_world, g_vsObject.m_viewWorld, g_vsScene.m_view, b0, b0b, sk);
    int se = PesSkinNearest(sw);
    o6.w = se >= 0 ? 1000.0 + se + 0.4 * PesTorsoMask(v0.xyz) : 999.0;   // (24/10 g) + cuanto de torso (solo el torso)
  }
  return;
}
