float PESStripeST(float3 g, float3 r4) {
  float3 t = float3(g.x * g.y, -(g.z * g.z + g.x * g.x), g.z * g.y);

  return g_tex_turfSlopTable.SampleLevel(g_samplerLinear_Clamp_s, uv, 0).y;
}

      // (25/10 c) umbrales fijos, los mismos que la DLL (Thresholds): la densidad ya no los corre
      float T2 = 45.0, T3 = 15.0;
      float kFar = 1.0 - smoothstep(T3 * 0.75, T2 * 1.25, pxm);
      if (kFar > 0.001) { float3 cg = o0.xyz; float sp = o2.y; PESFarTurf(cg, sp, pesV, v8.xz, pxm, kFar * saturate(Q61.y) * (1.0 - pesLine), Q61.z, Q61.w); o0.xyz = cg; o2.y = sp; }
    }
  }
  return;
