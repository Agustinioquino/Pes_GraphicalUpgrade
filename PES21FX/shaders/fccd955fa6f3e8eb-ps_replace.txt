  r1.xyz = r1.xyz * r0.www;

  r0.w = rsqrt(r0.w);

  }

      float l = dot(c, float3(0.2126, 0.7152, 0.0722));
      c = lerp(c, saturate(lerp(l.xxx, c, 1.1)), smoothstep(0.4, 0.8, eyeIris));
      o0.xyz = saturate(c);
      o2.x = min(o2.x, 0.12);
      o2.y = max(o2.y, 0.55);
    }
  }
  return;
