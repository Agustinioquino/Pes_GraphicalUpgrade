cbuffer cPSObject : register(b5)
{

  struct
  {
    float4x4 m_viewWorld;
    float4x4 m_world;
    float4 m_useWeightCount;
    float4 m_localParam[9];
  } g_psObject : packoffset(c0);

}

SamplerState g_samplerLinear_Clamp_s : register(s11);
Texture2D<float4> inImage : register(t0);


// 3Dmigoto declarations
#define cmp -
Texture1D<float4> IniParams : register(t120);
Texture2D<float4> StereoParams : register(t125);


// PES21 bloom v2 (Claude, 16/10): reduccion de 13 muestras (estilo CoD/Jimenez) en vez de 4 -> bloom suave, sin bloques
// y con supresion de "luciernagas" (puntos brillantes que titilan). x40 = 1 activa, y40 supresion (0..1), w40 on/off [Alt+Shift+7]
float3 PESBl(float2 uv) { return inImage.SampleLevel(g_samplerLinear_Clamp_s, clamp(uv, 0, g_psObject.m_localParam[2].zw), 0).xyz; }
float PESKw(float3 c, float k) { return 1.0 / (1.0 + k * dot(c, float3(0.2126, 0.7152, 0.0722))); }
void main(
  float2 v0 : TEXCOORD0,
  float4 v1 : SV_Position0,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2;
  uint4 bitmask, uiDest;
  float4 fDest;
  float4 BL = IniParams.Load(int2(40,0));
  if (BL.w > 0.5) {
    uint sw, sh; inImage.GetDimensions(sw, sh);
    float2 tx = 1.0 / float2(max(sw, 1u), max(sh, 1u));
    float2 uv = v1.xy * g_psObject.m_localParam[3].xy + float2(g_psObject.m_localParam[0].x + g_psObject.m_localParam[0].y, g_psObject.m_localParam[0].z + g_psObject.m_localParam[0].w) * 0.5;
    float3 A = PESBl(uv + tx * float2(-2, -2)), B = PESBl(uv + tx * float2(0, -2)), C = PESBl(uv + tx * float2(2, -2));
    float3 D = PESBl(uv + tx * float2(-1, -1)), E = PESBl(uv + tx * float2(1, -1));
    float3 F = PESBl(uv + tx * float2(-2, 0)),  G = PESBl(uv),                     H = PESBl(uv + tx * float2(2, 0));
    float3 I = PESBl(uv + tx * float2(-1, 1)),  J = PESBl(uv + tx * float2(1, 1));
    float3 K = PESBl(uv + tx * float2(-2, 2)),  L = PESBl(uv + tx * float2(0, 2)),  M = PESBl(uv + tx * float2(2, 2));
    float k = saturate(BL.y) * 4.0;
    float3 g0 = (D + E + I + J) * 0.25, g1 = (A + B + G + F) * 0.25, g2 = (B + C + H + G) * 0.25, g3 = (F + G + L + K) * 0.25, g4 = (G + H + M + L) * 0.25;
    float w0 = 0.5 * PESKw(g0, k), w1 = 0.125 * PESKw(g1, k), w2 = 0.125 * PESKw(g2, k), w3 = 0.125 * PESKw(g3, k), w4 = 0.125 * PESKw(g4, k);
    float3 res = (g0 * w0 + g1 * w1 + g2 * w2 + g3 * w3 + g4 * w4) / max(w0 + w1 + w2 + w3 + w4, 1e-5);
    float2 hiA = g_psObject.m_localParam[2].zw;
    float aw = (inImage.SampleLevel(g_samplerLinear_Clamp_s, clamp(uv + tx * float2(-1, -1), 0, hiA), 0).w + inImage.SampleLevel(g_samplerLinear_Clamp_s, clamp(uv + tx * float2(1, -1), 0, hiA), 0).w
              + inImage.SampleLevel(g_samplerLinear_Clamp_s, clamp(uv + tx * float2(-1, 1), 0, hiA), 0).w + inImage.SampleLevel(g_samplerLinear_Clamp_s, clamp(uv + tx * float2(1, 1), 0, hiA), 0).w) * 0.25;
    o0 = float4(res, aw) * g_psObject.m_localParam[3].z;
    return;
  }
  r0.xyzw = v1.xyxy * g_psObject.m_localParam[3].xyxy + g_psObject.m_localParam[0].xzyz;
  r0.xyzw = max(float4(0,0,0,0), r0.xyzw);
  r0.xyzw = min(g_psObject.m_localParam[2].zwzw, r0.xyzw);
  r1.xyzw = inImage.Sample(g_samplerLinear_Clamp_s, r0.xy).xyzw;
  r0.xyzw = inImage.Sample(g_samplerLinear_Clamp_s, r0.zw).xyzw;
  r0.xyzw = r1.xyzw + r0.xyzw;
  r1.xyzw = v1.xyxy * g_psObject.m_localParam[3].xyxy + g_psObject.m_localParam[0].xwyw;
  r1.xyzw = max(float4(0,0,0,0), r1.xyzw);
  r1.xyzw = min(g_psObject.m_localParam[2].zwzw, r1.xyzw);
  r2.xyzw = inImage.Sample(g_samplerLinear_Clamp_s, r1.xy).xyzw;
  r1.xyzw = inImage.Sample(g_samplerLinear_Clamp_s, r1.zw).xyzw;
  r0.xyzw = r2.xyzw + r0.xyzw;
  r0.xyzw = r0.xyzw + r1.xyzw;
  r1.x = 0.25 * g_psObject.m_localParam[3].z;
  o0.xyzw = r1.xxxx * r0.xyzw;
  return;
}
