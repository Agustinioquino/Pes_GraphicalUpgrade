// (13/11 tanda 2) la correccion del amarillo (x29) del natural, la misma en la hoja y en el color plano de y330: asi rel = o0 / flatC ya no
// la lleva y la base (que ya la tuvo) no la recibe dos veces
void PESNatX29(inout float3 c) {
  float4 GVn = IniParams.Load(int2(29,0));
  if (GVn.w > 0.5) {
    const float3 LWn = float3(0.2126, 0.7152, 0.0722);
    float lgn = dot(c, LWn);
    float yeln = saturate((c.r / max(c.g, 1e-3) - 0.55) * 2.5);
    yeln *= saturate(((max(c.r, max(c.g, c.b)) - min(c.r, min(c.g, c.b))) / max(max(c.r, max(c.g, c.b)), 1e-3) - 0.12) * 5.0);
    float3 natn = c * float3(0.80, 1.0, 0.95);
    natn *= lgn / max(dot(natn, LWn), 1e-4);
    c = lerp(c, natn, saturate(GVn.x) * yeln);
    c = lerp(dot(c, LWn).xxx, c, GVn.z > 0 ? GVn.z : 1.0);
    c *= GVn.y > 0 ? GVn.y : 1.0;
  }
}

    if (wet > 0.001) {
      float3 cw = o0.xyz; float lw = dot(cw, float3(0.2126, 0.7152, 0.0722));
      cw = lerp(lw.xxx, cw, 1.0 + 0.30 * wet) * (1.0 - 0.20 * wet);
      o0.xyz = saturate(cw);
      float wt = wet * (0.55 + 0.45 * smoothstep(0.3, 1.0, relH));
      o2.y = saturate(o2.y + 0.35 * wt);
      o2.x = lerp(o2.x, 0.14, 0.65 * wt);
    }
    if (pesArt) { o2.y = saturate(o2.y + 0.12 * (1.0 - pesPaintF)); o2.x = lerp(o2.x, 0.30, 0.5); }
  }
  r0.x = dot(v4.xyz, v4.xyz);
  r0.x = rsqrt(r0.x);
  r0.xyz = v4.xyz * r0.xxx;
  if (pesV3 && Q6.w > 0.001 && !pesFar && !pesBladeM) {   // (1/11) lejos nm = 0: la normal queda la del suelo; (11/11) hojas 3D: la normal ya es la de la hoja
    // normal por hoja: la normal del juego es la de la cancha (hacia arriba): todas las hojas se iluminan igual.
    // Se le suma la normal del atlas en un marco con "a lo ancho" = la derecha de la camara proyectada, asi cada hoja
    // se ve cilindrica (clara de un lado, oscura del otro) y el sol da brillos por hoja.
    float3 N = r0.xyz;
    float3 T = float3(1.0, 0.0, 0.0) - N * N.x; T = normalize(T + float3(0.0, 0.0, 1e-4));
    float3 B = cross(N, T);
    float2 nm = (pesN.xy * 2.0 - 1.0) * clamp(Q6.w, 0.0, 1.5) * pesCf;   // (26/10 f) lejos la normal del suelo
    float nz = sqrt(saturate(1.0 - dot(nm, nm)));
    r0.xyz = normalize(T * nm.x + B * nm.y + N * nz);
  }
  // (12/11, Agustin: "el brillo del pasto al mover la camara se nota bastante mas que al estar quieto") x436 BRILLO ESTABLE: rugosidad filtrada
  // por cuanto cambia la normal entre pixeles vecinos (filtro de Tokuyoshi-Kaplanyan): donde la hoja mide 1-2 px el reflejo del sol, la cera y
  // el rocio se reparten en un lobulo mas ancho y mas bajo, el mismo promedio que deja el TAA con la camara quieta; de cerca (normal pareja
  // entre pixeles) no cambia. Con la camara en movimiento (o3.xy = px por cuadro, lo mismo que lee el TAA) el filtro sube hasta el doble:
  // el TAA ya no promedia
  { float3 dNx = ddx(r0.xyz), dNy = ddy(r0.xyz);
    float kSt = pesStK;
    if (pesV3 && kSt > 0.001) {
      // (13/11 tanda 2) x436: el movimiento de la CAMARA (fila 498 de la DLL, px por cuadro a 1080 lineas). o3.xy aca es solo el viento y la
      // pelota (no hay jugadores en este shader): de cerca el viento mueve las puntas 2-5 px y parecia un paneo. Sin la fila: como antes
      float lP = length(o3.xy); float4 T498 = IniParams.Load(int2(498,0));
      float mvP = (T498.w > 0.5) ? T498.z * g_psSystem.m_renderBuffer.y / 1080.0 : lP;
      float sig2 = 0.15915494 * kSt * (1.0 + smoothstep(0.75, 4.0, mvP));
      float kern = min(2.0 * sig2 * (dot(dNx, dNx) + dot(dNy, dNy)), 0.18 * saturate(kSt));
      // (12/11, "le falta el brillo en las puntas") hoja 3D que ya mide 2-4 px de ancho con la camara quieta: el cambio de la normal de un borde
      // al otro (la V de la hoja angosta) es forma real de la hoja, no un serrucho que el TAA vaya a promediar: el filtro le apagaba el brillo
      // de la punta y del rocio. Ahi baja al 20 % (en movimiento sigue entero)
      if (pesBladeM) kern *= 1.0 - 0.8 * smoothstep(1.5, 4.0, 1.0 / max(pesFwU, 1e-4)) * (1.0 - smoothstep(0.75, 4.0, mvP));
      float rgA = saturate(o2.x), a2 = rgA * rgA * rgA * rgA;
      o2.x = sqrt(sqrt(saturate(a2 + kern)));
    }
  }
  float pesTr = v1.w;
  if (pesNat && !pesFar) pesTr = lerp(v1.w, saturate(v1.w * lerp(1.0, 1.55 - 1.1 * pesN.w, natTr)), pesCf);   // (11/11) w409: lamina fina translucida, nervio y venas opacos
  else if (pesV3 && !pesFar) pesTr = lerp(v1.w, saturate(v1.w * (1.35 - 0.8 * pesN.w)), pesCf);   // puntas finas mas translucidas (A = grosor); lejos sin variar
  // (27/10 h) z295: la punta deja pasar mas luz a contraluz (y57 lee esto del G-buffer); la pintura tapa
  if (pesV3) pesTr = saturate(pesTr * (1.0 + (pesNat ? 0.55 : 0.35) * PESTipS(0.45, 1.0, relT, pesTipAA) * capT * pesSunV * kT * pesCfT) * (1.0 - 0.85 * pesPaintF));   // (11/11 n) natural: la punta deja pasar mas; (13/11 tanda 2) y496 solo arriba, w447 nada en la sombra
  if (pesSeK > 0.001) pesTr = saturate(pesTr * (1.0 + pesSeK * pesSunV * PESTipS(0.45, 1.0, relT, pesTipAA) * capT * (0.45 * pesSeLo - 0.25 * pesSeHi)));   // (13/11) x506
  if (pesV3) pesTr = saturate(pesTr + 0.30 * pesTxGb);   // (08/11 aa) w413: la pasada del sol tambien la enciende mas a contraluz   // (27/10 i) pintura: casi opaca (salia lima a contraluz)
  // (27/10 n) tanda 11, x329: la punta deja pasar mas luz (translucidez al G-buffer, la lee la pasada del sol): hojas encendidas a contraluz
  if (pesV3 && v4.w > 0.0) { float4 P329t = IniParams.Load(int2(329,0)); pesTr = saturate(pesTr * (1.0 + 0.6 * clamp(P329t.x, 0.0, 2.0) * PESTipS(0.3, 1.0, relT, pesTipAA) * capT * pesSunV * pesCfT * (1.0 - pesPaintF))); }   // (13/11 tanda 2) y496, w447
  // (04/11) de cerca cada hoja con su translucidez (unas finas que se encienden a contraluz, otras gruesas): mismo promedio
  if (pesV3 && v4.w > 0.0) pesTr = saturate(pesTr * lerp(1.0, 0.6 + 0.8 * frac(pesHvR * 7.37 + 0.21), nearF * pesCfN * (1.0 - pesPaintF)));
  if (pesDrop > 0.001) pesTr = saturate(pesTr + 0.25 * pesDrop);   // (27/10 p) la gota se enciende a contraluz
  if (pesV3 && v4.w > 0.0) pesTr *= lerp(0.45, 1.0, pesCf);   // (02/11 paso 4) lejos: menos translucidez (la pasada del sol encendia las hojas lejanas a contraluz)
  r0.w = pesTr * 63 + 0.5;
  r1.x = abs(r0.x) + abs(r0.y);
  r1.x = r1.x + abs(r0.z);
  r0.xy = r0.xy / r1.xx;
  r0.z = cmp(0 < -r0.z);
  r1.xy = float2(1,1) + -abs(r0.yx);
  r1.zw = cmp(r0.xy >= float2(0,0));
  r1.zw = r1.zw ? float2(1,1) : float2(-1,-1);
  r1.xy = r1.xy * r1.zw;
  r0.xy = r0.zz ? r0.xy : r1.xy;
  r0.xy = r0.xy * float2(0.5,0.5) + float2(0.5,0.5);
  r0.xy = float2(2047,2047) * r0.xy;
  r0.xyw = (uint3)r0.xyw;
  r1.xy = (uint2)r0.xy >> int2(1,1);
  bitmask.x = ((~(-1 << 1)) << 1) & 0xffffffff;  r0.x = (((uint)r0.x << 1) & bitmask.x) | ((uint)0 & ~bitmask.x);
  bitmask.x = ((~(-1 << 1)) << 0) & 0xffffffff;  r0.x = (((uint)r0.y << 0) & bitmask.x) | ((uint)r0.x & ~bitmask.x);
  r1.xy = (uint2)r1.xy;
  r1.w = (uint)r0.x;
  o1.xyw = float3(0.000977517106,0.000977517106,0.333333343) * r1.xyw;
  bitmask.x = ((~(-1 << 28)) << 4) & 0xffffffff;  r0.x = (((uint)r0.w << 4) & bitmask.x) | ((uint)3 & ~bitmask.x);
  r0.x = (uint)r0.x;
  o1.z = 0.000977517106 * r0.x;
  o0.w = 1.0;   // (26/10 g) el alfa de RT0 no se escribe con la mezcla lejana (queda el de la base); la mezcla va al final con la copia del suelo
  o2.z = 0;
  o2.w = v5.z;
  o3.zw = float2(0,0);
  o4.xyzw = float4(0,0,0,1);
  // (11/11 j, captura 15.21 de Agustin: "ves el color de las sombras en las partes cortas") con y290 = 0 ("Propia") las hojas escribian RT4 = 0
  // y perdian la luz difusa / horneada del estadio: en la sombra el pasto quedaba gris oscuro contra las puntas lima al sol (con y290 = 1 la
  // tenian, pero delante de la pelota quedaba la de la pelota). Ahora la DLL enlaza en t44 una copia del RT4 de la base y la hoja escribe la luz
  // del suelo en su RAIZ (v7.zw); si la raiz cae fuera de pantalla o tapada por algo mas oscuro, la del suelo bajo su pixel (los recortes que
  // vuelan no tienen raiz: la de su pixel). y290 = 1 vuelve a lo de antes (RT4 sin escribir)
  { uint bw4 = 0, bh4 = 0; PESBaseRT4.GetDimensions(bw4, bh4);
    if (bw4 > 0 && bh4 > 0) {
      float2 rb4 = float2(bw4, bh4);
      float3 lP = PESBaseRT4.Load(int3(int2(clamp(v0.xy, 0.0, rb4 - 1.0)), 0)).xyz, lR = lP;
      if (all(abs(v7.zw) < 1.0)) lR = PESBaseRT4.Load(int3(int2(clamp((v7.zw * float2(0.5, -0.5) + 0.5) * rb4, 0.0, rb4 - 1.0)), 0)).xyz;
      const float3 LW4 = float3(0.2126, 0.7152, 0.0722);
      o4.xyz = max(dot(lR, LW4) >= 0.6 * dot(lP, LW4) ? lR : lP, 0.0);
    } }
    {
    // === PES21 mod: hojas de pasto 3D con variacion por hoja ===
    float4 G = IniParams.Load(int2(10,0));
    if (G.y > 0 && !pesFar) {
      float n = lerp(0.5, bh21(floor(v1.xy * 96.0)), pesCf);   // (26/10 f) lejos sin el valor al azar por tarjeta
      float3 c = o0.xyz * (0.90 + 0.18 * n * G.z);
      c = lerp(c, dot(c, float3(0.2126,0.7152,0.0722)) * float3(1.12, 1.04, 0.74), saturate(n * 3.0 - 2.2) * 0.18 * G.z * (1.0 - pesPaintF));
      o0.xyz = c;
    }
    // (27/10 h) y295: variacion POR HOJA (v8.z, el VS la manda por tarjeta; el n de arriba se repite igual en todas las tarjetas): cada hoja con
    // su claridad (+-14%) y su tono (unas mas amarillas, otras mas azuladas). Asi de cerca y a media distancia se distingue una hoja de la de al
    // lado en vez de un verde parejo. Lejos (pesCfN) y en las hojas pintadas no. (27/10 j) fuera del "if (G.y > 0)": con y10 = 0 (el ini de
    // Agustin) no corria y todas las hojas salian iguales; solo con v3
    if (pesV3 && kC > 0.001 && !pesFar) {
      float kV = 0.5 * kC * kVar * pesCfN * (1.0 - pesPaintF);   // (2/11) y325 escala la variacion
      float hv = pesHvR;
      float3 c = o0.xyz * lerp(1.0, 0.86 + 0.28 * hv, kV);
      float3 tn = lerp(float3(0.94, 1.0, 1.08), float3(1.08, 1.0, 0.90), frac(hv * 5.17 + 0.31));
      c *= lerp(float3(1.0, 1.0, 1.0), tn, kV);
      // (2/11) y325 > 0.5: una de cada ~6 hojas tira a seca (paja) de cerca: verde, amarillo y seco mezclados como en el pasto real
      float dryB = step(0.84, frac(hv * 3.77 + 0.17)) * saturate(kVar - 0.5) * pesCfN * (1.0 - pesPaintF);
      c = PESDryTint(c, 0.35 * dryB);
      o0.xyz = saturate(c);
    }
  }
  // (12/11, Agustin con la captura vieja de pasto largo: "verde medio en la base y solo la punta lima") z439 BASE MENOS SATURADA: a lo largo
  // de la hoja natural, la parte de abajo pierde saturacion (verde medio, mismo brillo) y el tramo de la punta (en mm, relT) la gana un poco:
  // el lima queda solo en las puntas. Lo que se ve desde la camara es sobre todo la parte alta de las hojas, asi que el color promedio
  // de la cancha casi no cambia (y lejos, donde la hoja toma el color del suelo, no corre). 0 = como antes
  { float kDs = pesNat ? clamp(IniParams.Load(int2(439,0)).z, 0.0, 1.0) : 0.0;
    if (kDs > 0.001 && !pesFar) {
      float lmD = dot(o0.xyz, float3(0.2126, 0.7152, 0.0722));
      float sD = lerp(1.0 - 0.38 * kDs, 1.0 + 0.14 * kDs, smoothstep(0.10, 0.85, relT));
      sD = lerp(1.0, sD, pesCfN * (1.0 - pesPaintF));
      o0.xyz = saturate(lmD + (o0.xyz - lmD) * sD);
    } }

  // capas de hojas gana la de adelante (antes cada capa volvia a mezclar sobre la anterior y el resultado dependia del orden). z59 sin mezclar
  // (27/10 l) tanda 9a, z299: las hojas pintadas por la linea toman el color del suelo (la linea de la base, nitida) hasta que la hoja mide en
  // pantalla (rampa pesCfN, desde z291): a media distancia la linea se veia punteada por las hojas blancas sombreadas
  // (03/11) y330: color del suelo en las hojas de CERCA. En canchas de parche (id 9, 03/11) el color del pasto del juego (v3) y su mascara de
  // lineas no coinciden con la base: las hojas quedaban a 0.63x del original y tapaban la cal. La hoja toma el color de la base dibujada
  // detras (copia de RT0: color, franjas de corte, cal, desgaste de la cancha) y conserva su propio sombreado (raiz, claridad, puntas, seco,
  // matas, lado al sol) como un factor relativo a su color plano. 0 = como antes, 1 = todo el color del suelo
  // (04/11) el suelo se lee en la raiz de la hoja (v7.zw del VS): en el pixel de la hoja la copia de RT0 tiene lo que esta DETRAS (botin,
  // pelota, piernas) y la hoja delante de un botin tomaba su color: parecia transparente. La raiz esta en el suelo, delante del jugador
  // (04/11 b) dos muestras del suelo: la de la raiz y la del pixel. La raiz puede quedar TAPADA por el botin (hojas detras del botin cuyas
  // puntas asoman por encima: tomaban el color del botin, pinches rosas / blancos) y el pixel puede tener el botin DETRAS (hoja delante: se
  // veia transparente). Se usa la que mas se parece al color plano de la propia hoja (las dos son suelo cuando no hay nadie: misma eleccion
  // que antes, con preferencia por la raiz)
  float3 pesGnd = 0;
  if (pesFB && v5.w < 0.5) {
    float3 gP = PESBaseRT0.Load(int3(v0.xy, 0)).xyz, gR = gP;
    if (all(abs(v7.zw) < 1.0)) {
      float2 rb = g_psSystem.m_renderBuffer.xy;
      if (rb.x < 1.0) { uint bw, bh; PESBaseRT0.GetDimensions(bw, bh); rb = float2(bw, bh); }
      gR = PESBaseRT0.Load(int3(int2(clamp((v7.zw * float2(0.5, -0.5) + 0.5) * rb, 0.0, rb - 1.0)), 0)).xyz;
    }
    float3 fc = saturate(v1.zzz * v3.xyz);
    { float3 cg = fc; PESGrassColorGradeApply(pesCG, cg); fc = saturate(cg); }

    float3 fn = fc / max(dot(fc, float3(1,1,1)), 0.03), rn = gR / max(dot(gR, float3(1,1,1)), 0.03), pn = gP / max(dot(gP, float3(1,1,1)), 0.03);
    float lf = dot(fc, float3(0.333,0.333,0.333)), dR = dot(rn - fn, rn - fn) + 0.05 * abs(dot(gR, float3(0.333,0.333,0.333)) - lf), dP = dot(pn - fn, pn - fn) + 0.05 * abs(dot(gP, float3(0.333,0.333,0.333)) - lf);
    pesGnd = (dR <= dP * 1.5 + 0.002) ? gR : gP;
  }
  float4 P330 = IniParams.Load(int2(330,0));
  if (pesFB && v5.w < 0.5 && !pesFar && P330.y > 0.001) {
    float3 gnd = pesGnd;
    float3 flatC = saturate(v1.zzz * v3.xyz * pesNatG);
    if (pesNat) { PESNatX29(flatC); float3 cg = flatC / max(pesNatG, 1e-3); PESGrassColorGradeApply(pesCG, cg); flatC = saturate(cg * pesNatG); }   // (12/11 20:50) como la hoja: color antes de la ganancia; (13/11 tanda 2) y x29 como la hoja
    else { float3 cg = flatC; PESGrassColorGradeApply(pesCG, cg); flatC = saturate(cg); }

    o0.xyz = lerp(o0.xyz, saturate(gnd * rel), saturate(P330.y));
  }

  if (pesFB && v5.w < 0.5) o0.xyz = lerp(pesGnd, o0.xyz, pesCfB);

  // cortas reciben menos luz que las altas que asoman; mas en la raiz que en la punta. Va DESPUES de la mezcla con el suelo para que a media
  // distancia (30-110 px/m), donde la hoja ya toma el color del suelo, el pasto siga leyendose como matas de 12 cm (claras / oscuras,
  // coherentes) y no como un verde liso con ruido por pixel. Se apaga por debajo de 15-40 px/m (ahi titilaria)

  if (pesV3 && v4.w > 0.0 && P328.z > 0.001 && v8.y < 0.999) {
    float kA = clamp(P328.z, 0.0, 2.0) * smoothstep(15.0, 40.0, v4.w) * (1.0 - 0.5 * pesPaintF) * lerp(0.35, 1.0, pesCf);   // (02/11 paso 4) lejos se funde con el suelo
    // (04/11) la oclusion de la mata promedia ~0.78: de cerca y a media distancia toda la cancha bajaba ~20 % contra lejos. Ahora reparte:
    // matas bajas y bordes mas oscuros, las altas un poco mas claras, con el mismo promedio (solo hojas del mod: las del juego llevan 1)

    o0.xyz = saturate(o0.xyz * ao);
  }
  // (27/10 p) tanda 12, z334 sombra por hoja de cerca: cada hoja tiene su orientacion (al azar, fija por hoja: v8.z) y segun mire al sol o le de
  // la espalda queda mas clara o mas oscura (en la foto, de cerca, hay hojas encendidas y hojas en sombra una al lado de la otra; la luz del

  if (pesV3 && nearF > 0.001 && kS > 0.001) {

    float expo = lerp(1.0, lerp(0.82, 1.16, face), kS * nearF * sunny * (0.5 + 0.5 * saturate(S.y)) * (1.0 - 0.7 * pesPaintF));
    o0.xyz = saturate(o0.xyz * expo);
  }
  if (pesV3) {   // depuracion w107: 1 = uv del atlas (rojo u, verde v), 2 = desgaste (rojo seco, verde oclusion de la mata), 3 = normal; (12/11 23:45) 5 = w447 (blanco = al sol, negro = sombra)
    int dbg = (int)(Q7.w + 0.5);
    if (dbg == 1) o0.xyz = float3(v1.x, v1.y, 0.0);
    else if (dbg == 2) o0.xyz = float3(v8.x, v8.y, 0.0);
    else if (dbg == 3) o0.xyz = pesN.xyz;
    else if (dbg == 4) o0.xyz = float3(1.0 - pesCf, pesCf, pesFB ? 1.0 : 0.0);   // (26/10 f) rojo = lejos (color del suelo), verde = cerca; azul = mezcla activa

    if (dbg >= 1 && dbg <= 5) o0.w = 1.0;
  }
