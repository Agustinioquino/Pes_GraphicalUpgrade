// PES21 DOF v2 (16/10): las capas desenfocadas (cercana y lejana, a media resolucion) se ampliaban con filtro lineal
// -> bordes en bloques. Ahora con B-spline cubico (4 lecturas), suave y sin escalones.
float4 PESBicubic(Texture2D<float4> tex, float2 uv, float2 hi)
{
  uint tw, th; tex.GetDimensions(tw, th);
  float2 ts = float2(max(tw, 1u), max(th, 1u));
  float2 st = uv * ts - 0.5;
  float2 i = floor(st);
  float2 f = st - i;
  float2 f2 = f * f, f3 = f2 * f;
  float2 w0 = (-f3 + 3.0 * f2 - 3.0 * f + 1.0) / 6.0;
  float2 w1 = (3.0 * f3 - 6.0 * f2 + 4.0) / 6.0;
  float2 w2 = (-3.0 * f3 + 3.0 * f2 + 3.0 * f + 1.0) / 6.0;
  float2 w3 = f3 / 6.0;
  float2 g0 = w0 + w1, g1 = w2 + w3;
  float2 h0 = (i - 1.0 + w1 / max(g0, 1e-5) + 0.5) / ts;
  float2 h1 = (i + 1.0 + w3 / max(g1, 1e-5) + 0.5) / ts;
  float4 a = tex.SampleLevel(g_samplerLinear_Clamp_s, clamp(float2(h0.x, h0.y), 0, hi), 0);
  float4 b = tex.SampleLevel(g_samplerLinear_Clamp_s, clamp(float2(h1.x, h0.y), 0, hi), 0);
  float4 c = tex.SampleLevel(g_samplerLinear_Clamp_s, clamp(float2(h0.x, h1.y), 0, hi), 0);
  float4 d = tex.SampleLevel(g_samplerLinear_Clamp_s, clamp(float2(h1.x, h1.y), 0, hi), 0);
  return g0.y * (g0.x * a + g1.x * b) + g1.y * (g0.x * c + g1.x * d);
}


// PES21 (20/10) bokeh real (137): desenfoque de fondo calculado a resolucion completa con un disco de lente en vez de las
// capas chicas del juego (480x270 / 240x135 ampliadas: el fondo quedaba como una mancha lavada y con halos alrededor de
// los jugadores). 48 muestras en espiral de Vogel dentro del radio de cada pixel; cada muestra solo cuenta si su propio
// circulo de confusion llega hasta el pixel (dispersion como recoleccion): el jugador enfocado (radio 0) nunca se mezcla
// con el fondo -> bordes limpios, y las luces fuertes del estadio forman discos de bokeh porque la imagen es HDR.
// (02/11) bokeh v2: con 48 muestras fijas el disco quedaba con huecos (a 12 px de radio una muestra cada ~3 px, a 4K cada
// ~6 px): el fondo mostraba el dibujo de la espiral y las luces fuertes salian como puntos sueltos que titilaban.
// Ahora: cantidad de muestras segun el radio de cada pixel (32..160, una cada ~2.5 px), cada lectura cae entre 4 texeles
// (el filtro lineal promedia 2x2: el disco se cubre entero), la espiral gira distinto en cada pixel (ruido fino en vez de
// anillos) y cada muestra pesa 1/(1+brillo/4): las luces siguen formando discos pero sin puntos sueltos que titilan;
// z137 realza las luces fuertes igual que antes.
float3 PESBokehFar(float2 uv, float2 pos, float4 c0, float maxPx, float boost) {
  float R0 = saturate((c0.w - 0.5) * 2.0) * maxPx;
  if (R0 < 0.6) return c0.xyz;
  float2 px = g_psSystem.m_renderBuffer.zw;
  int n = (int)clamp(ceil(R0 * R0 * 0.5), 32.0, 160.0);
  float rot = 6.2831853 * frac(52.9829189 * frac(dot(pos, float2(0.06711056, 0.00583715))));   // ruido de gradiente intercalado
  float3 col0 = min(c0.xyz, 64.0);
  float w0 = 1.0 / (1.0 + 0.25 * dot(col0, float3(0.2126, 0.7152, 0.0722)));
  float3 acc = col0 * w0; float wsum = w0;
  [loop] for (int k = 0; k < n; k++) {
    float r = sqrt((k + 0.5) / n) * R0;
    float a = k * 2.39996323 + rot;
    float2 o = float2(cos(a), sin(a)) * r + 0.5;              // +0.5 px: la lectura lineal promedia 2x2 texeles
    float4 sm = inImage.SampleLevel(g_samplerLinear_Clamp_s, saturate(uv + o * px), 0);
    float Rs = saturate((sm.w - 0.5) * 2.0) * maxPx;
    float w = saturate(Rs - r + 1.0);
    float3 col = min(sm.xyz, 64.0);
    float l = dot(col, float3(0.2126, 0.7152, 0.0722));
    w *= (1.0 + boost * saturate(l - 1.0)) / (1.0 + 0.25 * l);   // z137 realce de luces; 1/(1+l/4): sin puntos que titilan
    acc += col * w; wsum += w;
  }
  return acc / max(wsum, 1e-5);
}

void main(
  float4 v0 : SV_Position0,
  float2 v1 : TEXCOORD0,
  out float4 o0 : SV_Target0)
{
  float4 r0,r1,r2,r3;
  uint4 bitmask, uiDest;
  float4 fDest;

  r0.xyzw = g_psSystem.m_renderBuffer.zwzw * v0.xyxy;
  r1.xyzw = inImage.Sample(g_samplerPoint_Clamp_s, r0.zw).xyzw;
  r0.xyzw = max(float4(0,0,0,0), r0.xyzw);
  r0.xyzw = min(g_psObject.m_localParam[2].zwxy, r0.xyzw);
  r1.w = -0.5 + r1.w;
  r1.w = saturate(r1.w + r1.w);
  r1.w = g_psObject.m_localParam[0].x * r1.w;
  r1.w = saturate(g_psObject.m_localParam[0].z * r1.w);
  r1.w = log2(r1.w);
  r1.w = 0.174999997 * r1.w;
  r1.w = exp2(r1.w);
  r1.w = -0.850000024 + r1.w;
  r1.w = 6.66666794 * r1.w;
  r1.w = max(0, r1.w);
  r2.x = r1.w * -2 + 3;
  r1.w = r1.w * r1.w;
  r1.w = r2.x * r1.w;
  r2.x = log2(r1.w);
  float4 DFc = IniParams.Load(int2(41,0));
  if (DFc.w > 0.5) {
    r2.yzw = PESBicubic(inImageFar, r0.zw, g_psObject.m_localParam[2].xy).xyz;
    r0.xyzw = PESBicubic(inImageNear, r0.xy, g_psObject.m_localParam[2].zw);
  } else {
    r2.yzw = inImageFar.Sample(g_samplerLinear_Clamp_s, r0.zw).xyz;
    r0.xyzw = inImageNear.Sample(g_samplerLinear_Clamp_s, r0.xy).xyzw;
  }
  float4 BK = IniParams.Load(int2(137,0));
  if (BK.x > 0.5) {
    float maxPx = clamp(BK.y > 0.5 ? BK.y : 12.0, 2.0, 32.0) * (1.0 / max(g_psSystem.m_renderBuffer.w * 1080.0, 1e-3));
    float2 uvF = g_psSystem.m_renderBuffer.zw * v0.xy;
    float3 g = PESBokehFar(uvF, v0.xy, inImage.SampleLevel(g_samplerPoint_Clamp_s, uvF, 0), maxPx, max(BK.z, 0.0));
    r1.xyz = g; r2.yzw = g;
  }
  r3.x = dot(float3(0.212599993,0.715200007,0.0722000003), r2.yzw);
  r3.y = saturate(1 + -r3.x);
  r3.x = saturate(3 * r3.x);
  r3.y = r3.y * r3.y;
  r3.y = max(0.5, r3.y);
  r3.z = dot(float3(0.212599993,0.715200007,0.0722000003), r1.xyz);
  r3.z = saturate(1 + -r3.z);
  r3.z = max(0.25, r3.z);
  r3.y = min(r3.z, r3.y);
  r2.x = r3.y * r2.x;
  r2.x = exp2(r2.x);
  r3.yzw = saturate(r1.xyz);
  r3.yzw = r3.yzw + -r1.xyz;
  r1.xyz = r2.xxx * r3.yzw + r1.xyz;
  r2.yzw = r2.yzw + -r1.xyz;
  r1.xyz = r2.xxx * r2.yzw + r1.xyz;
  r0.xyz = -r1.xyz + r0.xyz;
  r0.w = saturate(g_psObject.m_localParam[0].y * r0.w);
  r1.w = r0.w * g_psObject.m_localParam[0].w + r1.w;
  r0.w = g_psObject.m_localParam[0].w * r0.w;
  r1.w = saturate(-1 + r1.w);
  r0.w = -r1.w * r3.x + r0.w;
  o0.xyz = r0.www * r0.xyz + r1.xyz;
  o0.w = 1;
  return;
}
